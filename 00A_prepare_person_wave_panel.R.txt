# =============================================================================
# BHPS + UKHLS
# FIRST-MARRIAGE PRE-TRANSITION STUDY
# T-5 TO T0 ELIGIBILITY / COUNTING DIAGNOSTIC
#
# PURPOSE
# -------
# Reframe the study around marriage rather than requiring respondents to be
# single throughout the five pre-event waves.
#
# T0 is the first interview wave in the focal event window at which the person
# is observed as married. The actual marriage occurred between T-1 and T0.
#
# The key window-based eligibility rule is:
#
#     T-5:T-1 = any OBSERVED NON-MARRIED marital status
#     T0      = married
#
# Thus, prior cohabitation is explicitly allowed. The pre-marriage window may
# include:
#     - single / never married
#     - living as a couple / cohabiting
#     - registered civil partnership
#     - other non-married statuses
#
# Complete marital-status observation at all six waves T-5:T0 is required for
# the eligibility diagnostic because the study explicitly characterises the
# pre-marriage marital-status pathway.
#
#
# THREE NESTED MARRIAGE DEFINITIONS ARE REPORTED
# -----------------------------------------------
# A. WINDOW-ONLY MARRIAGE TRANSITION
#       non-married at every wave T-5:T-1 -> married at T0.
#
# B. FIRST OBSERVED MARRIAGE  [PLANNED PRIMARY]
#       definition A +
#       no earlier observed "married" status anywhere in the combined
#       BHPS+UKHLS history before T0.
#
# C. STRICT FIRST-MARRIAGE PROXY  [DIAGNOSTIC / SENSITIVITY]
#       definition B +
#       no earlier observed status label indicating a prior marriage
#       (e.g., divorced, widowed, or marital separation).
#
# Definition C is more conservative because panel observation may start after
# a person's true first marriage. It is reported so that the cost of excluding
# likely remarriages can be assessed empirically rather than assumed.
#
#
# EXPLICIT PRE-MARRIAGE STATUS OUTPUTS
# ------------------------------------
# For first-observed marriers, the script reports at T-5,...,T-1:
#
#     n and % single
#     n and % cohabiting
#     n and % in civil partnership
#     n and % other non-married
#
# It also reports:
#
#     - whether each future marrier ever cohabited in T-5:T-1;
#     - whether they cohabited at T-1;
#     - number of cohabiting pre-waves;
#     - most common T-5:T-1 status sequences;
#     - first observed cohabiting event-time within the five-wave window.
#
#
# CONTEMPORANEOUS UNMARRIED RISK-SET CONTROLS
# --------------------------------------------
# The script constructs two control definitions for later analysis:
#
# 1. FIRST-MARRIAGE RISK-SET CONTROLS  [PLANNED PRIMARY]
#       non-married at T-5:T0 in the same survey/event-wave cohort and no
#       previously observed marriage by that focal T0.
#
# 2. ALL UNMARRIED RISK-SET CONTROLS  [SENSITIVITY]
#       non-married at T-5:T0 regardless of earlier marriage history.
#
# Controls may marry after the focal T0 and may contribute to multiple risk
# sets. No post-T0 condition is imposed.
#
#
# OUTCOMES
# --------
# GHQ-12:
#     BHPS + UKHLS
#
# Life satisfaction:
#     BHPS variable lfsato
#     UKHLS variable sclfsato
#
# Outcome completeness NEVER determines marital-status/event eligibility.
# Outcome availability is diagnosed separately.
#
#
# IMPORTANT INTERPRETATION
# ------------------------
# This diagnostic supports a study of WELL-BEING TRAJECTORIES LEADING UP TO
# MARRIAGE. It does not by itself identify a causal effect of legal marriage.
# Changes before T0 may reflect relationship formation, cohabitation,
# engagement, marriage planning, selection, or other processes.
#
# =============================================================================

read_data_file <- function(
    path
) {

  ext <- tolower(
    tools::file_ext(
      path
    )
  )

  if (ext == "dta") {
    return(
      haven::read_dta(
        path
      )
    )
  }

  if (ext == "sav") {
    return(
      haven::read_sav(
        path
      )
    )
  }

  stop(
    "Unsupported file extension: ",
    ext
  )
}


# BHPS files may use:
#   a_indresp.dta
# or:
#   ba_indresp.dta
#
# UKHLS files normally use:
#   a_indresp.dta, b_indresp.dta, ...

find_indresp_file <- function(
    root,
    survey,
    wave_letter
) {

  survey <- match.arg(
    survey,
    c(
      "BHPS",
      "UKHLS"
    )
  )

  prefixes <- if (
    survey == "BHPS"
  ) {
    c(
      wave_letter,
      paste0(
        "b",
        wave_letter
      )
    )
  } else {
    wave_letter
  }

  patterns <- paste0(
    "^",
    prefixes,
    "_indresp\\.(dta|sav)$"
  )

  all_files <- list.files(
    root,
    recursive = TRUE,
    full.names = TRUE,
    ignore.case = TRUE
  )

  hits <- all_files[
    Reduce(
      `|`,
      lapply(
        patterns,
        function(p) {
          grepl(
            p,
            basename(
              all_files
            ),
            ignore.case = TRUE
          )
        }
      )
    )
  ]

  if (length(hits) == 0L) {
    stop(
      "No individual-response file found for ",
      survey,
      " wave ",
      wave_letter,
      " under:\n",
      root
    )
  }

  if (length(hits) > 1L) {
    stop(
      "More than one possible individual-response file found for ",
      survey,
      " wave ",
      wave_letter,
      ":\n",
      paste(
        hits,
        collapse = "\n"
      )
    )
  }

  hits
}


prefix_from_indresp_filename <- function(
    path
) {

  nm <- basename(
    path
  )

  stringr::str_match(
    nm,
    "^(.+?)_indresp\\.(dta|sav)$"
  )[, 2]
}


find_variable <- function(
    dat,
    file_prefix,
    basenames,
    required = TRUE
) {

  candidates <- unique(
    unlist(
      lapply(
        basenames,
        function(v) {
          c(
            paste0(
              file_prefix,
              "_",
              v
            ),
            v
          )
        }
      )
    )
  )

  hit <- candidates[
    candidates %in%
      names(
        dat
      )
  ]

  if (length(hit) == 0L) {

    if (required) {
      stop(
        "Could not find any of these variables:\n",
        paste(
          candidates,
          collapse = ", "
        )
      )
    }

    return(
      NA_character_
    )
  }

  hit[1L]
}


clean_numeric <- function(
    x
) {

  z <- suppressWarnings(
    as.numeric(
      x
    )
  )

  z[
    !is.na(z) &
      z < 0
  ] <- NA_real_

  z
}


label_or_numeric_character <- function(
    x
) {

  if (
    inherits(
      x,
      "haven_labelled"
    )
  ) {

    lab <- as.character(
      haven::as_factor(
        x,
        levels = "labels"
      )
    )

    if (
      any(
        !is.na(
          lab
        )
      )
    ) {
      return(
        lab
      )
    }
  }

  as.character(
    x
  )
}

# 4. MARITAL-STATUS HARMONISATION
# =============================================================================

clean_label <- function(
    x
) {

  z <- as.character(
    haven::as_factor(
      x,
      levels = "labels"
    )
  )

  z |>
    stringr::str_squish() |>
    stringr::str_to_lower() |>
    stringr::str_replace_all(
      "[’`]",
      "'"
    )
}


is_survey_missing_label <- function(
    x
) {

  is.na(x) |
    stringr::str_detect(
      x,
      paste0(
        "^-[0-9]+$|",
        "^missing$|",
        "inapplicable|",
        "not applicable|",
        "not asked|",
        "^proxy$|",
        "refus|",
        "don't know|",
        "dont know|",
        "unable|",
        "no answer|",
        "child under|",
        "under 16"
      )
    )
}


classify_status <- function(
    x,
    survey
) {

  survey <- match.arg(
    survey,
    c(
      "BHPS",
      "UKHLS"
    )
  )

  raw_label <- as.character(
    haven::as_factor(
      x,
      levels = "labels"
    )
  )

  clean <- clean_label(
    x
  )

  missing_flag <- is_survey_missing_label(
    clean
  )

  status <- if (
    survey == "BHPS"
  ) {

    dplyr::case_when(

      missing_flag ~
        NA_character_,

      clean == "married" ~
        "married",

      clean == "living as couple" ~
        "cohab",

      clean == "civil partnership" ~
        "civil_partner",

      clean == "never married" ~
        "single",

      TRUE ~
        "other"
    )

  } else {

    dplyr::case_when(

      missing_flag ~
        NA_character_,

      clean == "married" ~
        "married",

      clean == "living as couple" ~
        "cohab",

      stringr::str_detect(
        clean,
        "^in a registered.*civil partnership$"
      ) ~
        "civil_partner",

      stringr::str_detect(
        clean,
        "^single.*never married"
      ) ~
        "single",

      TRUE ~
        "other"
    )
  }

  tibble(
    raw_status_label =
      raw_label,
    status =
      status
  )
}


# =============================================================================
# 5. AGE / SEX HARMONISATION
# =============================================================================

clean_sex <- function(
    x
) {

  if (is.null(x)) {
    return(
      character(0)
    )
  }

  lab <- label_or_numeric_character(
    x
  ) |>
    stringr::str_squish() |>
    stringr::str_to_lower()

  num <- suppressWarnings(
    as.numeric(
      x
    )
  )

  dplyr::case_when(

    stringr::str_detect(
      lab,
      "^male$|^man$"
    ) ~
      "male",

    stringr::str_detect(
      lab,
      "^female$|^woman$"
    ) ~
      "female",

    !is.na(num) &
      num == 1 ~
      "male",

    !is.na(num) &
      num == 2 ~
      "female",

    TRUE ~
      NA_character_
  )
}


clean_age <- function(
    x
) {

  z <- clean_numeric(
    x
  )

  # Conservative plausibility bounds.
  z[
    !is.na(z) &
      (
        z < 16 |
          z > 100
      )
  ] <- NA_real_

  z
}


# =============================================================================
# 6. READ ONE BHPS WAVE
# =============================================================================

read_bhps_wave <- function(
    wave_num
) {

  wave_letter <- letters[
    wave_num
  ]

  f <- find_indresp_file(
    root = bhps_dir,
    survey = "BHPS",
    wave_letter = wave_letter
  )

  file_prefix <- prefix_from_indresp_filename(
    f
  )

  message(
    "Reading BHPS wave ",
    wave_num,
    " (",
    wave_letter,
    "; prefix ",
    file_prefix,
    "): ",
    basename(
      f
    )
  )

  dat <- read_data_file(
    f
  )

  pidp_var <- find_variable(
    dat,
    file_prefix,
    "pidp"
  )

  mastat_var <- find_variable(
    dat,
    file_prefix,
    "mastat"
  )

  ghq_var <- find_variable(
    dat,
    file_prefix,
    "scghq1_dv",
    required = FALSE
  )

  life_var <- find_variable(
    dat,
    file_prefix,
    "lfsato",
    required = FALSE
  )

  age_var <- find_variable(
    dat,
    file_prefix,
    c(
      "age_dv",
      "age"
    ),
    required = FALSE
  )

  sex_var <- find_variable(
    dat,
    file_prefix,
    c(
      "sex_dv",
      "sex"
    ),
    required = FALSE
  )

  st <- classify_status(
    dat[[mastat_var]],
    survey = "BHPS"
  )

  n <- nrow(
    dat
  )

  tibble(
    pidp =
      dat[[pidp_var]],

    survey =
      "BHPS",

    wave_num =
      wave_num,

    wave_letter =
      wave_letter,

    file_prefix =
      file_prefix,

    raw_status_label =
      st$raw_status_label,

    status =
      st$status,

    ghq12 =
      if (
        !is.na(
          ghq_var
        )
      ) {
        clean_numeric(
          dat[[ghq_var]]
        )
      } else {
        rep(
          NA_real_,
          n
        )
      },

    life_sat =
      if (
        !is.na(
          life_var
        )
      ) {
        clean_numeric(
          dat[[life_var]]
        )
      } else {
        rep(
          NA_real_,
          n
        )
      },

    age =
      if (
        !is.na(
          age_var
        )
      ) {
        clean_age(
          dat[[age_var]]
        )
      } else {
        rep(
          NA_real_,
          n
        )
      },

    sex =
      if (
        !is.na(
          sex_var
        )
      ) {
        clean_sex(
          dat[[sex_var]]
        )
      } else {
        rep(
          NA_character_,
          n
        )
      },

    ghq_var_found =
      ifelse(
        is.na(
          ghq_var
        ),
        NA_character_,
        ghq_var
      ),

    life_var_found =
      ifelse(
        is.na(
          life_var
        ),
        NA_character_,
        life_var
      ),

    age_var_found =
      ifelse(
        is.na(
          age_var
        ),
        NA_character_,
        age_var
      ),

    sex_var_found =
      ifelse(
        is.na(
          sex_var
        ),
        NA_character_,
        sex_var
      )
  )
}

make_status_wide <- function(
    dat,
    n_waves
) {

  out <- dat |>
    select(
      pidp,
      wave_num,
      status
    ) |>
    pivot_wider(
      names_from = wave_num,
      values_from = status,
      names_prefix = "w"
    )

  needed <- paste0(
    "w",
    1:n_waves
  )

  for (v in needed) {
    if (!v %in% names(out)) {
      out[[v]] <- NA_character_
    }
  }

  out |>
    select(
      pidp,
      all_of(
        needed
      )
    )
}


bhps_status_wide <- make_status_wide(
  bhps_long,
  N_BHPS
)

ukhls_status_wide <- make_status_wide(
  ukhls_long,
  N_UKHLS

row_all_equal <- function(
    dat,
    cols,
    allowed_values
) {

  z <- as.matrix(
    dat[
      ,
      cols,
      drop = FALSE
    ]
  )

  rowSums(
    !is.na(
      z
    ) &
      z %in%
      allowed_values
  ) ==
    length(
      cols
    )
}


row_all_nonmissing <- function(
    dat,
    cols
) {

  z <- as.matrix(
    dat[
      ,
      cols,
      drop = FALSE
    ]
  )

  rowSums(
    !is.na(
      z
    )
  ) ==
    length(
      cols
    )

event_time_order <- function(
    survey,
    wave_num
) {

  ifelse(
    survey == "BHPS",
    wave_num,
    N_BHPS + wave_num
  )
}


# "Prior marriage evidence" is deliberately conservative.
#
# We flag:
#   - observed married status;
#   - divorced;
#   - widowed;
#   - separated, unless the label clearly refers to civil partnership.
#
# Former/dissolved civil partnership alone is NOT treated as evidence of
# prior marriage.

marriage_history_evidence <- function(
    status,
    raw_status_label
) {

  clean <- raw_status_label |>
    stringr::str_squish() |>
    stringr::str_to_lower()

  observed_married <-
    !is.na(status) &
    status == "married"

  divorced_or_widowed <-
    !is.na(clean) &
    stringr::str_detect(
      clean,
      "divorc|widow"
    )

  marital_separation <-
    !is.na(clean) &
    stringr::str_detect(
      clean,
      "separat"
    ) &
    !stringr::str_detect(
      clean,
      "civil partner|civil partnership"
    )

  observed_married |
    divorced_or_widowed |
    marital_separation
}


combined_history <- combined_long |>

# =============================================================================
# 15. BUILD WINDOW-BASED MARRIAGE CANDIDATES
# =============================================================================

row_all_nonmarried <- function(
    dat,
    cols
) {

  z <- as.matrix(
    dat[
      ,
      cols,
      drop = FALSE
    ]
  )

  rowSums(
    !is.na(
      z
    ) &
      z != "married"
  ) ==
    length(
      cols
    )
}


build_marriage_candidates <- function(
    status_wide,
    survey_name,
    candidate_t0
) {

  purrr::map_dfr(
    candidate_t0,
    function(
        t0
    ) {

      pre_cols <- paste0(
        "w",
        (t0 - 5L):(t0 - 1L)
      )

      full_cols <- paste0(
        "w",
        (t0 - 5L):t0
      )

      missing_cols <- setdiff(
        full_cols,
        names(
          status_wide
        )
      )

      if (length(
        missing_cols
      ) > 0L) {
        stop(
          "Missing wave columns for ",
          survey_name,
          " T0=",
          t0,
          ": ",
          paste(
            missing_cols,
            collapse = ", "
          )
        )
      }

      status_0 <-
        status_wide[[
          paste0(
            "w",
            t0
          )
        ]]

      married_T0 <-
        !is.na(
          status_0
        ) &
        status_0 == "married"

      complete_window <-
        row_all_nonmissing(
          status_wide,
          full_cols
        )

      pre_all_nonmarried <-
        row_all_nonmarried(
          status_wide,
          pre_cols
        )

      tibble(
        pidp =
          status_wide$pidp,

        survey =
          survey_name,

        event_wave =
          t0,

        event_order =
          event_time_order(
            survey_name,
            t0
          ),

        stage1_married_T0 =
          married_T0,

        stage2_complete_Tminus5_T0_status =
          married_T0 &
            complete_window,

        stage3_nonmarried_Tminus5_Tminus1 =
          married_T0 &
            complete_window &
            pre_all_nonmarried,

        window_eligible =
          married_T0 &
            complete_window &
            pre_all_nonmarried
      )
    }
  ) |>
    left_join(
      first_observed_married,
      by =
        "pidp",
      relationship =
        "many-to-one"
    ) |>
    left_join(
      first_observed_marriage_evidence,
      by =
        "pidp",
      relationship =
        "many-to-one"
    ) |>
    mutate(

      first_observed_marriage =
        window_eligible &
          !is.na(
            first_observed_married_order
          ) &
          event_order ==
            first_observed_married_order,

      strict_first_marriage_proxy =
        first_observed_marriage &
          !is.na(
            first_marriage_evidence_order
          ) &
          event_order ==
            first_marriage_evidence_order
    )
}


bhps_marriage_candidates <- build_marriage_candidates(

    "BHPS",
  candidate_t0 =

# 16. ELIGIBILITY ATTRITION
# =============================================================================

make_first_marriage_attrition <- function(
    dat,
    survey_name
) {

  x <- dat |>
    filter(
      survey ==
        .env$survey_name
    )

  stages <- c(
    "stage1_married_T0",
    "stage2_complete_Tminus5_T0_status",
    "stage3_nonmarried_Tminus5_Tminus1",
    "first_observed_marriage",
    "strict_first_marriage_proxy"
  )

  stage_labels <- c(
    "Married at T0",
    "Married at T0 + complete status T-5:T0",
    "Window-only: non-married T-5:T-1 -> married T0",
    "FIRST OBSERVED MARRIAGE: window-only + no earlier observed married status",
    "STRICT FIRST-MARRIAGE PROXY: first observed + no earlier marriage evidence"
  )

  map2_dfr(
    stages,
    stage_labels,
    function(
        v,
        lab
    ) {

      keep <- x[[
        v
      ]]

      tibble(
        survey =
          survey_name,
        stage =
          lab,
        eligible_event_episodes =
          sum(
            keep,
            na.rm =
              TRUE
          ),
        unique_people =
          n_distinct(
            x$pidp[
              !is.na(
                keep
              ) &
                keep
            ]
          )
      )
    }
  )
}


eligibility_attrition <- bind_rows(
  make_first_marriage_attrition(
    marriage_candidates,
    "BHPS"
  ),
  make_first_marriage_attrition(

  )
)

# 18. BUILD EXACT T-5:T0 EVENT ROWS

build_exact_event_rows <- function(
    event_keys,
    condition_label
) {

  keys <- event_keys |>
    select(
      pidp,
      survey,
      event_wave
    ) |>
    distinct() |>
    mutate(
      condition =
        condition_label
    )

  expanded <- keys |>
    tidyr::crossing(
      event_time =
        -5:0
    ) |>
    mutate(
      wave_num =
        event_wave +
        event_time
    )

  expanded |>
    left_join(
      combined_long |>
        select(
          pidp,
          survey,
          wave_num,
          status,
          raw_status_label,
          ghq12,
          life_sat,
          age,
          sex
        ),
      by = c(
        "pidp",
        "survey",
        "wave_num"
      ),
      relationship =
        "many-to-one"
    ) |>
    arrange(
      survey,
      event_wave,
      pidp,
      event_time
    )
}

# 21. CONTEMPORANEOUS UNMARRIED RISK-SET CONTROLS
# =============================================================================

build_unmarried_controls <- function(
    status_wide,
    survey_name,
    treated_events,
    require_no_prior_observed_marriage = TRUE
) {

  survey_events <- treated_events |>
    filter(
      survey ==
        .env$survey_name
    )

  cohorts <- sort(
    unique(
      survey_events$event_wave
    )
  )

  if (length(
    cohorts
  ) == 0L) {
    return(
      tibble(
        pidp =
          status_wide$pidp[0],
        survey =
          character(0),
        event_wave =
          integer(0)
      )
    )
  }

  purrr::map_dfr(
    cohorts,
    function(
        t0
    ) {

      full_cols <- paste0(
        "w",
        (t0 - 5L):t0
      )

      missing_cols <- setdiff(
        full_cols,
        names(
          status_wide
        )
      )

      if (length(
        missing_cols
      ) > 0L) {
        stop(
          "Requested control wave columns do not exist for ",
          survey_name,
          " at event wave ",
          t0,
          ": ",
          paste(
            missing_cols,
            collapse = ", "
          )
        )
      }

      complete_window <-
        row_all_nonmissing(
          status_wide,
          full_cols
        )

      all_nonmarried <-
        row_all_nonmarried(
          status_wide,
          full_cols
        )

      out <- tibble(
        pidp =
          status_wide$pidp,
        survey =
          survey_name,
        event_wave =
          t0,
        event_order =
          event_time_order(
            survey_name,
            t0
          ),
        complete_window =
          complete_window,
        all_nonmarried =
          all_nonmarried
      ) |>
        filter(
          complete_window,
          all_nonmarried
        )

      if (
        require_no_prior_observed_marriage
      ) {

        out <- out |>
          left_join(
            first_observed_married,
            by =
              "pidp",
            relationship =
              "many-to-one"
          ) |>
          filter(
            is.na(
              first_observed_married_order
            ) |
              first_observed_married_order >
                event_order
          ) |>
          select(
            -first_observed_married_order
          )
      }

      out |>
        select(
          pidp,
          survey,
          event_wave
        )
    }
  )
}

# 24. SAME-SEX AGE-CALIPER CONTROL AVAILABILITY
# =============================================================================

age_caliper_diagnostic <- function(
    demo_dat
) {

  treated <- demo_dat |>
    filter(
      condition == "treated"
    ) |>
    transmute(
      treated_id =
        row_number(),
      treated_pidp =
        pidp,
      survey,
      event_wave,
      treated_age =
        age_T0,
      treated_sex =
        sex_T0
    )

  controls_by_age <- demo_dat |>
    filter(
      condition == "unmarried control",
      !is.na(
        age_T0
      ),
      !is.na(
        sex_T0
      )
    ) |>
    count(
      survey,
      event_wave,
      control_sex =
        sex_T0,
      control_age =
        age_T0,
      name =
        "n_controls_at_age"
    )

  candidates <- treated |>
    filter(
      !is.na(
        treated_age
      ),
      !is.na(
        treated_sex
      )
    ) |>
    tidyr::crossing(
      age_shift =
        -5:5
    ) |>
    mutate(
      control_age =
        treated_age +
        age_shift
    ) |>
    left_join(
      controls_by_age,
      by = c(
        "survey",
        "event_wave",
        "treated_sex" =
          "control_sex",
        "control_age"
      ),
      relationship =
        "many-to-one"
    ) |>
    mutate(
      n_controls_at_age =
        coalesce(
          n_controls_at_age,
          0L
        ),
      age_distance =
        abs(
          age_shift
        )
    )

  counts_with_demo <- candidates |>
    group_by(
      treated_id,
      treated_pidp,
      survey,
      event_wave,
      treated_age,
      treated_sex
    ) |>
    summarise(
      n_same_sex_age_pm1 =
        sum(
          n_controls_at_age[
            age_distance <= 1
          ]
        ),
      n_same_sex_age_pm2 =
        sum(
          n_controls_at_age[
            age_distance <= 2
          ]
        ),
      n_same_sex_age_pm3 =
        sum(
          n_controls_at_age[
            age_distance <= 3
          ]
        ),
      n_same_sex_age_pm5 =
        sum(
          n_controls_at_age[
            age_distance <= 5
          ]
        ),
      .groups =
        "drop"
    )

  missing_demo <- treated |>
    filter(
      is.na(
        treated_age
      ) |
        is.na(
          treated_sex
        )
    ) |>
    transmute(
      treated_id,
      treated_pidp,
      survey,
      event_wave,
      treated_age,
      treated_sex,
      n_same_sex_age_pm1 =
        0L,
      n_same_sex_age_pm2 =
        0L,
      n_same_sex_age_pm3 =
        0L,
      n_same_sex_age_pm5 =
        0L
    )

  per_treated <- bind_rows(
    counts_with_demo,
    missing_demo
  ) |>
    arrange(
      survey,
      event_wave,
      treated_pidp
    )

  summary <- per_treated |>
    summarise(
      treated_events =
        n(),
      treated_with_observed_age_sex =
        sum(
          !is.na(
            treated_age
          ) &
            !is.na(
              treated_sex
            )
        ),
      zero_matches_pm1 =
        sum(
          n_same_sex_age_pm1 == 0L
        ),
      zero_matches_pm2 =
        sum(
          n_same_sex_age_pm2 == 0L
        ),
      zero_matches_pm3 =
        sum(
          n_same_sex_age_pm3 == 0L
        ),
      zero_matches_pm5 =
        sum(
          n_same_sex_age_pm5 == 0L
        ),
      median_matches_pm1 =
        median(
          n_same_sex_age_pm1
        ),
      median_matches_pm2 =
        median(
          n_same_sex_age_pm2
        ),
      median_matches_pm3 =
        median(
          n_same_sex_age_pm3
        ),
      median_matches_pm5 =
        median(
          n_same_sex_age_pm5
        )
    )

  list(
    per_treated =
      per_treated,
    summary =
      summary
  )
}


age_caliper <- age_caliper_diagnostic(

        ),
      observed_unique_people =

                outcome
              ]]
            )
          ]
        ),
      percent_observed =
        100 *
          observed_rows /
          person_stack_rows,
      .groups =

    )

# 26. OUTCOME SAMPLE FLOW
# =============================================================================

outcome_sample_flow_one <- function(
    dat,
    outcome
) {

  person_stack <- dat |>
    group_by(
      pidp,
      survey,
      event_wave,
      condition
    ) |>
    summarise(
      observed_Tminus5 =
        any(
          event_time == -5L &
            !is.na(
              .data[[
                outcome
              ]]
            )
        ),
      observed_Tminus1 =
        any(
          event_time == -1L &
            !is.na(
              .data[[
                outcome
              ]]
            )
        ),
      n_observed =
        sum(
          !is.na(
            .data[[
              outcome
            ]]
          )
        ),
      .groups =
        "drop"
    )

  make_stage <- function(
      keep,
      label
  ) {

    person_stack |>
      filter(
        {{ keep }}
      ) |>
      group_by(
        survey,
        condition
      ) |>
      summarise(
        outcome =
          outcome,
        stage =
          label,
        person_stack_episodes =
          n(),
        unique_people =
          n_distinct(
            pidp
          ),
        .groups =
          "drop"
      )
  }

  bind_rows(

    person_stack |>
      group_by(
        survey,
        condition
      ) |>
      summarise(
        outcome =
          outcome,
        stage =
          "Status-eligible first-marriage risk-set episode",
        person_stack_episodes =
          n(),
        unique_people =
          n_distinct(
            pidp
          ),
        .groups =
          "drop"
      ),

    make_stage(
      observed_Tminus5,
      "Outcome observed at T-5"
    ),

    make_stage(
      observed_Tminus1,
      "Outcome observed at T-1"
    ),

    make_stage(
      observed_Tminus5 &
        observed_Tminus1,
      "Outcome observed at both T-5 and T-1"
    ),

    make_stage(
      observed_Tminus5 &
        n_observed >= 2L,
      "T-5 observed + >=1 additional outcome"
    ),

    make_stage(
      n_observed == 6L,
      "All six outcome occasions observed"
    )
  )
}


outcome_sample_flow <- bind_rows(

  outcome_sample_flow_one(
    primary_event_rows,
    "ghq12"
  ),

  outcome_sample_flow_one(

  )
)
