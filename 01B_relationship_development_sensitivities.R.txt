################################################################################
# UKHLS ADDITIONAL RELATIONSHIP-DEVELOPMENT SENSITIVITIES - VERSION 1
# Add-on to executed v4.1 + corrected targeted relationship-development analysis
#
# PURPOSE
# -------
# This script implements three follow-up analyses motivated by the corrected
# C1/C2 results:
#
#   B. BALANCED / COMMON-SPELL C1-C2 SENSITIVITY
#      For each later relationship-duration bin through 60-83 months, retain
#      only relationship spells with the relevant outcome observed both at
#      0-11 months and in that later bin. Re-estimate the within-spell change
#      and marriage/breakup-vs-ongoing differential change using relationship-
#      spell FE + wave FE and respondent-clustered SEs.
#
#   R. RDAS MARRIAGE-PROXIMITY ANALYSES
#      Test whether RDAS satisfaction/cohesion are already different in LEVEL
#      and/or show RECENT CHANGE as marriage approaches (25-60, 13-24, 1-12m).
#      R1 reproduces the v4.1 stable-60m comparator logic; R2 is an at-risk
#      landmark sensitivity that does NOT require controls to remain intact and
#      unmarried for 60 further months.
#
#   N. STRICT NON-OVERLAPPING LAGGED CHANGE SENSITIVITY
#      Re-run C3/C4 only when the predictor-change interval has completely ended
#      by the start of the outcome-change interval.
#
# IMPORTANT INTERPRETATION
# ------------------------
# These analyses are observational diagnostics/sensitivities. They do NOT
# identify causal relationship-development effects, causal mediation, or causal
# anticipation effects. RDAS proximity is especially informative when level and
# recent-change results are considered together.
################################################################################

zsafe <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  s <- stats::sd(x, na.rm = TRUE)
  if (is.na(s) || s == 0) return(rep(NA_real_, length(x)))
  (x - mean(x, na.rm = TRUE)) / s
}

scale_params <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  list(
    mean = mean(x, na.rm = TRUE),
    sd = stats::sd(x, na.rm = TRUE)
  )
}

z_with_params <- function(x, pars) {
  x <- suppressWarnings(as.numeric(x))
  if (is.null(pars) || is.na(pars$sd) || pars$sd == 0) {
    return(rep(NA_real_, length(x)))
  }
  (x - pars$mean) / pars$sd
}

finite_num <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  x[is.finite(x) & !is.na(x)]
}

safe_min <- function(x) {
  x <- finite_num(x)
  if (length(x) == 0L) NA_real_ else min(x)
}

safe_median <- function(x) {
  x <- finite_num(x)
  if (length(x) == 0L) NA_real_ else stats::median(x)
}

safe_max <- function(x) {
  x <- finite_num(x)
  if (length(x) == 0L) NA_real_ else max(x)
}

safe_quantile <- function(x, prob) {
  x <- finite_num(x)
  if (length(x) == 0L) {
    NA_real_
  } else {
    as.numeric(stats::quantile(x, probs = prob, na.rm = TRUE, names = FALSE))
  }
}

safe_model_nobs <- function(model) {
  if (is.null(model)) return(NA_integer_)

  n_direct <- tryCatch(model$nobs, error = function(e) NULL)
  if (!is.null(n_direct) && length(n_direct) >= 1L) {
    n_direct <- suppressWarnings(as.integer(n_direct[[1L]]))
    if (!is.na(n_direct)) return(n_direct)
  }

  n_resid <- tryCatch(
    length(stats::residuals(model)),
    error = function(e) NA_integer_
  )
  n_resid <- suppressWarnings(as.integer(n_resid))
  if (!is.na(n_resid) && n_resid > 0L) return(n_resid)

  NA_integer_
}

save_fixest_model <- function(model, name) {
  if (is.null(model)) return(invisible(NULL))

  saveRDS(model, file.path(OUT_DIR, "models", paste0(name, ".rds")))

  cf <- stats::coef(model)
  se_ <- fixest::se(model)
  p_ <- fixest::pvalue(model)
  ci_ <- stats::confint(model)

  out <- tibble(
    term = names(cf),
    estimate = as.numeric(cf),
    std_error = as.numeric(se_),
    conf_low = as.numeric(ci_[, 1]),
    conf_high = as.numeric(ci_[, 2]),
    p_value = as.numeric(p_)
  )

  write_csv_quiet(out, paste0(name, "_coefficients.csv"))
  write_txt(capture.output(summary(model)), paste0(name, "_summary.txt"))

  invisible(model)
}

safe_feols <- function(formula, data, name, ...) {
  model <- tryCatch(
    fixest::feols(formula, data = data, ...),
    error = function(e) e
  )

  if (inherits(model, "error")) {
    write_txt(
      paste0("MODEL FAILED\n\n", conditionMessage(model)),
      paste0(name, "_ERROR.txt")
    )
    return(NULL)
  }

  save_fixest_model(model, name)
}

extract_term_row <- function(
  model,
  term,
  analysis,
  outcome,
  window = NA_character_,
  comparator = NA_character_,
  estimand = NA_character_
) {
  base <- tibble(
    analysis = analysis,
    outcome = outcome,
    window = window,
    comparator = comparator,
    estimand = estimand,
    term = term,
    estimate = NA_real_,
    std_error = NA_real_,
    conf_low = NA_real_,
    conf_high = NA_real_,
    p_value = NA_real_,
    nobs = if (is.null(model)) NA_integer_ else safe_model_nobs(model),
    status = if (is.null(model)) "model_missing" else "term_missing"
  )

  if (is.null(model)) return(base)

  b <- stats::coef(model)
  if (!term %in% names(b)) return(base)

  se_ <- fixest::se(model)
  p_ <- fixest::pvalue(model)
  ci_ <- stats::confint(model)

  tibble(
    analysis = analysis,
    outcome = outcome,
    window = window,
    comparator = comparator,
    estimand = estimand,
    term = term,
    estimate = as.numeric(b[term]),
    std_error = as.numeric(se_[term]),
    conf_low = as.numeric(ci_[term, 1]),
    conf_high = as.numeric(ci_[term, 2]),
    p_value = as.numeric(p_[term]),
    nobs = safe_model_nobs(model),
    status = "estimated"
  )
}

lincom_row <- function(
  model,
  weights_named,
  analysis,
  outcome,
  duration_bin = NA_character_,
  spell_outcome = NA_character_,
  contrast = NA_character_
) {
  empty <- tibble(
    analysis = analysis,
    outcome = outcome,
    duration_bin = duration_bin,
    spell_outcome = spell_outcome,
    contrast = contrast,
    estimate = NA_real_,
    std_error = NA_real_,
    conf_low = NA_real_,
    conf_high = NA_real_,
    z = NA_real_,
    p_value = NA_real_,
    nobs = if (is.null(model)) NA_integer_ else safe_model_nobs(model),
    status = if (is.null(model)) "model_missing" else "term_missing"
  )

  if (is.null(model)) return(empty)

  b <- stats::coef(model)
  V <- stats::vcov(model)
  needed <- names(weights_named)

  if (any(!needed %in% names(b))) {
    empty$status <- paste0(
      "term_missing: ",
      paste(setdiff(needed, names(b)), collapse = " | ")
    )
    return(empty)
  }

  w <- as.numeric(weights_named)
  bb <- as.numeric(b[needed])
  VV <- as.matrix(V[needed, needed, drop = FALSE])

  est <- sum(w * bb)
  var_est <- as.numeric(t(w) %*% VV %*% w)

  if (!is.finite(var_est) || var_est < 0) {
    empty$estimate <- est
    empty$status <- "invalid_variance"
    return(empty)
  }

  se_est <- sqrt(var_est)
  z <- ifelse(se_est > 0, est / se_est, NA_real_)
  p <- ifelse(
    is.finite(z),
    2 * stats::pnorm(abs(z), lower.tail = FALSE),
    NA_real_
  )

  tibble(
    analysis = analysis,
    outcome = outcome,
    duration_bin = duration_bin,
    spell_outcome = spell_outcome,
    contrast = contrast,
    estimate = est,
    std_error = se_est,
    conf_low = est - 1.96 * se_est,
    conf_high = est + 1.96 * se_est,
    z = z,
    p_value = p,
    nobs = safe_model_nobs(model),
    status = "estimated"
  )
}

safe_wald_row <- function(
  model,
  keep,
  test_name,
  outcome,
  analysis
) {
  empty <- tibble(
    analysis = analysis,
    test = test_name,
    outcome = outcome,
    stat = NA_real_,
    p_value = NA_real_,
    df = NA_real_,
    n_terms = NA_integer_,
    vcov_rank = NA_integer_,
    matched_terms = NA_character_,
    nobs = if (is.null(model)) NA_integer_ else safe_model_nobs(model),
    status = if (is.null(model)) "model_missing" else "no_matching_coefficients"
  )

  if (is.null(model)) return(empty)

  b_all <- tryCatch(stats::coef(model), error = function(e) NULL)
  V_all <- tryCatch(stats::vcov(model), error = function(e) NULL)
  if (is.null(b_all) || is.null(V_all)) {
    empty$status <- "coef_or_vcov_unavailable"
    return(empty)
  }

  idx <- grep(keep, names(b_all), perl = TRUE)
  if (length(idx) == 0L) return(empty)

  terms <- names(b_all)[idx]
  b <- as.numeric(b_all[terms])
  V <- as.matrix(V_all[terms, terms, drop = FALSE])

  if (any(!is.finite(b)) || any(!is.finite(V))) {
    empty$n_terms <- length(terms)
    empty$matched_terms <- paste(terms, collapse = " | ")
    empty$status <- "nonfinite_coef_or_vcov"
    return(empty)
  }

  ee <- tryCatch(eigen(V, symmetric = TRUE), error = function(e) NULL)
  if (is.null(ee)) {
    empty$n_terms <- length(terms)
    empty$matched_terms <- paste(terms, collapse = " | ")
    empty$status <- "vcov_eigendecomposition_failed"
    return(empty)
  }

  max_eig <- max(abs(ee$values), na.rm = TRUE)
  tol <- max(1, nrow(V)) * max(max_eig, 1) * .Machine$double.eps * 100
  keep_eig <- which(ee$values > tol)
  rank_v <- length(keep_eig)

  if (rank_v == 0L) {
    empty$n_terms <- length(terms)
    empty$vcov_rank <- 0L
    empty$matched_terms <- paste(terms, collapse = " | ")
    empty$status <- "vcov_rank_zero"
    return(empty)
  }

  Q <- ee$vectors[, keep_eig, drop = FALSE]
  V_inv <- Q %*% diag(1 / ee$values[keep_eig], nrow = rank_v) %*% t(Q)
  W <- as.numeric(t(b) %*% V_inv %*% b)
  p <- stats::pchisq(W, df = rank_v, lower.tail = FALSE)

  tibble(
    analysis = analysis,
    test = test_name,
    outcome = outcome,
    stat = W,
    p_value = p,
    df = as.numeric(rank_v),
    n_terms = as.integer(length(terms)),
    vcov_rank = as.integer(rank_v),
    matched_terms = paste(terms, collapse = " | "),
    nobs = safe_model_nobs(model),
    status = if_else(
      rank_v == length(terms),
      "estimated",
      "estimated_reduced_rank"
    )
  )
}

add_riskset_id <- function(dat) {
  dat %>%
    mutate(
      duration_stratum = floor(cohab_duration_months / RISKSET_DURATION_BIN_MONTHS),
      age_stratum = floor(age / RISKSET_AGE_BIN_YEARS),
      riskset_stratum = interaction(
        wave, sex, duration_stratum, age_stratum, drop = TRUE
      )
    )
}

make_att_riskset <- function(dat, treated_var = "treated") {
  dat <- dat %>%
    filter(
      !is.na(riskset_stratum),
      !is.na(.data[[treated_var]])
    )

  counts <- dat %>%
    group_by(riskset_stratum) %>%
    summarise(
      n_treated = sum(.data[[treated_var]] == 1, na.rm = TRUE),
      n_control = sum(.data[[treated_var]] == 0, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    filter(n_treated > 0, n_control > 0)

  dat %>%
    inner_join(counts, by = "riskset_stratum") %>%
    mutate(
      att_weight = if_else(
        .data[[treated_var]] == 1,
        1,
        n_treated / n_control
      )
    )
}

# ==============================================================================
# 5. B: BALANCED / COMMON-SPELL C1-C2 SENSITIVITY
# ==============================================================================
#
# For each target duration bin, a relationship spell is included only if the
# relevant outcome is observed at least once in BOTH 0-11m and the target bin.
# The model then uses all nonmissing observations from those two bins:
#
#   outcome_z ~ target + marriage_x_target + breakup_x_target |
#               relationship_spell_id + wave
#
# Therefore:
#   target                  = within-spell change for ongoing cohabitations
#   target + marriage_x     = within-spell change for future-marriage spells
#   target + breakup_x      = within-spell change for future-breakup spells
#   marriage_x              = differential change: marriage vs ongoing
#   breakup_x               = differential change: breakup vs ongoing
#
# This is a balanced-overlap sensitivity, not a new primary estimator.
# ==============================================================================

balanced_outcomes <- tribble(
  ~outcome_var, ~outcome_label, ~stub,
  "ghq_z", "GHQ distress", "GHQ",
  "ls_z", "Life satisfaction", "LS",
  "rdas_satisfaction_z", "RDAS satisfaction", "RDAS_SAT",
  "rdas_cohesion_z", "RDAS cohesion", "RDAS_COH"
)

balanced_models <- list()
balanced_counts <- list()
balanced_contrasts <- list()

fit_balanced_one <- function(outcome_var, outcome_label, stub, target_bin) {

  nonmissing <- development %>%
    filter(
      spell_outcome %in% c("ongoing", "marriage", "breakup"),
      duration_bin %in% c("0-11", target_bin),
      !is.na(.data[[outcome_var]])
    )

  eligible_spells <- nonmissing %>%
    distinct(relationship_spell_id, duration_bin) %>%
    group_by(relationship_spell_id) %>%
    summarise(
      has_ref = any(duration_bin == "0-11"),
      has_target = any(as.character(duration_bin) == target_bin),
      .groups = "drop"
    ) %>%
    filter(has_ref, has_target) %>%
    pull(relationship_spell_id)

  dat <- nonmissing %>%
    filter(relationship_spell_id %in% eligible_spells) %>%
    mutate(
      target_bin_indicator = as.integer(as.character(duration_bin) == target_bin),
      marriage_x_target = as.integer(spell_outcome == "marriage") * target_bin_indicator,
      breakup_x_target = as.integer(spell_outcome == "breakup") * target_bin_indicator
    )

  counts <- dat %>%
    distinct(pidp, relationship_spell_id, spell_outcome) %>%
    count(spell_outcome, name = "n_relationship_spells") %>%
    mutate(
      outcome = outcome_label,
      target_bin = target_bin,
      n_rows_model_pool = nrow(dat),
      n_persons_model_pool = dplyr::n_distinct(dat$pidp)
    ) %>%
    select(
      outcome, target_bin, spell_outcome,
      n_relationship_spells, n_rows_model_pool, n_persons_model_pool
    )

  model_name <- paste0(
    "B_BALANCED_", stub, "_0_11_vs_", gsub("-", "_", target_bin)
  )

  model <- safe_feols(
    stats::as.formula(
      paste0(
        outcome_var,
        " ~ target_bin_indicator + marriage_x_target + breakup_x_target | ",
        "relationship_spell_id + wave"
      )
    ),
    data = dat,
    name = model_name,
    vcov = ~pidp

# ==============================================================================
# 6. R: RDAS MARRIAGE-PROXIMITY ANALYSES
# ==============================================================================
#
# We estimate both:
#   LEVEL: current standardized RDAS level ~ treated | risk-set stratum
#   CHANGE: standardized recent RDAS change ~ treated + prior RDAS level |
#           risk-set stratum
#
# The CHANGE model is the more direct diagnostic of a proximal increase.
# The LEVEL model shows whether relationship-quality differences are already
# present at that proximity window.
#
# Two comparator definitions are reported:
#
# R1 stable60:
#   exact v4.1 logic: marriage in target window vs continued cohabitation >=60m.
#
# R2 at_risk:
#   treated = exact marriage in target window;
#   controls = currently cohabiting observations with NO KNOWN linked marriage
#              within the upper bound of that window. Controls may later marry
#              or may dissolve; no 60m future-survival condition is imposed.
#
# R2 is a sensitivity to selection induced by the stable-60m control definition.
# ==============================================================================

windows <- tribble(

  "25-60m", 25L, 60L, "marriage_25_60m"
)

rdas_specs <- tribble(

  "RDAS satisfaction", "rdas_satisfaction_z_all", "rdas_satisfaction_change_z_all",
  "prior_rdas_satisfaction_z_all", "prior_rdas_satisfaction_ym",
  "RDAS cohesion", "rdas_cohesion_z_all", "rdas_cohesion_change_z_all",
  "prior_rdas_cohesion_z_all", "prior_rdas_cohesion_ym"
)

prepare_rdas_pair <- function(
  comparator_type = c("stable60", "at_risk"),
  lower,
  upper,
  treated_state,
  level_var,
  change_var,
  prior_level_var
) {
  comparator_type <- match.arg(comparator_type)

  if (comparator_type == "stable60") {
    dat <- primary_support %>%
      filter(
        marriage_horizon_state %in% c(treated_state, "continued_cohab_60m")
      ) %>%
      mutate(
        treated = as.integer(marriage_horizon_state == treated_state)
      )
  } else {
    # Landmark risk-set sensitivity. Controls are cohabiting at the current
    # interview and have no KNOWN linked marriage by the upper edge of the
    # target window. They are NOT required to remain together for 60 months.
    # Rows with any known marriage inside <= upper months are excluded from the
    # control pool even if its date flag is not exact; this avoids knowingly
    # classifying near-marriage rows as controls.
    dat <- primary_support %>%
      mutate(
        treated = as.integer(
          exact_marriage_date &
            !is.na(months_to_marriage) &
            months_to_marriage >= lower &
            months_to_marriage <= upper
        ),
        eligible_at_risk_control =
          treated == 0 &
          (
            is.na(months_to_marriage) |
              months_to_marriage > upper
          )
      ) %>%
      filter(treated == 1 | eligible_at_risk_control)
  }

  dat %>%
    mutate(
      outcome_level = .data[[level_var]],
      outcome_change = .data[[change_var]],
      prior_level = .data[[prior_level_var]]
    )
}

summarise_arm_counts <- function(dat, model_label) {
  if (nrow(dat) == 0L) {
    return(
      tibble(
        treated = integer(),
        n_rows = integer(),
        n_persons = integer(),
        n_spells = integer(),
        weighted_mass_ATT = numeric(),
        model = character()
      )
    )
  }

  dat %>%
    group_by(treated) %>%
    summarise(
      n_rows = n(),
      n_persons = dplyr::n_distinct(pidp),
      n_spells = dplyr::n_distinct(relationship_spell_id),
      weighted_mass_ATT = sum(att_weight, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(model = model_label)
}

run_rdas_one <- function(
  comparator_type,
  window_name,
  lower,
  upper,
  treated_state,
  outcome_label,
  level_var,
  change_var,
  prior_level_var,
  prior_time_var
) {

  pair <- prepare_rdas_pair(
    comparator_type = comparator_type,
    lower = lower,
    upper = upper,
    treated_state = treated_state,
    level_var = level_var,
    change_var = change_var,
    prior_level_var = prior_level_var
  ) %>%
    add_riskset_id()

  level_pool <- pair %>%
    filter(!is.na(outcome_level))
  level_rs <- make_att_riskset(level_pool)

  # Both recent-change models use the same complete sample so the difference
  # between them is due only to adjustment for the prior RDAS level.
  change_pool <- pair %>%
    filter(!is.na(outcome_change), !is.na(prior_level))
  change_rs <- make_att_riskset(change_pool)

  stub <- paste0(
    "R_", comparator_type, "_",
    ifelse(outcome_label == "RDAS satisfaction", "SAT", "COH"), "_",
    gsub("-", "_", window_name)
  )

  level_model <- safe_feols(
    outcome_level ~ treated | riskset_stratum,
    data = level_rs,
    name = paste0(stub, "_LEVEL"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  change_unadjusted_model <- safe_feols(
    outcome_change ~ treated | riskset_stratum,
    data = change_rs,
    name = paste0(stub, "_CHANGE_unadjusted"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  change_adjusted_model <- safe_feols(
    outcome_change ~ treated + prior_level | riskset_stratum,
    data = change_rs,
    name = paste0(stub, "_CHANGE_plus_prior_level"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  level_result <- extract_term_row(
    level_model,
    "treated",
    analysis = "RDAS_marriage_proximity_level",
    outcome = outcome_label,
    window = window_name,
    comparator = comparator_type,
    estimand = "ATT-oriented current-level contrast"
  )

  change_unadjusted_result <- extract_term_row(
    change_unadjusted_model,
    "treated",
    analysis = "RDAS_marriage_proximity_recent_change_unadjusted",
    outcome = outcome_label,
    window = window_name,
    comparator = comparator_type,
    estimand = "ATT-oriented recent-change contrast"
  )

  change_adjusted_result <- extract_term_row(
    change_adjusted_model,
    "treated",
    analysis = "RDAS_marriage_proximity_recent_change_prior_level_adjusted",
    outcome = outcome_label,
    window = window_name,
    comparator = comparator_type,
    estimand = "ATT-oriented recent-change contrast adjusted for prior RDAS level"
  )

  counts <- bind_rows(
    summarise_arm_counts(level_rs, "level"),
    summarise_arm_counts(change_rs, "recent_change_common_sample")
  ) %>%
    mutate(
      comparator = comparator_type,
      window = window_name,
      outcome = outcome_label,
      treatment_group = if_else(treated == 1, "marriage_window", "control")
    ) %>%
    select(
      comparator, window, outcome, model, treatment_group,
      n_rows, n_persons, n_spells, weighted_mass_ATT
    )

  # Timing audit for treated recent-change observations. The prior RDAS level
  # time is the START of the recent RDAS change interval.
  timing_dat <- change_rs %>%
    filter(treated == 1) %>%
    mutate(
      prior_measurement_ym = .data[[prior_time_var]],
      change_gap_months = interview_ym - prior_measurement_ym,
      months_to_marriage_at_change_start = marriage_start_ym - prior_measurement_ym,
      months_to_marriage_at_change_end = months_to_marriage
    )

  timing <- tibble(
    comparator = comparator_type,
    window = window_name,
    outcome = outcome_label,
    n = nrow(timing_dat),
    n_persons = dplyr::n_distinct(timing_dat$pidp),
    n_spells = dplyr::n_distinct(timing_dat$relationship_spell_id),
    gap_min = safe_min(timing_dat$change_gap_months),
    gap_median = safe_median(timing_dat$change_gap_months),
    gap_p90 = safe_quantile(timing_dat$change_gap_months, .90),
    start_mte_min = safe_min(timing_dat$months_to_marriage_at_change_start),
    start_mte_median = safe_median(timing_dat$months_to_marriage_at_change_start),
    start_mte_p90 = safe_quantile(timing_dat$months_to_marriage_at_change_start, .90),
    end_mte_min = safe_min(timing_dat$months_to_marriage_at_change_end),
    end_mte_median = safe_median(timing_dat$months_to_marriage_at_change_end),
    end_mte_max = safe_max(timing_dat$months_to_marriage_at_change_end)
  )

  list(
    level_model = level_model,
    change_unadjusted_model = change_unadjusted_model,
    change_adjusted_model = change_adjusted_model,
    level_result = level_result,
    change_unadjusted_result = change_unadjusted_result,
    change_adjusted_result = change_adjusted_result,
    counts = counts,
    timing = timing
  )
}

# ==============================================================================
#
# C3 strict rule:
#   both prior RDAS-change endpoints <= start of current psychological-change
#   interval.
#
# C4 strict rule:
#   prior psychological-change endpoint <= start of current RDAS-change interval.

# ----------------------------------------------------------------------------

build_C3_ghq_pair <- function() {
  base <- development %>%
    filter(
      !is.na(spell_ghq_change),
      usable_prior_rdas_change,
      !is.na(prior_spell_ghq),
      !is.na(prior_spell_ghq_ym),
      !is.na(prior_rdas_sat_change_ym),
      !is.na(prior_rdas_coh_change_ym)
    )

  pars_y <- scale_params(base$spell_ghq_change)
  pars_level <- scale_params(base$prior_spell_ghq)
  pars_sat <- scale_params(base$prior_rdas_sat_change)
  pars_coh <- scale_params(base$prior_rdas_coh_change)

  original <- base %>%
    mutate(
      ghq_change_z = z_with_params(spell_ghq_change, pars_y),
      prior_ghq_level_z = z_with_params(prior_spell_ghq, pars_level),
      prior_rdas_sat_change_z = z_with_params(prior_rdas_sat_change, pars_sat),
      prior_rdas_coh_change_z = z_with_params(prior_rdas_coh_change, pars_coh)
    )

  strict <- original %>%
    filter(
      prior_rdas_sat_change_ym <= prior_spell_ghq_ym,
      prior_rdas_coh_change_ym <= prior_spell_ghq_ym
    )

  list(original = original, strict = strict)
}

build_C4_pair <- function(
  outcome_change_var,
  prior_psych_change_var,
  prior_psych_ok_var,
  prior_psych_change_time_var,
  prior_rdas_level_var,
  outcome_interval_start_var
) {
  base <- development %>%
    mutate(
      y_raw = .data[[outcome_change_var]],
      psych_change_raw = .data[[prior_psych_change_var]],
      psych_ok = .data[[prior_psych_ok_var]],
      psych_change_end_ym = .data[[prior_psych_change_time_var]],
      prior_rdas_level_raw = .data[[prior_rdas_level_var]],
      outcome_start_ym = .data[[outcome_interval_start_var]]
    ) %>%
    filter(
      !is.na(y_raw),
      psych_ok,
      !is.na(psych_change_raw),
      !is.na(psych_change_end_ym),
      !is.na(prior_rdas_level_raw),
      !is.na(outcome_start_ym)
    )

  pars_y <- scale_params(base$y_raw)
  pars_psych <- scale_params(base$psych_change_raw)
  pars_level <- scale_params(base$prior_rdas_level_raw)

  original <- base %>%
    mutate(
      y_z = z_with_params(y_raw, pars_y),
      prior_psych_change_z = z_with_params(psych_change_raw, pars_psych),
      prior_rdas_level_z = z_with_params(prior_rdas_level_raw, pars_level)
    )

  strict <- original %>%
    filter(psych_change_end_ym <= outcome_start_ym)

  list(original = original, strict = strict)
}
