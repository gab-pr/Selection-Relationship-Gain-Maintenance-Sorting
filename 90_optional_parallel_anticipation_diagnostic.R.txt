################################################################################
# UKHLS PARALLEL ANTICIPATION WORKFLOW V1
# COHABITATION AND MARRIAGE
#
# COMPLEMENTARY / EXPLORATORY ANALYSES
#
# PURPOSE
# -------
# To test anticipation of TWO formal relationship transitions with as nearly
# parallel an analytic strategy as the UKHLS data permit.
#
# TRANSITION 1: LAT -> COHABITATION
#   Objective estimand:
#     Does recent psychological change become more favorable as cohabitation
#     approaches among respondents already in a steady non-residential (LAT)
#     relationship, relative to comparable LAT relationships that remain at
#     risk over the same event horizon?
#   Subjective estimand:
#     Does NCRR11 intention to begin living together predict subsequent
#     psychological improvement while the respondent is still in the same LAT
#     relationship?
#
# TRANSITION 2: COHABITATION -> MARRIAGE
#   Objective estimand:
#     Does recent psychological change become more favorable as marriage
#     approaches among respondents already cohabiting and unmarried, relative
#     to comparable cohabiting relationships that remain unmarried and intact
#     over the same event horizon?
#   Subjective estimand:
#     If UKHLS contains a repeated cohabiter-specific marriage-expectation item,
#     does stronger expectation of marrying the CURRENT cohabiting partner
#     predict subsequent psychological improvement while the respondent is
#     still unmarried and in the same cohabiting relationship?
#
# IMPORTANT CONCEPTUAL DISTINCTIONS
# ---------------------------------
# * Psychological LEVEL differences are not primary anticipation evidence.
#   They may reflect selection, differentiation, or sorting.
# * Primary anticipation outcomes are RECENT WITHIN-PERSON CHANGES.
# * Controls are NOT defined as never-transitioners. A control may cohabit or
#   marry later; it must only remain transition-free and in the same risk state
#   through the treated person's analogous event horizon.
# * Objective anticipation is most convincing if the treated-control change
#   contrast becomes increasingly favorable as the event approaches:
#       beta_1-12m > beta_13-24m > beta_25-60m.
# * Subjective anticipation is most convincing if an expectation measure that
#   demonstrably predicts the transition also predicts subsequent PRE-EVENT
#   psychological improvement.
#
# PRIMARY MATCHING CHANGE RELATIVE TO THE EARLIER COHABITATION WORKFLOW
# --------------------------------------------------------------------
# The primary risk-set estimator now requires >=1 outcome-complete control per
# set. >=3 and >=5 controls are saved as sensitivities. This avoids the severe
# information loss observed when >=5 controls was required as the primary rule.
#
# OUTCOME ORIENTATION
# -------------------
# Life satisfaction: higher = better.
# GHQ well-being: -GHQ12, so higher = better / less distress.
# Positive change always means more favorable psychological functioning.
#
# INFERENTIAL STATUS
# ------------------
# Post-hoc complementary/exploratory analyses. Holm-adjusted p values are saved
# as descriptive multiplicity control; they do not make these analyses
# confirmatory.
################################################################################

write_csv_quiet <- function(x, filename, subdir = NULL) {
  path <- if (is.null(subdir)) {
    file.path(RESULTS_DIR, filename)
  } else {
    dir.create(file.path(RESULTS_DIR, subdir), recursive = TRUE,
               showWarnings = FALSE)
    file.path(RESULTS_DIR, subdir, filename)
  }
  readr::write_csv(x, path, na = "")
}

num_clean <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z[z < 0] <- NA_real_
  z
}

safe_sd <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) < 2L) return(NA_real_)
  stats::sd(z)
}

normal_ci <- function(est, se, level = .95) {
  q <- qnorm(1 - (1 - level) / 2)
  c(est - q * se, est + q * se)
}

safe_model_nobs <- function(model) {
  if (is.null(model)) return(NA_integer_)
  n1 <- tryCatch(stats::nobs(model), error = function(e) NA_real_)
  if (length(n1) >= 1L && is.finite(n1[1])) return(as.integer(n1[1]))
  n2 <- tryCatch(model$nobs, error = function(e) NULL)
  if (!is.null(n2) && length(n2) >= 1L && is.finite(n2[1])) {
    return(as.integer(n2[1]))
  }
  r <- tryCatch(model$residuals, error = function(e) NULL)
  if (!is.null(r) && length(r) > 0L) return(as.integer(length(r)))
  NA_integer_
}

safe_feols <- function(formula, data, weights = NULL, vcov = NULL) {
  tryCatch(
    fixest::feols(formula, data = data, weights = weights, vcov = vcov,
                  warn = FALSE, notes = FALSE),
    error = function(e) {
      message("feols failed: ", conditionMessage(e))
      NULL
    }
  )
}

safe_feglm <- function(formula, data, family, vcov = NULL) {
  tryCatch(
    fixest::feglm(formula, data = data, family = family, vcov = vcov,
                  warn = FALSE, notes = FALSE),
    error = function(e) {
      message("feglm failed: ", conditionMessage(e))
      NULL
    }
  )
}

extract_fixest_term <- function(model, term, model_name, outcome, analysis) {
  if (is.null(model)) return(tibble())
  ct <- tryCatch(as.data.frame(fixest::coeftable(model)),
                 error = function(e) NULL)
  if (is.null(ct) || nrow(ct) == 0L || !term %in% rownames(ct)) return(tibble())
  est <- unname(ct[term, 1])
  se <- unname(ct[term, 2])
  p <- unname(ct[term, 4])
  ci <- normal_ci(est, se)
  tibble(
    analysis = analysis,
    model = model_name,
    outcome = outcome,
    term = term,
    estimate = est,
    std_error = se,
    ci_low = ci[1],
    ci_high = ci[2],
    p_value = p,
    n_obs = safe_model_nobs(model)
  )
}

extract_fixest_all <- function(model, model_name, outcome, analysis) {
  if (is.null(model)) return(tibble())
  ct <- tryCatch(as.data.frame(fixest::coeftable(model)),
                 error = function(e) NULL)
  if (is.null(ct) || nrow(ct) == 0L) return(tibble())
  out <- tibble(
    term = rownames(ct),
    estimate = ct[[1]],
    std_error = ct[[2]],
    statistic = ct[[3]],
    p_value = ct[[4]]
  ) %>%
    rowwise() %>%
    mutate(
      ci_low = normal_ci(estimate, std_error)[1],
      ci_high = normal_ci(estimate, std_error)[2]
    ) %>%
    ungroup() %>%
    mutate(
      analysis = analysis,
      model = model_name,
      outcome = outcome,
      n_obs = safe_model_nobs(model),
      .before = 1
    )
  out
}

linear_contrast <- function(model, weights_named, contrast_name, outcome,
                            analysis) {
  if (is.null(model)) return(tibble())
  b <- stats::coef(model)
  V <- tryCatch(stats::vcov(model), error = function(e) NULL)
  if (is.null(V)) return(tibble())
  needed <- names(weights_named)
  if (!all(needed %in% names(b))) return(tibble())
  w <- rep(0, length(b)); names(w) <- names(b)
  w[needed] <- weights_named
  est <- sum(w * b)
  se <- sqrt(as.numeric(t(w) %*% V %*% w))
  ci <- normal_ci(est, se)
  z <- est / se
  p <- 2 * pnorm(abs(z), lower.tail = FALSE)
  tibble(
    analysis = analysis,
    model = "direct_proximity_gradient",
    outcome = outcome,
    contrast = contrast_name,
    estimate = est,
    std_error = se,
    ci_low = ci[1],
    ci_high = ci[2],
    p_value = p,
    n_obs = safe_model_nobs(model)
  )
}

# Most recent nonmissing prior value within an episode/spell and gap restriction.
attach_prior_within_group <- function(df, group_col, value_col, prefix) {
  g <- rlang::sym(group_col)
  v <- rlang::sym(value_col)
  df %>%
    arrange(!!g, interview_ym, wave_n) %>%
    group_by(!!g) %>%
    group_modify(~ {
      x <- .x[[value_col]]
      tt <- .x$interview_ym
      prior_value <- rep(NA_real_, length(x))
      prior_time <- rep(NA_real_, length(x))
      last_val <- NA_real_
      last_time <- NA_real_
      for (i in seq_along(x)) {
        prior_value[i] <- last_val
        prior_time[i] <- last_time
        if (!is.na(x[i])) {
          last_val <- x[i]
          last_time <- tt[i]
        }
      }
      .x %>%
        mutate(
          "{prefix}_prior_value" := prior_value,
          "{prefix}_prior_ym" := prior_time,
          "{prefix}_prior_gap" := interview_ym - prior_time
        )
    }) %>%
    ungroup()
}

# Next nonmissing outcome within same relationship episode/spell.
attach_next_within_group <- function(df, group_col, value_col, prefix) {
  g <- rlang::sym(group_col)
  df %>%
    arrange(!!g, interview_ym, wave_n) %>%
    group_by(!!g) %>%
    group_modify(~ {
      x <- .x[[value_col]]
      tt <- .x$interview_ym
      next_value <- rep(NA_real_, length(x))
      next_time <- rep(NA_real_, length(x))
      next_val_seen <- NA_real_
      next_time_seen <- NA_real_
      for (i in rev(seq_along(x))) {
        next_value[i] <- next_val_seen
        next_time[i] <- next_time_seen
        if (!is.na(x[i])) {
          next_val_seen <- x[i]
          next_time_seen <- tt[i]
        }
      }
      .x %>%
        mutate(
          "{prefix}_next_value" := next_value,
          "{prefix}_next_ym" := next_time,
          "{prefix}_next_gap" := next_time - interview_ym
        )
    }) %>%
    ungroup()
}

  target1 <- paste0(tolower(wave), "_", tolower(stem))
  nlow <- tolower(nms)

# This function takes transition-specific treated landmarks and candidate
# controls already satisfying the design-specific eligibility/caliper rules.
# Both transitions are then estimated identically.
fit_parallel_objective_outcome <- function(
    transition,
    treated_df,
    candidates_df,
    treated_change_col,
    control_change_col,
    treated_prior_col,
    control_prior_col,
    outcome,
    min_controls,
    spec_name
) {

  treated_part <- treated_df %>%
    transmute(
      match_set_id,
      proximity_window,
      window_order,
      pidp,
      role = "treated",
      treated = 1,
      change_z = .data[[treated_change_col]],
      prior_level_z = .data[[treated_prior_col]]
    )

  control_part <- candidates_df %>%
    transmute(
      match_set_id,
      proximity_window,
      window_order,
      pidp = control_pidp,
      role = "control",
      treated = 0,
      change_z = .data[[control_change_col]],
      prior_level_z = .data[[control_prior_col]]
    )

  stack <- bind_rows(treated_part, control_part) %>%
    filter(!is.na(change_z), !is.na(prior_level_z)) %>%
    group_by(match_set_id) %>%
    mutate(
      n_treated = sum(role == "treated"),
      n_controls = sum(role == "control")
    ) %>%
    ungroup() %>%
    filter(n_treated == 1L, n_controls >= min_controls) %>%
    group_by(match_set_id) %>%
    mutate(att_weight = if_else(role == "treated", 1, 1 / n_controls)) %>%
    ungroup()

  results <- tibble()
  models <- list()

  for (w in PROXIMITY_WINDOWS$window) {
    d <- stack %>% filter(proximity_window == w)
    if (n_distinct(d$match_set_id) == 0L) next

    m0 <- safe_feols(
      change_z ~ treated | match_set_id,
      data = d,
      weights = ~att_weight,
      vcov = ~pidp + match_set_id
    )

    m1 <- safe_feols(
      change_z ~ treated + prior_level_z | match_set_id,
      data = d,
      weights = ~att_weight,
      vcov = ~pidp + match_set_id
    )

    models[[paste0(w, "_unadjusted")]] <- m0
    models[[paste0(w, "_prior_adjusted")]] <- m1

    r0 <- extract_fixest_term(
      m0, "treated", paste0(transition, "_", spec_name, "_", w),
      outcome, paste0(transition, " objective proximity")
    ) %>%
      mutate(
        transition = transition,
        proximity_window = w,
        specification = spec_name,
        model_specification = "preferred matched recent change",
        min_controls_required = min_controls,
        matched_sets = n_distinct(d$match_set_id),
        treated_people = n_distinct(d$pidp[d$role == "treated"]),
        unique_controls = n_distinct(d$pidp[d$role == "control"])
      )

    r1 <- extract_fixest_term(
      m1, "treated", paste0(transition, "_", spec_name, "_prior_", w),
      outcome, paste0(transition, " objective proximity")
    ) %>%
      mutate(
        transition = transition,
        proximity_window = w,
        specification = spec_name,
        model_specification = "prior-level-adjusted sensitivity",
        min_controls_required = min_controls,
        matched_sets = n_distinct(d$match_set_id),
        treated_people = n_distinct(d$pidp[d$role == "treated"]),
        unique_controls = n_distinct(d$pidp[d$role == "control"])
      )

    results <- bind_rows(results, r0, r1)
  }

  list(results = results, stack = stack, models = models)
}

fit_direct_gradient <- function(stack, transition, outcome, spec_name) {
  d <- stack %>%
    mutate(
      t_1_12 = as.numeric(treated == 1 & proximity_window == "1-12m"),
      t_13_24 = as.numeric(treated == 1 & proximity_window == "13-24m"),
      t_25_60 = as.numeric(treated == 1 & proximity_window == "25-60m")
    )

  if (n_distinct(d$proximity_window) < 2L) {
    return(list(model = NULL, coefficients = tibble(), contrasts = tibble()))
  }

  m <- safe_feols(
    change_z ~ t_1_12 + t_13_24 + t_25_60 | match_set_id,
    data = d,
    weights = ~att_weight,
    vcov = ~pidp + match_set_id
  )

  coefs <- extract_fixest_all(
    m,
    paste0(transition, "_", spec_name, "_direct_gradient"),
    outcome,
    paste0(transition, " objective direct proximity gradient")
  )

  contrasts <- bind_rows(
    linear_contrast(
      m, c(t_1_12 = 1, t_13_24 = -1),
      "1-12m minus 13-24m", outcome,
      paste0(transition, " objective direct proximity gradient")
    ),
    linear_contrast(
      m, c(t_1_12 = 1, t_25_60 = -1),
      "1-12m minus 25-60m", outcome,
      paste0(transition, " objective direct proximity gradient")
    ),
    linear_contrast(
      m, c(t_13_24 = 1, t_25_60 = -1),
      "13-24m minus 25-60m", outcome,
      paste0(transition, " objective direct proximity gradient")
    )
  ) %>%
    mutate(specification = spec_name)

  list(model = m, coefficients = coefs, contrasts = contrasts)
}

fit_treated_only_gradient <- function(df, outcome_col, outcome_label, transition,
                                      group_id) {
  d <- df %>%
    filter(!is.na(.data[[outcome_col]])) %>%
    mutate(
      proximity_window = factor(
        proximity_window,
        levels = c("25-60m", "13-24m", "1-12m")
      )
    )

  if (nrow(d) == 0L || n_distinct(d$proximity_window) < 2L) return(tibble())

  f <- stats::as.formula(
    paste0(outcome_col, " ~ proximity_window | ", group_id)
  )
  m <- safe_feols(f, data = d, vcov = ~pidp)
  extract_fixest_all(
    m, paste0(transition, "_treated_only_within_relationship_gradient"),
    outcome_label,
    paste0(transition, " auxiliary treated-only proximity gradient")
  )
}

  fit_treated_only_gradient(
    cohab_treated_landmarks, "par_ghqwb_recent_change_z",
    "GHQ well-being (-GHQ12)", "cohabitation", "lat_episode_id"
  )
)

scan_wave_labels <- function(path, wave, wave_n) {
  d <- tryCatch(haven::read_dta(path, n_max = 1L), error = function(e) NULL)
  if (is.null(d)) return(tibble())
  nms <- names(d)
  labs <- vapply(
    d,
    function(x) {
      z <- attr(x, "label", exact = TRUE)
      if (is.null(z)) "" else as.character(z)
    },
    character(1)
  )
  tibble(
    wave = wave,
    wave_n = wave_n,
    file = path,
    variable = nms,
    stem = strip_wave_prefix(nms),
    label = labs,
    label_lower = tolower(labs)
  )
}

  function(file, wave, wave_n) scan_wave_labels(file, wave, wave_n)
)

# Read a candidate stem across all waves and merge to cohab_panel to determine
# whether it actually has substantive responses among cohabiters.
read_candidate_stem <- function(stem) {
  map2_dfr(wave_index$file, wave_index$wave, function(path, wave) {
    nms <- safe_read_names(path)
    pidp_var <- resolve_wave_variable(nms, wave, "pidp")
    target_var <- resolve_wave_variable(nms, wave, stem)
    if (is.na(pidp_var) || is.na(target_var)) return(tibble())
    dat <- haven::read_dta(path, col_select = any_of(c(pidp_var, target_var)))
    tibble(
      wave = wave,
      pidp = suppressWarnings(as.numeric(dat[[pidp_var]])),
      marriage_expect_raw = suppressWarnings(as.numeric(dat[[target_var]]))
    )
  })
}

      cohab_rows_with_codes_1_to_5 = sum(substantive_1_5, na.rm = TRUE),

        chk$relationship_spell_id[substantive_1_5]

      observed_min = ifelse(any(substantive_1_5), min(vals[substantive_1_5]), NA),
      observed_max = ifelse(any(substantive_1_5), max(vals[substantive_1_5]), NA)
    )
  )
}
