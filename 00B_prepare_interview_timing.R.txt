# =============================================================================
# BHPS + UKHLS
# FIRST-MARRIAGE CALENDAR-TIME DIAGNOSTIC
#
# PURPOSE
# -------
# Diagnose whether nominal wave-based event time (T-5,...,T0) adequately
# represents elapsed calendar time in the first-marriage study.
#
# The script:
#
#   1. Re-reads BHPS and UKHLS INDRESP files and extracts interview day/month/year.
#   2. Audits which interview-date variables are available in every wave.
#   3. Describes actual elapsed months between consecutive interviews.
#   4. For first-observed marriers, compares nominal T-k with actual months
#      before the T0 interview.
#   5. Quantifies how often nominal T-1 is actually 9-15, >15-21, or >21
#      months before T0, and provides analogous diagnostics for T-2,...,T-5.
#   6. Re-evaluates control support using CALENDAR-TIME T0 calipers of
#      +/-3, +/-6, and +/-12 months, without requiring the same wave number,
#      while retaining exact sex and age +/-1-year matching.
#   7. Optionally reads the Understanding Society marital/cohabitation history
#      file (phistory_long) and evaluates completeness of actual first-marriage
#      start month/year and whether that date falls between T-1 and T0.
#
# IMPORTANT
# ---------
# This is a DIAGNOSTIC. It does not overwrite the existing wave-based analysis.
# Its purpose is to determine the preferred calendar-time specification before
# the main models are rewritten.
#
# Interview date documentation:
#   UKHLS derived interview month/year:
#   https://www.understandingsociety.ac.uk/documentation/mainstage/variables/intdatm_dv/
#
# Marital/cohabitation histories:
#   phistory_long contains status, starty, startm, start_if, etc.
#   https://www.understandingsociety.ac.uk/documentation/mainstage/user-guides/
#
# =============================================================================

read_data_file <- function(
    path
) {

  ext <- tolower(
    tools::file_ext(
      path
    )
  )

  if (ext == "dta") {
    return(
      haven::read_dta(
        path
      )
    )
  }

  if (ext == "sav") {
    return(
      haven::read_sav(
        path
      )
    )
  }

  stop(
    "Unsupported file extension: ",
    ext
  )
}


find_indresp_file <- function(
    root,
    survey,
    wave_letter
) {

  survey <- match.arg(
    survey,
    c(
      "BHPS",
      "UKHLS"
    )
  )

  prefixes <- if (
    survey == "BHPS"
  ) {
    c(
      wave_letter,
      paste0(
        "b",
        wave_letter
      )
    )
  } else {
    wave_letter
  }

  patterns <- paste0(
    "^",
    prefixes,
    "_indresp\\.(dta|sav)$"
  )

  all_files <- list.files(
    root,
    recursive = TRUE,
    full.names = TRUE,
    ignore.case = TRUE
  )

  hits <- all_files[
    Reduce(
      `|`,
      lapply(
        patterns,
        function(p) {
          grepl(
            p,
            basename(
              all_files
            ),
            ignore.case = TRUE
          )
        }
      )
    )
  ]

  if (length(hits) == 0L) {
    stop(
      "No INDRESP file found for ",
      survey,
      " wave ",
      wave_letter,
      " under:\n",
      root
    )
  }

  if (length(hits) > 1L) {
    stop(
      "More than one INDRESP file found for ",
      survey,
      " wave ",
      wave_letter,
      ":\n",
      paste(
        hits,
        collapse = "\n"
      )
    )
  }

  hits
}


prefix_from_indresp_filename <- function(
    path
) {

  nm <- basename(
    path
  )

  stringr::str_match(
    nm,
    "^(.+?)_indresp\\.(dta|sav)$"
  )[, 2]
}


# Flexible finder because BHPS legacy variables can appear with different
# prefix conventions across releases.

find_variable_flexible <- function(
    dat,
    file_prefix,
    basenames,
    required = TRUE
) {

  short_prefix <- sub(
    "^b",
    "",
    file_prefix
  )

  candidates <- unique(
    unlist(
      lapply(
        basenames,
        function(v) {
          c(
            paste0(
              file_prefix,
              "_",
              v
            ),
            paste0(
              file_prefix,
              v
            ),
            paste0(
              short_prefix,
              "_",
              v
            ),
            paste0(
              short_prefix,
              v
            ),
            v
          )
        }
      )
    )
  )

  hit <- candidates[
    candidates %in%
      names(
        dat
      )
  ]

  if (length(hit) == 0L) {

    if (required) {
      stop(
        "Could not find any candidate variable:\n",
        paste(
          candidates,
          collapse = ", "
        )
      )
    }

    return(
      NA_character_
    )
  }

  hit[1L]
}


clean_integer <- function(
    x
) {

  z <- suppressWarnings(
    as.integer(
      as.numeric(
        x
      )
    )
  )

  z[
    !is.na(z) &
      z < 0
  ] <- NA_integer_

  z
}


clean_day <- function(
    x
) {

  z <- clean_integer(
    x
  )

  z[
    !is.na(z) &
      !z %in%
        1:31
  ] <- NA_integer_

  z
}


clean_month <- function(
    x
) {

  z <- clean_integer(
    x
  )

  z[
    !is.na(z) &
      !z %in%
        1:12
  ] <- NA_integer_

  z
}


clean_year <- function(
    x
) {

  z <- clean_integer(
    x
  )

  # Accommodate occasional two-digit legacy year fields.
  z[
    !is.na(z) &
      z >= 0L &
      z <= 30L
  ] <-
    2000L +
    z[
      !is.na(z) &
        z >= 0L &
        z <= 30L
    ]

  z[
    !is.na(z) &
      z >= 31L &
      z <= 99L
  ] <-
    1900L +
    z[
      !is.na(z) &
        z >= 31L &
        z <= 99L
    ]

  z[
    !is.na(z) &
      (
        z < 1980L |
          z > 2035L
      )
  ] <- NA_integer_

  z
}


make_interview_date <- function(
    year,
    month,
    day
) {

  out <- rep(
    as.Date(
      NA
    ),
    length(
      year
    )
  )

  complete_ym <- !is.na(
    year
  ) &
    !is.na(
      month
    )

  exact <- complete_ym &
    !is.na(
      day
    )

  if (any(
    exact
  )) {

    exact_chr <- sprintf(
      "%04d-%02d-%02d",
      year[
        exact
      ],
      month[
        exact
      ],
      day[
        exact
      ]
    )

    out[
      exact
    ] <- suppressWarnings(
      as.Date(
        exact_chr
      )
    )
  }

  fallback <- complete_ym &
    is.na(
      out
    )

  if (any(
    fallback
  )) {

    out[
      fallback
    ] <- as.Date(
      sprintf(
        "%04d-%02d-15",
        year[
          fallback
        ],
        month[
          fallback
        ]
      )
    )
  }

  out
}


month_index <- function(
    year,
    month
) {

  ifelse(
    !is.na(
      year
    ) &
      !is.na(
        month
      ),
    year *
      12L +
      month,
    NA_real_
  )
}


q_safe <- function(
    x,
    prob
) {

  x <- x[
    is.finite(
      x
    )
  ]

  if (length(x) == 0L) {
    return(
      NA_real_
    )
  }

  as.numeric(
    quantile(
      x,
      probs =
        prob,
      na.rm =
        TRUE,
      names =
        FALSE
    )
  )
}

# =============================================================================
# 6. READ INTERVIEW MONTH/YEAR FROM ONE WAVE
# =============================================================================

read_timing_wave <- function(
    survey,
    wave_num,
    root
) {

  wave_letter <- letters[
    wave_num
  ]

  f <- find_indresp_file(
    root =
      root,
    survey =
      survey,
    wave_letter =
      wave_letter
  )

  file_prefix <- prefix_from_indresp_filename(
    f
  )

  message(
    "Timing read: ",
    survey,
    " wave ",
    wave_num,
    " (",
    wave_letter,
    "): ",
    basename(
      f
    )
  )

  dat <- read_data_file(
    f
  )

  pidp_var <- find_variable_flexible(
    dat,
    file_prefix,
    "pidp",
    required =
      TRUE
  )

  day_var <- find_variable_flexible(
    dat,
    file_prefix,
    interview_date_candidates[[
      survey
    ]][[
      "day"
    ]],
    required =
      FALSE
  )

  month_var <- find_variable_flexible(
    dat,
    file_prefix,
    interview_date_candidates[[
      survey
    ]][[
      "month"
    ]],
    required =
      FALSE
  )

  year_var <- find_variable_flexible(
    dat,
    file_prefix,
    interview_date_candidates[[
      survey
    ]][[
      "year"
    ]],
    required =
      FALSE
  )

  n <- nrow(
    dat
  )

  interview_day <- if (
    is.na(
      day_var
    )
  ) {
    rep(
      NA_integer_,
      n
    )
  } else {
    clean_day(
      dat[[
        day_var
      ]]
    )
  }

  interview_month <- if (
    is.na(
      month_var
    )
  ) {
    rep(
      NA_integer_,
      n
    )
  } else {
    clean_month(
      dat[[
        month_var
      ]]
    )
  }

  interview_year <- if (
    is.na(
      year_var
    )
  ) {
    rep(
      NA_integer_,
      n
    )
  } else {
    clean_year(
      dat[[
        year_var
      ]]
    )
  }

  timing <- tibble(
    pidp =
      dat[[
        pidp_var
      ]],

    survey =
      survey,

    wave_num =
      wave_num,

    wave_letter =
      wave_letter,

    interview_year =
      interview_year,

    interview_month =
      interview_month,

    interview_day =
      interview_day,

    interview_month_index =
      month_index(
        interview_year,
        interview_month
      ),

    interview_date =
      make_interview_date(
        interview_year,
        interview_month,
        interview_day
      ),

    interview_date_exact =
      !is.na(
        interview_year
      ) &
        !is.na(
          interview_month
        ) &
        !is.na(
          interview_day
        )
  )

  audit <- tibble(
    survey =
      survey,
    wave_num =
      wave_num,
    wave_letter =
      wave_letter,
    file =
      basename(
        f
      ),
    file_prefix =
      file_prefix,
    pidp_var =
      pidp_var,
    day_var_found =
      day_var,
    month_var_found =
      month_var,
    year_var_found =
      year_var,
    n_indresp_rows =
      n,
    n_day_observed =
      sum(
        !is.na(
          interview_day
        )
      ),
    pct_day_observed =
      100 *
        mean(
          !is.na(
            interview_day
          )
        ),
    n_year_observed =
      sum(
        !is.na(
          interview_year
        )
      ),
    pct_year_observed =
      100 *
        mean(
          !is.na(
            interview_year
          )
        ),
    n_month_observed =
      sum(
        !is.na(
          interview_month
        )
      ),
    pct_month_observed =
      100 *
        mean(
          !is.na(
            interview_month
          )
        ),
    n_complete_month_year =
      sum(
        !is.na(
          interview_year
        ) &
          !is.na(
            interview_month
          )
      ),
    pct_complete_month_year =
      100 *
        mean(
          !is.na(
            interview_year
          ) &
            !is.na(
              interview_month
            )
        )
  )

  rm(
    dat
  )

  list(
    timing =
      timing,
    audit =
      audit
  )
}

# =============================================================================
# 9. ACTUAL ELAPSED MONTHS BETWEEN CONSECUTIVE INTERVIEWS
# =============================================================================

consecutive_intervals <- timing_panel |>
  arrange(
    pidp,
    survey,
    wave_num
  ) |>
  group_by(
    pidp,
    survey
  ) |>
  mutate(

    previous_wave =
      lag(
        wave_num
      ),

    previous_year =
      lag(
        interview_year
      ),

    previous_month =
      lag(
        interview_month
      ),

    previous_month_index =
      lag(
        interview_month_index
      ),

    consecutive_wave =
      !is.na(
        previous_wave
      ) &
        wave_num ==
          previous_wave +
            1L,

    months_since_previous_interview =
      if_else(
        consecutive_wave &
          !is.na(
            interview_month_index
          ) &
          !is.na(
            previous_month_index
          ),
        interview_month_index -
          previous_month_index,
        NA_real_
      ),

    same_calendar_year_as_previous =
      consecutive_wave &
        !is.na(
          interview_year
        ) &
        !is.na(
          previous_year
        ) &
        interview_year ==
          previous_year
  ) |>
  ungroup() |>
  filter(
    consecutive_wave
  ) |>
  mutate(
    interval_category =
      case_when(

        is.na(
          months_since_previous_interview
        ) ~
          "Missing month/year",

        months_since_previous_interview < 9 ~
          "<9 months",

        months_since_previous_interview <= 15 ~
          "9-15 months",

        months_since_previous_interview <= 21 ~
          ">15-21 months",

        TRUE ~
          ">21 months"
      )
  )

write_csv(
  consecutive_intervals,
  file.path(
    output_dir,
    "consecutive_interview_intervals_person_level.csv"
  )
)


consecutive_interval_summary <- consecutive_intervals |>
  group_by(
    survey
  ) |>
  summarise(
    consecutive_wave_pairs =
      n(),
    complete_month_year_pairs =
      sum(
        !is.na(
          months_since_previous_interview
        )
      ),
    pct_complete_month_year_pairs =
      100 *
        mean(
          !is.na(
            months_since_previous_interview
          )
        ),
    mean_months =
      mean(
        months_since_previous_interview,
        na.rm =
          TRUE
      ),
    sd_months =
      sd(
        months_since_previous_interview,
        na.rm =
          TRUE
      ),
    p10_months =
      q_safe(
        months_since_previous_interview,
        0.10
      ),
    p25_months =
      q_safe(
        months_since_previous_interview,
        0.25
      ),
    median_months =
      q_safe(
        months_since_previous_interview,
        0.50
      ),
    p75_months =
      q_safe(
        months_since_previous_interview,
        0.75
      ),
    p90_months =
      q_safe(
        months_since_previous_interview,
        0.90
      ),
    pct_9_to_15_months =
      100 *
        mean(
          months_since_previous_interview >= 9 &
            months_since_previous_interview <= 15,
          na.rm =
            TRUE
        ),
    pct_over_15_to_21_months =
      100 *
        mean(
          months_since_previous_interview > 15 &
            months_since_previous_interview <= 21,
          na.rm =
            TRUE
        ),
    pct_over_21_months =

# =============================================================================
# 14. CALENDAR-TIME CONTROL SUPPORT:
#     SAME SURVEY + EXACT SEX + AGE +/-1 YEAR + DATE CALIPER
# =============================================================================
#
# Important:
# The join below intentionally does NOT require the same wave number.
# A control anchor from another wave is allowed if its interview date is close
# to the treated person's T0 date.
#
# The current primary-control RDS still ensures that each candidate control
# anchor:
#   - is unmarried through its focal T0,
#   - has complete T-5:T0 status,
#   - has no previously observed marriage by that focal T0.
#
# This therefore gives a conservative diagnostic of calendar-time support.
# =============================================================================

treated_calendar_ready <- treated_T0_demo |>
  filter(
    !is.na(
      treated_month_index
    ),
    !is.na(
      treated_age_key
    ),
    !is.na(
      treated_sex
    )
  ) |>
  tidyr::crossing(
    age_shift =
      (-age_caliper_years):age_caliper_years
  ) |>
  mutate(
    allowed_control_age_key =
      treated_age_key +
        age_shift
  )


control_calendar_ready <- control_anchor_demo |>
  filter(
    !is.na(
      control_month_index
    ),
    !is.na(
      control_age_key
    ),
    !is.na(
      control_sex
    )
  )


calendar_candidates <- treated_calendar_ready |>
  inner_join(
    control_calendar_ready,
    by = c(
      "survey",
      "treated_sex" =
        "control_sex",
      "allowed_control_age_key" =
        "control_age_key"
    ),
    relationship =
      "many-to-many"
  ) |>
  filter(
    treated_pidp !=
      control_pidp
  ) |>
  mutate(
    absolute_T0_month_difference =
      abs(
        treated_month_index -
          control_month_index
      )
  )

discover_phistory <- function() {

  explicit_ok <-
    length(
      phistory_path
    ) == 1L &&
    !is.na(
      phistory_path
    ) &&
    nzchar(
      phistory_path
    ) &&
    file.exists(
      phistory_path
    )

  if (explicit_ok) {
    return(
      normalizePath(
        phistory_path,
        winslash =
          "/",
        mustWork =
          TRUE
      )
    )
  }

  search_roots <- unique(
    c(
      common_stata_dir,
      dirname(
        common_stata_dir
      )
    )
  )

  hits <- unique(
    unlist(
      lapply(
        search_roots,
        function(root) {

          if (!dir.exists(
            root
          )) {
            return(
              character(0)
            )
          }

          list.files(
            root,
            pattern =
              "^phistory_long\\.(dta|sav)$",
            recursive =
              TRUE,
            full.names =
              TRUE,
            ignore.case =
              TRUE
          )
        }
      )
    )
  )

  if (length(hits) == 0L) {
    return(
      NA_character_
    )
  }

  hits <- sort(
    hits
  )

  if (length(hits) > 1L) {
    warning(
      "More than one phistory_long file found. Using the first:\n",
      hits[1L],
      "\nAll candidates:\n",
      paste(
        hits,
        collapse =
          "\n"
      )
    )
  }

  hits[1L]
}


phistory_found_path <- discover_phistory()

        "but actual marriage-date completeness cannot be assessed. ",
        "Download the marital/cohabitation history dataset or set phistory_path manually."
      )
    )
)
