################################################################################
# UKHLS WAVES 3-15
# INCIDENT STEADY NON-RESIDENTIAL (LAT) RELATIONSHIP FORMATION
# PSYCHOLOGICAL TRAJECTORIES BEFORE AND ACROSS RELATIONSHIP FORMATION
#
# COMPLEMENTARY / EXPLORATORY ANALYSIS FOR THE
# SELECTION - DIFFERENTIATION - DEVELOPMENT - SORTING FRAMEWORK
#
# PURPOSE
# -------
# This workflow addresses a temporal gap left by the cohabitation-duration
# analyses. It asks whether psychological functioning changes BEFORE formation
# of a steady non-residential relationship and/or ACROSS formation of that
# relationship.
#
# It estimates two distinct quantities:
#
#   1. ABSOLUTE WITHIN-PERSON CHANGE among incident LAT formers.
#      This directly answers whether future relationship formers themselves
#      improve, remain stable, or deteriorate.
#
#   2. MATCHED DIFFERENTIAL CHANGE relative to contemporaneous respondents who
#      are observed as single and without a steady non-residential partner over
#      the corresponding risk interval.
#
# The workflow deliberately keeps these quantities separate. A favorable
# treated-control difference can arise from improvement among treated people,
# deterioration among controls, or both.
#
# INFERENTIAL STATUS
# ------------------
# These analyses are complementary/exploratory. They were motivated by the
# pattern observed in the current study and MUST NOT be described as part of
# the original confirmatory marriage-proximity hypothesis family. They can
# clarify the mechanism/timing underlying the framework but cannot rescue a
# nonsignificant confirmatory result.
#
# PRIMARY INCIDENT-LAT EVENT DEFINITION
# -------------------------------------
# A respondent must have two consecutive VALID NCRR1 observations such that:
#
#   prior valid observation:  status == "single" AND NCRR1 == 2
#                             (no steady non-resident partner)
#   current valid observation: status == "single" AND NCRR1 == 1
#                             (steady non-resident partner)
#
# At the first positive NCRR1 observation, the reported relationship start
# month/year (NCRRM/NCRRY4) must:
#
#   - be valid;
#   - occur AFTER the prior NCRR1=2 interview;
#   - occur ON OR BEFORE the first NCRR1=1 interview;
#   - be no more than 12 months after the prior unpartnered interview;
#   - be no more than 12 months before the first positive interview;
#   - and the two valid NCRR1 interviews must be no more than 18 months apart.
#
# These rules identify a clean interval in which an incident steady LAT
# relationship is reported to have started between an observed unpartnered
# interview and an observed partnered interview.
#
# PRIMARY TEMPORAL LANDMARKS
# --------------------------
#   pre25_36          = closest psychological observation 25-36 months before
#                       the reported LAT relationship start
#   pre13_24          = closest psychological observation 13-24 months before
#                       the reported LAT relationship start
#   last_unpartnered  = the exact prior interview with single + NCRR1=2;
#                       by design 1-12 months before reported relationship start
#   first_LAT         = the exact first positive NCRR1 interview;
#                       by design 0-12 months after reported relationship start
#
# KEY CONTRASTS
# -------------
# A. PRE-FORMATION CHANGE:
#      pre25_36 -> last_unpartnered
#    Tests whether future LAT formers were already improving before the reported
#    relationship began. A favorable absolute change is consistent with dynamic
#    psychological selection, but is not causal evidence.
#
# B. FORMATION-ASSOCIATED CHANGE:
#      last_unpartnered -> first_LAT
#    Tests whether psychological functioning changes as an incident steady LAT
#    relationship forms. A favorable absolute and relative change is consistent
#    with an early relationship-formation gain.
#
# C. TOTAL EXTENDED CHANGE:
#      pre25_36 -> first_LAT
#    Describes the combined longer trajectory.
#
# CONTROL DESIGN
# --------------
# PRIMARY matched controls are NOT required to remain unpartnered indefinitely.
# For each treated event they must:
#
#   - be single + NCRR1=2 at a comparable baseline interview;
#   - be observed single + NCRR1=2 again at the treated event's positive wave;
#   - have no observed steady-partner or co-residential/formal transition
#     between those two observations;
#   - have no later reported LAT start date indicating that a relationship
#     actually began before that control follow-up interview.
#
# Matching uses exact baseline UKHLS wave, exact baseline interview year, exact
# sex, age +/- 1 year, and baseline interview month within +/- 3 months.
# The control pseudo-event month preserves the treated person's elapsed months
# from baseline interview to reported relationship start.
#
# Controls may form relationships AFTER the matched follow-up observation. This
# avoids defining the primary comparison group by future never-partnered status.
#
# STABLE-UNPARTNERED SENSITIVITY
# ------------------------------
# A secondary descriptive sensitivity restricts controls to those with observed
# unpartnered follow-up through 24 months after the pseudo-event. This is useful
# substantively but explicitly conditions on future relationship history and is
# therefore NOT the primary counterfactual.
#
# FIRST-LIKE HISTORY SENSITIVITY
# ------------------------------
# A second sensitivity restricts BOTH treated respondents and controls to
# baselines with no prior observed partnership of any type and no known dated
# PHISTORY co-residential/formal union beginning on/before baseline. Undated
# PHISTORY spells do not exclude observations because their temporal position is
# unknown.
#
# OUTCOMES
# --------
# life_sat : higher = better
# ghq_wb   : reverse of GHQ-12 distress, ghq_wb = -ghq12; higher = better
#
# Thus, positive changes consistently indicate more favorable psychological
# functioning in this workflow.
################################################################################

normalize_pidp_key <- function(x) {
  z <- x
  if (
    inherits(z, "haven_labelled") ||
      inherits(z, "labelled") ||
      inherits(z, "haven_labelled_spss")
  ) z <- unclass(z)
  if (is.factor(z)) z <- as.character(z)
  if (is.numeric(z) || is.integer(z)) {
    out <- ifelse(
      is.na(z), NA_character_,
      format(z, scientific = FALSE, trim = TRUE, digits = 22)
    )
    out <- sub("\\.0+$", "", out)
    return(out)
  }
  out <- trimws(as.character(z))
  out[out %in% c("", "NA", "NaN")] <- NA_character_
  sub("\\.0+$", "", out)
}

as_numeric_clean <- function(x) {
  z <- x
  if (
    inherits(z, "haven_labelled") ||
      inherits(z, "labelled") ||
      inherits(z, "haven_labelled_spss")
  ) z <- unclass(z)
  if (is.factor(z)) z <- as.character(z)
  suppressWarnings(as.numeric(z))
}

stata_month_from_year_month <- function(year, month_num) {
  y <- suppressWarnings(as.integer(year))
  m <- suppressWarnings(as.integer(month_num))
  ifelse(
    is.na(y) | is.na(m) | m < 1L | m > 12L,
    NA_integer_,
    as.integer((y - 1960L) * 12L + (m - 1L))
  )
}

stata_month_to_year <- function(m) {
  ifelse(is.na(m), NA_integer_, as.integer(1960L + floor(m / 12)))
}

stata_month_to_monthnum <- function(m) {
  ifelse(is.na(m), NA_integer_, as.integer((m %% 12L) + 1L))
}

pick_existing_name <- function(all_names, candidates, label, file_path) {
  hit <- candidates[candidates %in% all_names]
  if (length(hit) == 0L) {
    stop(
      "Could not find ", label, " in ", basename(file_path), ". Tried: ",
      paste(candidates, collapse = ", ")
    )
  }
  hit[[1L]]
}

pick_optional_name <- function(all_names, candidates) {
  hit <- candidates[candidates %in% all_names]
  if (length(hit) == 0L) NA_character_ else hit[[1L]]
}

first_future_from_candidate <- function(candidate) {
  dplyr::lead(rev(cummin(rev(candidate))), default = Inf)
}

normal_ci <- function(est, se, level = 0.95) {
  crit <- qnorm(1 - (1 - level) / 2)
  c(est - crit * se, est + crit * se)
}

safe_sd <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) < 2L) return(NA_real_)
  s <- sd(z)
  if (!is.finite(s) || s <= 0) NA_real_ else s
}

safe_weighted_mean <- function(x, w) {
  ok <- is.finite(x) & is.finite(w) & w > 0
  if (!any(ok)) return(NA_real_)
  weighted.mean(x[ok], w[ok])
}

write_csv_quiet <- function(x, filename) {
  readr::write_csv(x, file.path(results_dir, filename))
}

read_ncrr_wave <- function(wave_num, prefix) {
  file_path <- file.path(ukhls_dir, paste0(prefix, "_indresp.dta"))
  if (!file.exists(file_path)) {
    stop("Missing UKHLS INDRESP file for Wave ", wave_num, ": ", file_path)
  }

  header <- haven::read_dta(file_path, n_max = 0)
  nm <- names(header)

  pidp_nm <- pick_existing_name(
    nm, c("pidp", paste0(prefix, "_pidp")), "pidp", file_path
  )
  ncrr1_nm <- pick_existing_name(
    nm, c(paste0(prefix, "_ncrr1"), "ncrr1"), "NCRR1", file_path
  )
  inty_nm <- pick_existing_name(
    nm, c(paste0(prefix, "_intdaty_dv"), "intdaty_dv"),
    "interview year", file_path
  )
  intm_nm <- pick_existing_name(
    nm, c(paste0(prefix, "_intdatm_dv"), "intdatm_dv"),
    "interview month", file_path
  )

  ncrrm_nm <- pick_optional_name(
    nm, c(paste0(prefix, "_ncrrm"), "ncrrm")
  )
  ncrry4_nm <- pick_optional_name(
    nm, c(paste0(prefix, "_ncrry4"), "ncrry4")
  )

  selected <- unique(na.omit(c(
    pidp_nm, ncrr1_nm, inty_nm, intm_nm, ncrrm_nm, ncrry4_nm
  )))

  dat <- haven::read_dta(file_path, col_select = dplyr::all_of(selected))

  get_optional <- function(v) {
    if (is.na(v) || !v %in% names(dat)) {
      rep(NA_real_, nrow(dat))
    } else {
      as_numeric_clean(dat[[v]])
    }
  }

  out <- tibble(
    pidp_key = normalize_pidp_key(dat[[pidp_nm]]),
    wave_num = as.integer(wave_num),
    wave_prefix = prefix,
    ncrr_year_raw = suppressWarnings(as.integer(as_numeric_clean(dat[[inty_nm]]))),
    ncrr_month_raw = suppressWarnings(as.integer(as_numeric_clean(dat[[intm_nm]]))),
    ncrr1_raw = suppressWarnings(as.integer(as_numeric_clean(dat[[ncrr1_nm]]))),
    ncrrm_raw = suppressWarnings(as.integer(get_optional(ncrrm_nm))),
    ncrry4_raw = suppressWarnings(as.integer(get_optional(ncrry4_nm)))
  ) |>
    mutate(
      ncrr_year_raw = if_else(
        !is.na(ncrr_year_raw) & ncrr_year_raw >= 1900L & ncrr_year_raw <= 2100L,
        ncrr_year_raw, NA_integer_
      ),
      ncrr_month_raw = if_else(
        !is.na(ncrr_month_raw) & ncrr_month_raw >= 1L & ncrr_month_raw <= 12L,
        ncrr_month_raw, NA_integer_
      ),
      ncrr1 = if_else(ncrr1_raw %in% c(1L, 2L), ncrr1_raw, NA_integer_),
      ncrrm = if_else(
        !is.na(ncrrm_raw) & ncrrm_raw >= 1L & ncrrm_raw <= 12L,
        ncrrm_raw, NA_integer_
      ),
      ncrry4 = if_else(
        !is.na(ncrry4_raw) & ncrry4_raw >= 1900L & ncrry4_raw <= 2100L,
        ncrry4_raw, NA_integer_
      ),
      ncrr_month_index_raw = stata_month_from_year_month(ncrr_year_raw, ncrr_month_raw),
      lat_start_month_index = stata_month_from_year_month(ncrry4, ncrrm)
    )

  out
}

ncrr_raw <- map2_dfr(UKHLS_WAVES, UKHLS_PREFIX, read_ncrr_wave) |>
  filter(!is.na(pidp_key))

if (anyDuplicated(ncrr_raw[c("pidp_key", "wave_num")]) > 0L) {
  stop("NCRR extraction is not unique by pidp x wave.")
}

ncrr_availability_audit <- ncrr_raw |>
  group_by(wave_num, wave_prefix) |>
  summarise(
    rows = n(),
    valid_ncrr1 = sum(ncrr1 %in% c(1L, 2L), na.rm = TRUE),
    ncrr1_yes = sum(ncrr1 == 1L, na.rm = TRUE),
    ncrr1_no = sum(ncrr1 == 2L, na.rm = TRUE),
    positive_with_valid_reported_start = sum(
      ncrr1 == 1L & !is.na(lat_start_month_index), na.rm = TRUE
    ),
    pct_positive_with_valid_reported_start = if_else(
      ncrr1_yes > 0,
      100 * positive_with_valid_reported_start / ncrr1_yes,
      NA_real_
    ),
    .groups = "drop"
  )

write_csv_quiet(ncrr_availability_audit, "01_NCRR_LAT_start_availability_by_wave.csv")

  transmute(
    pidp_key = normalize_pidp_key(pidp),
    survey = as.character(survey),
    wave_num = suppressWarnings(as.integer(wave_num)),

    life_sat = suppressWarnings(as.numeric(life_sat)),
    age = suppressWarnings(as.numeric(age)),
    sex = as.character(sex)
  ) |>
  filter(
    survey == "UKHLS",
    wave_num %in% UKHLS_WAVES,
    !is.na(pidp_key)
  )

timing <- timing_raw |>
  transmute(
    pidp_key = normalize_pidp_key(pidp),
    survey = as.character(survey),
    wave_num = suppressWarnings(as.integer(wave_num)),
    interview_year = suppressWarnings(as.integer(interview_year)),
    interview_month = suppressWarnings(as.integer(interview_month))
  ) |>
  filter(
    survey == "UKHLS",
    wave_num %in% UKHLS_WAVES,
    !is.na(pidp_key)
  )

if (anyDuplicated(combined[c("pidp_key", "wave_num")]) > 0L) {
  stop("Combined UKHLS person-wave panel is not unique by pidp x wave.")
}
if (anyDuplicated(timing[c("pidp_key", "wave_num")]) > 0L) {
  stop("UKHLS timing panel is not unique by pidp x wave.")
}

wave_panel <- combined |>
  left_join(
    timing |> select(-survey),
    by = c("pidp_key", "wave_num"),
    relationship = "one-to-one"
  ) |>
  left_join(
    ncrr_raw |>
      select(
        pidp_key, wave_num, wave_prefix,
        ncrr_year_raw, ncrr_month_raw, ncrr_month_index_raw,
        ncrr1_raw, ncrr1, ncrrm, ncrry4, lat_start_month_index
      ),
    by = c("pidp_key", "wave_num"),
    relationship = "one-to-one"
  ) |>
  mutate(
    interview_month_index = stata_month_from_year_month(interview_year, interview_month),
    ghq12 = if_else(ghq12 >= 0 & ghq12 <= 36, ghq12, NA_real_),
    life_sat = if_else(life_sat >= 1 & life_sat <= 7, life_sat, NA_real_),
    ghq_wb = -ghq12
  ) |>
  filter(
    !is.na(interview_month_index),
    !is.na(age), age >= MIN_AGE, age <= MAX_AGE,
    !is.na(sex)
  )

# Same-interview timing consistency check, inherited from the existing NCRR1 V3 logic.
same_wave_timing_audit <- wave_panel |>
  summarise(
    rows = n(),
    rows_with_both_month_sources = sum(
      !is.na(interview_month_index) & !is.na(ncrr_month_index_raw)
    ),
    exact_month_matches = sum(
      !is.na(interview_month_index) & !is.na(ncrr_month_index_raw) &
        interview_month_index == ncrr_month_index_raw
    ),
    month_mismatches = sum(
      !is.na(interview_month_index) & !is.na(ncrr_month_index_raw) &
        interview_month_index != ncrr_month_index_raw
    )
  )
write_csv_quiet(same_wave_timing_audit, "02_same_wave_timing_audit.csv")

  filter(

  )

  ) |>

build_control_candidates <- function(
    treated_events,
    require_stable_24m = FALSE,
    require_firstlike_history = FALSE
) {
  t <- treated_events

  if (require_firstlike_history) {
    t <- t |>
      filter(baseline_no_known_prior_union_and_no_prior_LAT)
  }

  pool <- control_baselines_all
  if (require_firstlike_history) {
    pool <- pool |>
      filter(control_no_known_prior_union_and_no_prior_LAT)
  }

  d <- t |>
    inner_join(
      pool,
      by = c(
        "baseline_wave" = "control_baseline_wave",
        "baseline_year" = "control_baseline_year",
        "treated_sex" = "control_sex"
      ),
      relationship = "many-to-many"
    ) |>
    filter(
      control_pidp != treated_pidp,
      abs(control_age - treated_age) <= MATCH_AGE_CALIPER,
      abs(control_baseline_month - baseline_month) <= MATCH_BASELINE_MONTH_CALIPER
    ) |>
    left_join(
      followup_lookup,
      by = c("control_pidp", "positive_wave"),
      relationship = "many-to-one"
    ) |>
    mutate(
      control_followup_interval = control_followup_month - control_baseline_month,
      control_pseudo_event_month =
        control_baseline_month + event_offset_from_baseline,

      control_no_observed_partner_through_followup =
        is.na(control_first_future_any_partner_detection_month) |
        control_first_future_any_partner_detection_month > control_followup_month,

      # A later positive NCRR1 report can retrospectively date its relationship
      # start before the matched follow-up. Exclude such controls because they
      # may not truly have remained unpartnered through the comparison interval.
      control_no_reported_LAT_start_through_followup =
        is.na(control_first_future_lat_reported_start_month) |
        control_first_future_lat_reported_start_month > control_followup_month,
      control_remains_single_through_followup =
        is.na(control_first_future_non_single_month) |
        control_first_future_non_single_month > control_followup_month
    ) |>
    filter(
      !is.na(control_followup_month),
      control_followup_status == "single",
      control_followup_ncrr1 == 2L,
      control_followup_month > control_baseline_month,
      abs(control_followup_interval - baseline_to_positive_months) <=
        MATCH_FOLLOWUP_INTERVAL_CALIPER,
      control_pseudo_event_month > control_baseline_month,
      control_pseudo_event_month <= control_followup_month,
      control_no_observed_partner_through_followup,
      control_no_reported_LAT_start_through_followup,
      control_remains_single_through_followup
    )

  if (require_stable_24m) {
    d <- d |>
      mutate(
        stable_target_month =
          control_pseudo_event_month + STABLE_UNPARTNERED_HORIZON_MONTHS,
        stable_followup_observed =
          !is.na(control_last_valid_ncrr_month) &
          control_last_valid_ncrr_month >= stable_target_month,
        no_observed_partner_through_stable_horizon =
          is.na(control_first_future_any_partner_detection_month) |
          control_first_future_any_partner_detection_month > stable_target_month,
        no_reported_LAT_start_through_stable_horizon =
          is.na(control_first_future_lat_reported_start_month) |
          control_first_future_lat_reported_start_month > stable_target_month,
        no_coresident_detection_through_stable_horizon =
          is.na(control_first_future_coresident_month) |
          control_first_future_coresident_month > stable_target_month,
        remains_single_through_stable_horizon =
          is.na(control_first_future_non_single_month) |
          control_first_future_non_single_month > stable_target_month
      ) |>
      filter(
        stable_followup_observed,
        no_observed_partner_through_stable_horizon,
        no_reported_LAT_start_through_stable_horizon,
        no_coresident_detection_through_stable_horizon,
        remains_single_through_stable_horizon
      )
  }

  d <- d |>
    distinct(match_set_id, control_pidp, .keep_all = TRUE) |>
    group_by(match_set_id) |>
    mutate(n_controls = n()) |>
    ungroup() |>
    filter(n_controls >= MIN_CONTROLS_PER_SET)

  d
}

make_match_index <- function(control_candidates, treated_events, analysis_label) {
  retained_sets <- control_candidates |>
    distinct(match_set_id)

  t <- treated_events |>
    semi_join(retained_sets, by = "match_set_id") |>
    transmute(
      analysis_label = analysis_label,
      match_set_id,
      unit_pidp = treated_pidp,
      condition = "treated",
      event_month,
      baseline_wave,
      endpoint_wave = positive_wave,
      baseline_month,
      endpoint_month = positive_month
    )

  c <- control_candidates |>
    transmute(
      analysis_label = analysis_label,
      match_set_id,
      unit_pidp = control_pidp,
      condition = "unpartnered control",
      event_month = control_pseudo_event_month,
      baseline_wave,
      endpoint_wave = positive_wave,
      baseline_month = control_baseline_month,
      endpoint_month = control_followup_month
    )

  bind_rows(t, c) |>
    arrange(match_set_id, desc(condition == "treated"), unit_pidp)
}

panel_for_stack <- panel_features |>
  select(
    pidp_key, wave_num, interview_month_index, interview_year, interview_month,
    status, ncrr1, ghq12, ghq_wb, life_sat, age, sex

build_event_stack <- function(match_index) {
  if (nrow(match_index) == 0L) return(tibble())

  joined <- match_index |>
    inner_join(
      panel_for_stack,
      by = c("unit_pidp" = "pidp_key"),
      relationship = "many-to-many"
    ) |>
    mutate(rel_month = interview_month_index - event_month)

  # Select one observation in each distal pre-event window, closest to the
  # midpoint of that window.
  distal <- map_dfr(seq_len(nrow(PRE_WINDOWS)), function(i) {
    w <- PRE_WINDOWS[i, ]
    joined |>
      filter(
        rel_month >= w$lower_month,
        rel_month <= w$upper_month
      ) |>
      mutate(
        period = w$period,
        distance_to_target = abs(rel_month - w$target_month)
      ) |>
      group_by(analysis_label, match_set_id, unit_pidp, condition) |>
      arrange(distance_to_target, interview_month_index, .by_group = TRUE) |>
      slice_head(n = 1L) |>
      ungroup() |>
      select(-distance_to_target)
  })

  # Exact last observed unpartnered interview.
  last_unpartnered <- joined |>
    filter(
      wave_num == baseline_wave,
      interview_month_index == baseline_month
    ) |>
    mutate(period = "last_unpartnered")

  # Exact first observed steady LAT interview for treated respondents and the
  # corresponding still-unpartnered follow-up wave for controls.
  first_lat_or_control_followup <- joined |>
    filter(
      wave_num == endpoint_wave,
      interview_month_index == endpoint_month
    ) |>
    mutate(period = "first_LAT")

  bind_rows(distal, last_unpartnered, first_lat_or_control_followup) |>
    mutate(
      period = factor(period, levels = PERIOD_ORDER, ordered = TRUE)
    ) |>
    arrange(analysis_label, match_set_id, unit_pidp, period)
}

# 10. ABSOLUTE WITHIN-PERSON CHANGE AMONG INCIDENT LAT FORMERS
# =============================================================================

absolute_paired_change <- function(stack, outcome, contrast_row) {
  p0 <- contrast_row$period0[[1L]]
  p1 <- contrast_row$period1[[1L]]
  contrast_name <- contrast_row$contrast[[1L]]

  d <- stack |>
    filter(condition == "treated", period %in% c(p0, p1)) |>
    mutate(y = .data[[outcome]]) |>
    select(unit_pidp, period, y) |>
    filter(!is.na(y)) |>
    distinct(unit_pidp, period, .keep_all = TRUE) |>
    pivot_wider(names_from = period, values_from = y)

  if (!all(c(p0, p1) %in% names(d))) return(tibble())

  d <- d |>
    filter(!is.na(.data[[p0]]), !is.na(.data[[p1]])) |>
    mutate(change = .data[[p1]] - .data[[p0]])

  if (nrow(d) < 2L) return(tibble())

  est <- mean(d$change)
  se <- sd(d$change) / sqrt(nrow(d))
  ci <- normal_ci(est, se)
  sd0 <- get_baseline_sd(outcome)

  tibble(
    analysis = "absolute_within_person_treated",
    outcome = outcome,
    contrast = contrast_name,
    period0 = p0,
    period1 = p1,
    n_people = nrow(d),
    mean_period0 = mean(d[[p0]]),
    mean_period1 = mean(d[[p1]]),
    absolute_change = est,
    std_error = se,
    conf_low = ci[1],
    conf_high = ci[2],
    p_value = 2 * pnorm(abs(est / se), lower.tail = FALSE),
    baseline_sd_last_unpartnered = sd0,
    change_std = est / sd0,
    conf_low_std = ci[1] / sd0,
    conf_high_std = ci[2] / sd0
  )
}

absolute_results <- map_dfr(c("life_sat", "ghq_wb"), function(outcome) {
  map_dfr(seq_len(nrow(KEY_CONTRASTS)), function(i) {
    absolute_paired_change(treated_stack, outcome, KEY_CONTRASTS[i, ])
  })
}) |>
  group_by(contrast) |>
  mutate(
    p_holm_across_two_outcomes = p.adjust(p_value, method = "holm")
  ) |>
  ungroup()

write_csv_quiet(
  absolute_results,
  "09_absolute_within_person_LAT_former_changes.csv"
)


# =============================================================================
# 11. MATCHED TWO-PERIOD DIFFERENCE-IN-DIFFERENCES
# =============================================================================

prepare_matched_pair <- function(stack, outcome, p0, p1) {
  d <- stack |>
    filter(period %in% c(p0, p1)) |>
    mutate(y = .data[[outcome]]) |>
    filter(!is.na(y)) |>
    select(
      analysis_label, match_set_id, unit_pidp, condition,
      period, y
    ) |>
    distinct(match_set_id, unit_pidp, period, .keep_all = TRUE)

  eligible_units <- d |>
    count(
      analysis_label, match_set_id, unit_pidp, condition,
      name = "n_periods"
    ) |>
    filter(n_periods == 2L)

  set_eligibility <- eligible_units |>
    group_by(analysis_label, match_set_id) |>
    summarise(
      treated_complete = any(condition == "treated"),
      n_complete_controls = sum(condition == "unpartnered control"),
      .groups = "drop"
    ) |>
    filter(treated_complete, n_complete_controls >= 1L)

  d <- d |>
    semi_join(
      eligible_units,
      by = c("analysis_label", "match_set_id", "unit_pidp", "condition")
    ) |>
    semi_join(
      set_eligibility,
      by = c("analysis_label", "match_set_id")
    )

  unit_weights <- d |>
    distinct(analysis_label, match_set_id, unit_pidp, condition) |>
    group_by(analysis_label, match_set_id) |>
    mutate(
      n_controls = sum(condition == "unpartnered control"),
      analysis_weight = if_else(
        condition == "treated", 1, 1 / n_controls
      )
    ) |>
    ungroup() |>
    select(
      analysis_label, match_set_id, unit_pidp, condition,
      n_controls, analysis_weight
    )

  d |>
    left_join(
      unit_weights,
      by = c("analysis_label", "match_set_id", "unit_pidp", "condition"),
      relationship = "many-to-one"
    ) |>
    mutate(
      post = as.integer(as.character(period) == p1),
      treated = as.integer(condition == "treated"),
      treated_post = treated * post,
      match_person_id = paste(match_set_id, unit_pidp, sep = "__pid__"),
      match_time_id = paste(match_set_id, as.character(period), sep = "__period__")
    )
}

fit_matched_did <- function(stack, outcome, contrast_row) {
  p0 <- contrast_row$period0[[1L]]
  p1 <- contrast_row$period1[[1L]]
  contrast_name <- contrast_row$contrast[[1L]]

  d <- prepare_matched_pair(stack, outcome, p0, p1)
  if (nrow(d) == 0L) return(tibble())

  if (length(unique(d$treated_post)) < 2L) return(tibble())

  mod <- fixest::feols(
    y ~ treated_post | match_person_id + match_time_id,
    data = d,
    weights = ~ analysis_weight,
    vcov = ~ unit_pidp,
    notes = TRUE
  )

  b <- coef(mod)
  V <- vcov(mod)
  if (!"treated_post" %in% names(b)) return(tibble())

  est <- unname(b[["treated_post"]])
  se <- sqrt(V["treated_post", "treated_post"])
  ci <- normal_ci(est, se)
  sd0 <- get_baseline_sd(outcome)

  # Direct decomposition into weighted within-group change using exactly the
  # paired person-sets entering the model.
  deltas <- d |>
    select(
      analysis_label, match_set_id, unit_pidp, condition,
      period, y, analysis_weight
    ) |>
    distinct() |>
    pivot_wider(names_from = period, values_from = y) |>
    mutate(delta = .data[[p1]] - .data[[p0]])

  group_change <- deltas |>
    group_by(condition) |>
    summarise(
      weighted_change = safe_weighted_mean(delta, analysis_weight),
      n_person_sets = n(),
      sum_weight = sum(analysis_weight),
      .groups = "drop"
    )

  treated_change <- group_change$weighted_change[
    group_change$condition == "treated"
  ]
  control_change <- group_change$weighted_change[
    group_change$condition == "unpartnered control"
  ]

  tibble(
    analysis_label = unique(d$analysis_label)[1],
    outcome = outcome,
    contrast = contrast_name,
    period0 = p0,
    period1 = p1,
    matched_sets = n_distinct(d$match_set_id),
    unique_treated_people = n_distinct(d$unit_pidp[d$condition == "treated"]),
    unique_control_people = n_distinct(d$unit_pidp[d$condition == "unpartnered control"]),
    treated_weighted_change = treated_change,
    control_weighted_change = control_change,
    did_estimate = est,
    std_error = se,
    conf_low = ci[1],
    conf_high = ci[2],
    p_value = 2 * pnorm(abs(est / se), lower.tail = FALSE),
    baseline_sd_last_unpartnered = sd0,
    did_std = est / sd0,
    conf_low_std = ci[1] / sd0,
    conf_high_std = ci[2] / sd0
  )
}

run_matched_bundle <- function(stack) {
  if (nrow(stack) == 0L) return(tibble())
  map_dfr(c("life_sat", "ghq_wb"), function(outcome) {
    map_dfr(seq_len(nrow(KEY_CONTRASTS)), function(i) {
      fit_matched_did(stack, outcome, KEY_CONTRASTS[i, ])
    })
  }) |>
    group_by(analysis_label, contrast) |>
    mutate(
      p_holm_across_two_outcomes = p.adjust(p_value, method = "holm")
    ) |>
    ungroup()
}
