# =============================================================================
# BHPS + UKHLS
# PHISTORY-BASED FIRST-MARRIAGE RECONSTRUCTION, UKHLS SPSS DATE REFINEMENT, AND VALIDATION - v7 MONTHLY-DATE FIX
#
# This script REPLACES the previous validation workflow.
#
# AUTHORITATIVE PARTNERSHIP-HISTORY SOURCE:
#   phistory_long.dta
#
# OPTIONAL CROSS-CHECK SOURCE:
#   phistory_wide.dta
#
# User-supplied paths:
#   D:/Documenti/psicologia/big data/Understanding Society+BHPS/
#      Marital and Cohabitation Histories/stata/phistory_long.dta
#
#   D:/Documenti/psicologia/big data/Understanding Society+BHPS/
#      Marital and Cohabitation Histories/stata/phistory_wide.dta
#
# PHISTORY_LONG variables used:
#   pidp       cross-wave person identifier
#   spellno    spell number, earliest first
#   spellnor   spell number, recent first [OPTIONAL; absent in some releases]
#   status     2 = marriage, 3 = civil partnership, 10 = cohabitation
#   partner    partner pidp
#   startdate  spell start date
#   start_if   start-date imputation flag
#                0 = no imputation
#                1 = month imputed
#                2 = whole date imputed
#   enddate    spell end date
#   end_if     end-date imputation flag
#   mrgend     how marriage/civil partnership ended
#   cohend     how cohabitation ended
#                0 = ongoing
#                1 = break up
#                2 = marriage
#   ongoing    ongoing spell indicator
#   ttl_married
#   ever_married
#   lastintdate / lastinty / lastintm
#   hhorig
#   starty / startm
#   endy / endm
#
# CORE DESIGN
# -----------
# 1. Keep the existing prospectively observed first-marriage candidates.
#
# 2. Use PHISTORY_LONG to identify each respondent's TRUE earliest marriage
#    spell (status == 2), not merely the first marriage observed in the panel.
#
# 3. Validate the true first-marriage start against the prospective transition:
#
#       last unmarried interview < marriage start <= first married interview
#
#    Month-level validation is primary because PHISTORY provides starty/startm
#    and may impute day/month components.
#
# 4. Classify prospective transitions as:
#       - validated genuine first marriage
#       - probable remarriage / earlier marriage exists
#       - marriage date before T-1
#       - marriage date after T0
#       - no marriage spell in PHISTORY
#       - incomplete marriage start month/year
#
# 5. Recover the cohabitation spell immediately preceding first marriage:
#       - preferably a cohabitation spell with cohend == 2
#       - ending in/near the first-marriage month
#       - same partner where partner pidp is usable
#
# 6. Create exact continuous interview time relative to ACTUAL first marriage:
#
#       years_from_first_marriage =
#           (interview_date - first_marriage_anchor_date) / 365.25
#
#    PHISTORY month/year is treated as month-level timing. If an exact usable
#    PHISTORY startdate is available, it is retained separately. For a common
#    deterministic month-level anchor, the 15th of the reported month is used.
#
# 7. Preserve familiar substantive labels T-5,...,T0 as YEAR landmarks, not
#    survey waves. The model itself should use continuous elapsed years.
#
# =============================================================================

# UKHLS direct current-marriage month/year reports.
#
# IMPORTANT:
# In the Stata release, f_indresp.dta does not contain the needed variables,
# whereas they are present in the SPSS release. Therefore this workflow uses
# the SPSS INDRESP files for this refinement step.
#
# The files are located recursively rather than assuming a flat directory.
find_exact_file_recursive <- function(
    root,
    filename
) {

  ff <- list.files(
    root,
    recursive = TRUE,
    full.names = TRUE,
    ignore.case = TRUE
  )

  hit <- ff[
    tolower(
      basename(
        ff
      )
    ) ==
      tolower(
        filename
      )
  ]

  if (length(hit) == 0L) {
    return(
      NA_character_
    )
  }

  if (length(hit) > 1L) {
    stop(
      "More than one file named ",
      filename,
      " found under:\n",
      root,
      "\n\n",
      paste(
        hit,
        collapse = "\n"
      )
    )
  }

  hit[1L]
}

read_survey_file <- function(
    path
) {

  if (is.na(path) ||
      !file.exists(path)) {
    stop(
      "Survey file not found: ",
      path
    )
  }

  ext <- tolower(
    tools::file_ext(
      path
    )
  )

  if (ext == "dta") {
    return(
      haven::read_dta(
        path
      )
    )
  }

  if (ext == "sav") {
    return(
      haven::read_sav(
        path
      )
    )
  }

  stop(
    "Unsupported survey file extension: ",
    ext,
    " for ",
    path
  )
}


clean_int <- function(
    x
) {

  z <- suppressWarnings(
    as.integer(
      as.numeric(
        x
      )
    )
  )

  z[
    !is.na(z) &
      z < 0
  ] <- NA_integer_

  z
}


clean_month <- function(
    x
) {

  z <- clean_int(
    x
  )

  z[
    !is.na(z) &
      !z %in%
        1:12
  ] <- NA_integer_

  z
}


clean_year <- function(
    x
) {

  z <- clean_int(
    x
  )

  z[
    !is.na(z) &
      (
        z < 1900 |
          z > 2035
      )
  ] <- NA_integer_

  z
}


clean_code_keep_zero <- function(
    x
) {

  z <- suppressWarnings(
    as.integer(
      as.numeric(
        x
      )
    )
  )

  z[
    !is.na(z) &
      z < 0
  ] <- NA_integer_

  z
}


month_index <- function(
    year,
    month
) {

  ifelse(
    !is.na(year) &
      !is.na(month),
    year *
      12L +
      month,
    NA_real_
  )
}


midmonth_date <- function(
    year,
    month
) {

  out <- rep(
    as.Date(
      NA
    ),
    length(
      year
    )
  )

  ok <- !is.na(year) &
    !is.na(month)

  out[
    ok
  ] <- as.Date(
    sprintf(
      "%04d-%02d-15",
      year[
        ok
      ],
      month[
        ok
      ]
    )
  )

  out
}


clean_stata_month_index <- function(
    x
) {

  z <- suppressWarnings(
    as.integer(
      as.numeric(
        x
      )
    )
  )

  z[
    !is.na(z) &
      z < 0
  ] <- NA_integer_

  z
}


expected_stata_month_index <- function(
    year,
    month
) {

  ifelse(
    !is.na(year) &
      !is.na(month),
    (year - 1960L) *
      12L +
      (month - 1L),
    NA_integer_
  )
}


safe_pct <- function(
    n,
    d
) {

  if (is.na(d) ||
      d == 0) {
    return(
      NA_real_
    )
  }

  100 *
    n /
    d
}


safe_median <- function(
    x
) {

  x <- x[
    is.finite(
      x
    )
  ]

  if (length(x) == 0L) {
    return(
      NA_real_
    )
  }

  median(
    x
  )
}


safe_quantile <- function(
    x,
    p
) {

  x <- x[
    is.finite(
      x
    )
  ]

  if (length(x) == 0L) {
    return(
      NA_real_
    )
  }

  as.numeric(
    quantile(
      x,
      p,
      names =
        FALSE
    )
  )
}

# =============================================================================
# 5. PROSPECTIVE T-1 AND T0 INTERVIEW DATES
# =============================================================================
#
# T-1 and T0 here retain their ORIGINAL prospective meaning only for
# validation:
#
#   T-1 = last interview at which the respondent was not married
#   T0  = first interview at which the respondent was observed married
#
# After validation, the definitive temporal zero will become ACTUAL marriage.
# =============================================================================

prospective_dates <- treated_base |>
  crossing(
    prospective_event_time =
      c(
        -1L,
        0L
      )
  ) |>
  mutate(
    wave_num =
      event_wave +
        prospective_event_time
  ) |>
  left_join(
    timing_panel |>
      select(
        pidp,
        survey,
        wave_num,
        interview_year,
        interview_month,
        interview_day,
        interview_month_index,
        interview_date
      ),
    by = c(
      "pidp",
      "survey",
      "wave_num"
    ),
    relationship =
      "many-to-one"
  ) |>
  # IMPORTANT:
  # wave_num differs between T-1 and T0. If it is left as an implicit ID
  # variable, pivot_wider() creates two rows per prospective transition.
  # Specify the intended transition-level ID columns explicitly.
  pivot_wider(
    id_cols =
      c(
        pidp,
        survey,
        event_wave
      ),
    names_from =
      prospective_event_time,
    values_from =
      c(
        interview_year,
        interview_month,
        interview_day,
        interview_month_index,
        interview_date
      ),
    names_glue =
      "{.value}_T{prospective_event_time}"
  ) |>
  rename(
    last_unmarried_year =
      `interview_year_T-1`,
    first_married_year =
      interview_year_T0,

    last_unmarried_month =
      `interview_month_T-1`,
    first_married_month =
      interview_month_T0,

    last_unmarried_day =
      `interview_day_T-1`,
    first_married_day =
      interview_day_T0,

    last_unmarried_month_index =
      `interview_month_index_T-1`,
    first_married_month_index =
      interview_month_index_T0,

    last_unmarried_interview_date =
      `interview_date_T-1`,
    first_married_interview_date =
      interview_date_T0
  )

# =============================================================================
# 6B. VERIFY PHISTORY MONTHLY DATE FORMAT
# =============================================================================
#
# PHISTORY startdate/enddate are Stata MONTHLY indices, not Stata daily dates.
# For example:
#   monthly_index = (year - 1960) * 12 + (month - 1)
#
# Therefore no exact marriage DAY is available from startdate. Calendar anchors
# are set to the 15th of the reported/imputed marriage month.
# =============================================================================

phistory_startdate_format_audit <- phistory_long |>
  filter(
    status == 2
  ) |>
  summarise(
    marriage_spells_with_both_representations =
      sum(
        !is.na(
          startdate_monthly_index
        ) &
          !is.na(
            expected_startdate_monthly_index
          )
      ),

    exact_month_index_matches =
      sum(
        startdate_matches_starty_startm,
        na.rm =
          TRUE
      ),

    pct_exact_month_index_matches =
      if_else(
        marriage_spells_with_both_representations >
          0,
        100 *
          exact_month_index_matches /
          marriage_spells_with_both_representations,
        NA_real_
      ),

    max_absolute_month_index_difference =
      {
        dd <- abs(
          startdate_monthly_index -
            expected_startdate_monthly_index
        )

        if (
          all(
            is.na(
              dd
            )
          )
        ) {
          NA_real_
        } else {
          max(
            dd,
            na.rm =
              TRUE
          )
        }
      }
  )


write_csv(
  phistory_startdate_format_audit,
  file.path(
    output_dir,
    "PHISTORY_startdate_monthly_format_audit.csv"
  )
)

      )
  )

# =============================================================================
# 10B. REFINE UKHLS IMPUTED FIRST-MARRIAGE DATES USING DIRECT CURRENT-MARRIAGE
#      MONTH/YEAR REPORTS FROM a_indresp AND f_indresp
# =============================================================================
#
# In this release, the literal variables:
#   a_lcmarm
#   a_lcmary4
# are wave-prefixed in the SPSS release:
#   a_indresp.sav -> a_lcmarm / a_lcmary4
#   f_indresp.sav -> f_lcmarm / f_lcmary4
#
# Refinement applies only to UKHLS validated genuine first marriages with
# PHISTORY start_if == 1 or 2.
#
# A direct report is eligible to replace the PHISTORY imputed month/year only
# when:
#   - month/year are complete and valid; AND
#   - the reported marriage month lies between the last prospectively unmarried
#     and first prospectively married interviews.
#
# If a_indresp and f_indresp yield the SAME eligible date, this is corroborated.
# If only one source yields an eligible date, that date is used.
# If they yield DIFFERENT eligible dates, the case is flagged as a conflict and
# PHISTORY is retained rather than choosing arbitrarily.
#
# For start_if == 0, PHISTORY remains authoritative.
# =============================================================================

required_ukhls_direct_paths <- c(
  a_indresp_spss =
    ukhls_a_indresp_path,
  f_indresp_spss =
    ukhls_f_indresp_path
)

missing_ukhls_direct_paths <- required_ukhls_direct_paths[
  !file.exists(
    required_ukhls_direct_paths
  )
]

if (length(missing_ukhls_direct_paths) > 0L) {
  stop(
    "Missing UKHLS INDRESP file(s) needed for direct marriage-date refinement:\n",
    paste(
      missing_ukhls_direct_paths,
      collapse = "\n"
    )
  )
}


read_ukhls_direct_marriage <- function(
    path,
    source_wave_label
) {

  dat <- read_survey_file(
    path
  )

  # SPSS variable names are matched case-insensitively.
  nm_lower <- tolower(
    names(
      dat
    )
  )

  get_var_name_ci <- function(
      target
  ) {

    hit <- which(
      nm_lower ==
        tolower(
          target
        )
    )

    if (length(hit) == 0L) {
      return(
        NA_character_
      )
    }

    names(
      dat
    )[
      hit[1L]
    ]
  }

  pidp_var <- get_var_name_ci(
    "pidp"
  )

  # Marriage variables are wave-prefixed in the SPSS release:
  #   Wave a -> a_lcmarm / a_lcmary4
  #   Wave f -> f_lcmarm / f_lcmary4
  #
  # Derive the expected names from source_wave_label rather than hard-coding
  # a particular wave prefix.
  expected_lcmarm_var <- paste0(
    source_wave_label,
    "_lcmarm"
  )

  expected_lcmary4_var <- paste0(
    source_wave_label,
    "_lcmary4"
  )

  lcmarm_var <- get_var_name_ci(
    expected_lcmarm_var
  )

  lcmary4_var <- get_var_name_ci(
    expected_lcmary4_var
  )

  missing <- c(
    pidp =
      pidp_var,
    marriage_month =
      lcmarm_var,
    marriage_year =
      lcmary4_var
  )

  missing <- names(
    missing
  )[
    is.na(
      missing
    )
  ]

  if (length(missing) > 0L) {
    stop(
      basename(
        path
      ),
      " is missing expected variable(s): ",
      paste(
        missing,
        collapse = ", "
      ),
      "\nExpected wave-specific variables: ",
      expected_lcmarm_var,
      ", ",
      expected_lcmary4_var,
      "\nAvailable variables containing 'lcmar': ",
      paste(
        names(
          dat
        )[
          grepl(
            "lcmar",
            names(
              dat
            ),
            ignore.case = TRUE
          )
        ],
        collapse = ", "
      )
    )
  }

  out <- tibble(
    pidp =
      dat[[
        pidp_var
      ]],

    direct_source_wave =
      source_wave_label,

    direct_source_file =
      basename(
        path
      ),

    direct_marriage_month =
      clean_month(
        dat[[
          lcmarm_var
        ]]
      ),

    direct_marriage_year =
      clean_year(
        dat[[
          lcmary4_var
        ]]
      )
  ) |>
    mutate(
      direct_marriage_month_index =
        month_index(
          direct_marriage_year,
          direct_marriage_month
        ),

      direct_marriage_midmonth_date =
        midmonth_date(
          direct_marriage_year,
          direct_marriage_month
        ),

      direct_marriage_complete =
        !is.na(
          direct_marriage_month_index
        )
    )

  audit <- tibble(
    source_wave =
      source_wave_label,
    source_file =
      basename(
        path
      ),
    source_path =
      path,
    pidp_variable_used =
      pidp_var,
    marriage_month_variable_used =
      lcmarm_var,
    marriage_year_variable_used =
      lcmary4_var,
    n_rows =
      nrow(
        dat
      ),
    complete_direct_marriage_dates =
      sum(
        out$direct_marriage_complete,
        na.rm =
          TRUE
      )
  )

  list(
    data =
      out,
    audit =
      audit
  )
}

# =============================================================================
# 10C. DEFINE REFINED FIRST-MARRIAGE TEMPORAL ANCHOR
# =============================================================================
#
# Hierarchy:
#
# 1. PHISTORY start_if == 0:
#      use reported PHISTORY month/year, anchored at the 15th.
#      PHISTORY startdate is a monthly index, not an exact calendar day.
#
# 2. UKHLS start_if in {1,2}:
#      if exactly one eligible direct month/year is identified from a/f
#      INDRESP, use its mid-month anchor.
#
# 3. Otherwise:
#      retain PHISTORY mid-month imputed anchor, explicitly flagged as imputed.
#
# This preserves all validated cases. "High precision" means directly
# reported month/year, not an exact day.
# =============================================================================

validation_refined <- validation |>
  left_join(
    ukhls_direct_resolution |>
      select(
        pidp,
        event_wave,
        direct_report_resolved,
        direct_report_conflict,
        eligible_direct_year,
        eligible_direct_month,
        eligible_direct_month_index,
        eligible_direct_midmonth_date,
        direct_sources_supporting_selected_date,
        direct_minus_phistory_months,
        direct_report_agrees_exactly_with_phistory_month
      ),
    by = c(
      "pidp",
      "event_wave"
    ),
    relationship =
      "one-to-one"
  ) |>
  mutate(
    use_ukhls_direct_month_year =
      survey == "UKHLS" &
        validated_genuine_first_marriage &
        first_marriage_start_if %in%
          c(
            1L,
            2L
          ) &
        coalesce(
          direct_report_resolved,
          FALSE
        ),

    refined_first_marriage_year =
      case_when(
        use_ukhls_direct_month_year ~
          eligible_direct_year,

        TRUE ~
          first_marriage_start_year
      ),

    refined_first_marriage_month =
      case_when(
        use_ukhls_direct_month_year ~
          eligible_direct_month,

        TRUE ~
          first_marriage_start_month
      ),

    refined_first_marriage_month_index =
      month_index(
        refined_first_marriage_year,
        refined_first_marriage_month
      ),

    refined_first_marriage_anchor_date =
      case_when(
        use_ukhls_direct_month_year ~
          eligible_direct_midmonth_date,

        TRUE ~
          first_marriage_start_midmonth_date
      ),

    refined_anchor_source =
      case_when(
        use_ukhls_direct_month_year ~
          "UKHLS direct current-marriage month/year",

        first_marriage_start_if == 0 ~
          "PHISTORY reported month/year",

        first_marriage_start_if == 1 ~
          "PHISTORY month-imputed start",

        first_marriage_start_if == 2 ~
          "PHISTORY whole-date-imputed start",

        TRUE ~
          "PHISTORY month/year"
      ),

    refined_anchor_precision =
      case_when(
        use_ukhls_direct_month_year ~
          "direct_reported_month_year",

        first_marriage_start_if == 0 ~
          "reported_month_year",

# 15. RECOVER COHABITATION SPELL IMMEDIATELY PRECEDING FIRST MARRIAGE
# =============================================================================
#
# Candidate premarital cohabitation spell:
#   status == 10
#   starts before first marriage
#
# Preferred ranking:
#   1. cohend == 2 (ended in marriage)
#   2. same partner as marriage, where both partner ids are usable
#   3. end month closest to marriage month
#   4. latest cohabitation spell before marriage
#
# This is deliberately diagnostic rather than blindly assuming adjacency.
# =============================================================================

    first_marriage_start_imputation

    status == 10

    pidp,

      end_month_index,

      end_midmonth_date,

      "many-to-many"

      !is.na(
        cohab_start_month_index

        !is.na(
          first_marriage_start_month_index

        cohab_start_month_index <
          first_marriage_start_month_index,

          first_marriage_start_month_index,

        first_marriage_partner

          cohab_partner

        first_marriage_partner >=
          0 &
        cohab_partner >=
          0,

      same_partner_known &
        first_marriage_partner ==
          cohab_partner,

      first_marriage_start_month_index -
        cohab_end_month_index,

    months_cohab_start_to_marriage =
      first_marriage_start_month_index -
        cohab_start_month_index

  filter(
    cohab_starts_before_marriage
  )

# =============================================================================
# 18. CREATE CONTINUOUS-TIME PERSON-WAVE PANEL
# =============================================================================
#
# Merge exact interview dates and define continuous time relative to the
# validated first-marriage month.
#
# The temporal anchor uses the precision hierarchy defined above.
# All sources are month-level. The deterministic calendar anchor is the 15th
# of the selected marriage month.
#
# IMPORTANT:
# This is NOT discretized into survey-wave T values.
#
# years_from_first_marriage is the inferential continuous-time variable.
#
# integer_year_landmark is supplied only for descriptive plotting / labels.
# =============================================================================

continuous_time_panel <- combined_panel |>
  inner_join(
    validated_first_marriage_file |>
      select(
        pidp,
        survey,
        first_marriage_start_year,
        first_marriage_start_month,
        first_marriage_start_midmonth_date,
        first_marriage_start_if,
        first_marriage_start_imputation,

        refined_first_marriage_year,
        refined_first_marriage_month,
        refined_first_marriage_month_index,
        refined_first_marriage_anchor_date,
        refined_anchor_source,
        refined_anchor_precision,
        high_precision_primary_anchor,
        medium_or_high_precision_anchor,

        premarital_cohab_found,
        premarital_cohab_start_midmonth_date
      ),
    by = c(
      "pidp",
      "survey"
    ),
    relationship =
      "many-to-one"
  ) |>
  left_join(
    timing_panel |>
      select(
        pidp,
        survey,
        wave_num,
        interview_date,
        interview_year,
        interview_month,
        interview_day
      ),
    by = c(
      "pidp",
      "survey",
      "wave_num"
    ),
    relationship =
      "many-to-one"
  ) |>
  mutate(

    days_from_first_marriage =
      as.numeric(
        interview_date -
          refined_first_marriage_anchor_date
      ),

    years_from_first_marriage =
      days_from_first_marriage /
        365.25,

    integer_year_landmark =
      round(
        years_from_first_marriage
      ),

    in_primary_minus5_to_0_window =
      years_from_first_marriage >=
        -5.5 &
        years_from_first_marriage <=
          0.5,

    in_extended_minus5_to_plus2_window =
      years_from_first_marriage >=
        -5.5 &
        years_from_first_marriage <=
          2.5,

    years_from_premarital_cohab_start =
      if_else(
        premarital_cohab_found &
          !is.na(
            premarital_cohab_start_midmonth_date
          ) &
          !is.na(
            interview_date
          ),
        as.numeric(
          interview_date -
            premarital_cohab_start_midmonth_date
        ) /
          365.25,
        NA_real_
      )
  )


saveRDS(
  continuous_time_panel,
  file.path(
    output_dir,
    "validated_first_marriage_continuous_time_person_wave_panel.rds"
  )
)


write_csv(

      refined_anchor_source,
      refined_anchor_precision,
      high_precision_primary_anchor,
      medium_or_high_precision_anchor,
      years_from_first_marriage,
      integer_year_landmark,
      in_primary_minus5_to_0_window,
      in_extended_minus5_to_plus2_window

# =============================================================================
# 19. CONTINUOUS-TIME COVERAGE BY YEAR LANDMARK
# =============================================================================
#
# A person's observation is considered near T-k if it lies within +/- 0.5 year
# of that literal year landmark.
#
# These are descriptive diagnostics only.
# =============================================================================

landmarks <- tibble(
  landmark =
    c(
      -5,
      -4,
      -3,
      -2,
      -1,
      0,
      1,
      2
    )
)


continuous_landmark_coverage <- crossing(
  validated_first_marriage_file |>
    select(
      pidp,
      survey
    ),
  landmarks
) |>
  left_join(
    continuous_time_panel |>
      select(
        pidp,
        survey,
        wave_num,
        years_from_first_marriage
      ),
    by = c(
      "pidp",
      "survey"
    ),
    relationship =
      "many-to-many"
  ) |>
  mutate(
    absolute_distance_to_landmark =
      abs(
        years_from_first_marriage -
          landmark
      )
  ) |>
  group_by(
    pidp,
    survey,
    landmark
  ) |>
  summarise(
    closest_distance_years =
      if (
        all(
          is.na(
            absolute_distance_to_landmark
          )
        )
      ) {
        NA_real_
      } else {
        min(
          absolute_distance_to_landmark,
          na.rm =
            TRUE
        )
      },

    has_interview_within_pm6_months =
      !is.na(
        closest_distance_years
      ) &
        closest_distance_years <=
          0.5,

    .groups =
      "drop"
  )


landmark_coverage_summary <- continuous_landmark_coverage |>
  group_by(
    survey,
    landmark
  ) |>
  summarise(
    validated_people =
      n(),

    with_interview_within_pm6_months =
      sum(
        has_interview_within_pm6_months
      ),

    pct_with_interview_within_pm6_months =
      100 *
        with_interview_within_pm6_months /
        validated_people,

    median_closest_distance_months =
      safe_median(
        closest_distance_years *
          12
      ),

    p90_closest_distance_months =
      safe_quantile(
