# PURPOSE
# -------
# Construct the non-overlapping clean-unpartnered episode identifiers and follow-
# up boundaries required by the latent stable/WPVS Cox selection workflow.
#
# This script contains DATA PREPARATION ONLY. It deliberately does NOT select
# psychological baselines using historical means/deviations and does NOT estimate

# the latent T-specific score lookups.
#

# validated UKHLS_H1_latent_stable_WPVS_selection_v1.R workflow.
################################################################################

id_chr <- function(x) {
  z <- x
  if (
    inherits(z, "haven_labelled") ||
      inherits(z, "labelled") ||
      inherits(z, "haven_labelled_spss")
  ) z <- unclass(z)
  if (is.factor(z)) z <- as.character(z)
  if (is.numeric(z) || is.integer(z)) {
    out <- ifelse(
      is.na(z), NA_character_,
      format(z, scientific = FALSE, trim = TRUE, digits = 22)
    )
    out <- sub("\\.0+$", "", out)
    return(out)
  }
  out <- trimws(as.character(z))
  out[out %in% c("", "NA", "NaN")] <- NA_character_
  sub("\\.0+$", "", out)
}

safe_min <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) == 0L) NA_real_ else min(z)
}

safe_median <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) == 0L) NA_real_ else stats::median(z)
}

safe_max <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) == 0L) NA_real_ else max(z)
}

write_csv_quiet <- function(x, filename) {
  readr::write_csv(
    x,
    file.path(DIAG_DIR, filename),
    na = "",
    progress = FALSE
  )
}

episode_uid_fun <- function(pidp, episode_num) {
  paste0(pidp, "__UNP_EP__", episode_num)
}

# ------------------------------------------------------------------------------
# 6. CONSTRUCT NON-OVERLAPPING CLEAN-UNPARTNERED EPISODES
# ------------------------------------------------------------------------------
clean_landmarks <- formation %>%
  dplyr::arrange(pidp, clean_ym, wave_n) %>%
  dplyr::distinct(pidp, wave_n, clean_ym, .keep_all = TRUE)

boundary_times_by_pidp <- split(boundary_events$event_ym, boundary_events$pidp)

assign_episode_numbers <- function(.x, .y) {
  person_id <- as.character(.y$pidp[[1L]])
  .x <- .x %>%
    dplyr::arrange(clean_ym, wave_n) %>%
    dplyr::mutate(previous_clean_ym = dplyr::lag(clean_ym))

  bt <- boundary_times_by_pidp[[person_id]]
  if (is.null(bt)) bt <- numeric(0)
  bt <- sort(unique(bt[is.finite(bt)]))

  intervening_boundary <- vapply(
    seq_len(nrow(.x)),
    function(i) {
      prev <- .x$previous_clean_ym[[i]]
      curr <- .x$clean_ym[[i]]
      if (!is.finite(prev) || !is.finite(curr) || length(bt) == 0L) return(FALSE)
      any(bt > prev & bt < curr)
    },
    logical(1)
  )

  same_month_boundary_at_current <- vapply(
    seq_len(nrow(.x)),
    function(i) {
      curr <- .x$clean_ym[[i]]
      if (!is.finite(curr) || length(bt) == 0L) return(FALSE)
      any(bt == curr)
    },
    logical(1)
  )

  new_episode <- seq_len(nrow(.x)) == 1L | intervening_boundary

  .x %>%
    dplyr::mutate(
      intervening_partnership_boundary = intervening_boundary,
      same_month_boundary_at_clean_landmark = same_month_boundary_at_current,
      episode_num = cumsum(new_episode),
      episode_uid = episode_uid_fun(person_id, episode_num)
    )
}

clean_ep <- clean_landmarks %>%
  dplyr::group_by(pidp) %>%
  dplyr::group_modify(assign_episode_numbers) %>%
  dplyr::ungroup()

person_followup <- panel %>%
  dplyr::group_by(pidp) %>%
  dplyr::summarise(
    panel_first_ym = min(interview_ym, na.rm = TRUE),
    panel_last_ym = max(interview_ym, na.rm = TRUE),
    .groups = "drop"
  )

episode_summary <- clean_ep %>%
  dplyr::group_by(pidp, episode_num, episode_uid) %>%
  dplyr::summarise(
    episode_start_ym = min(clean_ym),
    episode_last_clean_ym = max(clean_ym),
    n_clean_landmarks = dplyr::n(),
    any_same_month_boundary = any(same_month_boundary_at_clean_landmark, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  dplyr::arrange(pidp, episode_start_ym, episode_num) %>%
  dplyr::group_by(pidp) %>%
  dplyr::mutate(next_episode_start_ym = dplyr::lead(episode_start_ym)) %>%
  dplyr::ungroup() %>%
  dplyr::left_join(person_followup, by = "pidp", relationship = "many-to-one") %>%
  dplyr::mutate(
    end_is_next_episode =
      !is.na(next_episode_start_ym) & next_episode_start_ym <= panel_last_ym,
    episode_followup_end_ym = dplyr::case_when(
      end_is_next_episode ~ next_episode_start_ym,
      TRUE ~ panel_last_ym
    ),
    episode_followup_months = episode_followup_end_ym - episode_start_ym
  )

construction_summary <- tibble::tibble(
  metric = c(
    "clean-unpartnered landmarks",
    "people with >=1 clean landmark",
    "constructed non-overlapping unpartnered episodes",
    "people with >=2 constructed episodes",
    "episodes with >1 clean landmark",
    "clean landmarks starting a new episode after a partnership boundary",
    "episodes with positive follow-up"
  ),
  value = c(
    nrow(clean_ep),
    dplyr::n_distinct(clean_ep$pidp),
    nrow(episode_summary),
    episode_summary %>% dplyr::count(pidp) %>% dplyr::filter(n >= 2L) %>% nrow(),
    sum(episode_summary$n_clean_landmarks > 1L, na.rm = TRUE),
    sum(clean_ep$intervening_partnership_boundary, na.rm = TRUE),
    sum(episode_summary$episode_followup_months > 0, na.rm = TRUE)
  )
)
write_csv_quiet(construction_summary, "02_episode_construction_summary.csv")

followup_distribution <- episode_summary %>%

    episodes = dplyr::n(),

    max_months = safe_max(episode_followup_months),
    episodes_ge12m = sum(episode_followup_months >= 12, na.rm = TRUE),
    episodes_ge24m = sum(episode_followup_months >= 24, na.rm = TRUE),
    episodes_ge36m = sum(episode_followup_months >= 36, na.rm = TRUE),
    episodes_ge60m = sum(episode_followup_months >= 60, na.rm = TRUE)

write_csv_quiet(followup_distribution, "05_episode_followup_distribution.csv")
