################################################################################
# UKHLS H1 PSYCHOLOGICAL SELECTION
# FINAL WAVE-ADJUSTED STABLE-TRAIT / WPVS DECOMPOSITION
# Version 1.0
#
# PURPOSE
# -------
# Definitive second-stage H1 analysis using the wave-adjusted model-based
# decomposition produced by:
#
#   UKHLS_latent_stable_WPVS_final_measurement_stage_v2_1_WAVECENTERED_AGEFIX.R.txt
#
# The psychological predictors are:
#
#   1. stable_score_SD
#      Model-based stable between-person psychological component.
#
#   2. current_wpvs_SD
#      Current within-person variability score (WPVS) at the clean-unpartnered
#      baseline, conditional on the stable component.
#
# Both are expressed in units of the SAME outcome-specific raw current-score SD
# by the measurement-stage workflow.
#
# PRIMARY HISTORY SPECIFICATION
# -----------------------------
# T = 4:
#   current clean-unpartnered baseline + 3 immediately preceding UKHLS waves.
#
# HISTORY-LENGTH SENSITIVITIES
# ----------------------------
# T = 5 and T = 6.
#
# H1 DESIGN A: INCIDENT LAT FORMATION
# -----------------------------------
# Reuses the VALIDATED incident-LAT matched risk sets.
# A matched conditional-logistic model is fit separately for:
#   - life satisfaction
#   - favorable-oriented GHQ / lower psychological distress
#
# Model:
#
#   incident LAT ~ stable component + current WPVS
#                  + strata(matched set) + respondent-clustered robust SE
#
# Matched sets are pruned AFTER score eligibility so every retained set contains
# exactly one incident-LAT case and at least one eligible control.
#
# H1 DESIGN B: SUBSEQUENT COHABITATION / MARRIAGE
# ------------------------------------------------
# Reuses the already validated NON-OVERLAPPING clean-unpartnered episode
# construction. For each episode and psychological measure, this script selects
# the FIRST clean-unpartnered landmark with an available latent T-specific score.
#
# Thus the latent analysis does NOT merely retain the old >=2-prior baseline:
# if that earlier baseline lacks a latent score but a later clean landmark in the
# same episode has one, the later FIRST LATENT-SCORE-ELIGIBLE landmark becomes the
# baseline. Outcomes are then reconstructed prospectively from that new baseline.
#
# Endpoints:
#   A. Cohabitation
#      Cause-specific Cox model. Marriage before or in the same recorded month as
#      cohabitation is treated as a competing event and ends the risk interval.
#
#   B. Marriage
#      First subsequent marriage. Cohabitation is allowed as an intermediate
#      state and does not censor follow-up.
#
#   C. Cohabitation or marriage
#      First occurrence of either state.
#
# Primary Cox specification:
#   stable component + current WPVS
#   + age + age^2 + sex
#   + recurrent-episode indicator
#   + flexible adjustment for baseline delay within clean-unpartnered episode
#   + baseline-calendar-year strata
#   + respondent-clustered robust sandwich SE
#
# Time scale:
#   months since the selected psychological baseline.
#
# PRIMARY MULTIPLICITY
# --------------------
# The two H1 designs retain the manuscript's separate inferential families:
#
# Incident LAT:
#   2 outcomes x 2 components = 4 primary coefficient tests.
#   Holm correction across 4.
#   Two 2-df joint tests (one per psychological outcome), Holm across 2.
#
# Cox transition models:
#   3 endpoints x 2 outcomes x 2 components = 12 primary coefficient tests.
#   Holm correction across 12.
#   Six 2-df joint tests, Holm across 6.
#
# T=5 and T=6 are sensitivity families and receive the same within-design Holm
# correction separately at each T.
#
# INTERPRETATION
# --------------
# Higher predictor values always indicate MORE FAVORABLE psychological
# functioning:
#   - higher life satisfaction
#   - lower psychological distress for GHQ
#
# OR/HR > 1 therefore indicates greater subsequent transition odds/rate for a
# more favorable psychological component.
#
# These are prospective selection associations, NOT causal effects of
# psychological functioning on relationship transitions.
#
# IMPORTANT TWO-STEP INFERENCE NOTE
# ---------------------------------
# This script treats Step-1 correlation-preserving stable/WPVS scores as generated
# regressors and uses the same respondent-robust inferential structure as the
# existing manuscript analyses. It does NOT propagate SEM/scoring uncertainty.
# A full respondent-level two-step bootstrap that re-estimates the measurement
# model and scores is best treated as a later inferential sensitivity after the
# substantive H1 estimates have been inspected.
#
# RUN FROM THE BEGINNING IN A FRESH R SESSION.
################################################################################

normalize_pidp <- function(x) {

  z <- x

  if (
    inherits(z, "haven_labelled") ||
      inherits(z, "labelled") ||
      inherits(z, "haven_labelled_spss")
  ) {
    z <- unclass(z)
  }

  if (is.factor(z)) {
    z <- as.character(z)
  }

  if (is.numeric(z) || is.integer(z)) {
    out <- ifelse(
      is.na(z),
      NA_character_,
      format(
        z,
        scientific = FALSE,
        trim = TRUE,
        digits = 22
      )
    )
    out <- sub(
      "\\.0+$",
      "",
      out
    )
    return(out)
  }

  out <- trimws(
    as.character(z)
  )

  out[
    out %in% c(
      "",
      "NA",
      "NaN"
    )
  ] <- NA_character_

  sub(
    "\\.0+$",
    "",
    out
  )
}

safe_min <- function(x) {

  z <- suppressWarnings(
    as.numeric(x)
  )

  z <- z[
    is.finite(z)
  ]

  if (
    length(z) == 0L
  ) {
    NA_real_
  } else {
    min(z)
  }
}


safe_median <- function(x) {

  z <- suppressWarnings(
    as.numeric(x)
  )

  z <- z[
    is.finite(z)
  ]

  if (
    length(z) == 0L
  ) {
    NA_real_
  } else {
    stats::median(z)
  }
}


safe_max <- function(x) {

  z <- suppressWarnings(
    as.numeric(x)
  )

  z <- z[
    is.finite(z)
  ]

  if (
    length(z) == 0L
  ) {
    NA_real_
  } else {
    max(z)
  }
}


safe_cor <- function(x, y) {

  ok <-
    is.finite(x) &
    is.finite(y)

  if (
    sum(ok) < 3L
  ) {
    return(
      NA_real_
    )
  }

  suppressWarnings(
    stats::cor(
      x[ok],
      y[ok]
    )
  )
}


normal_ci <- function(
    est,
    se,
    level = 0.95
) {

  crit <- stats::qnorm(
    1 -
      (
        1 -
          level
      ) /
      2
  )

  c(
    est -
      crit *
      se,
    est +
      crit *
      se
  )
}


adjust_valid_p <- function(
    p,
    status = rep(
      "OK",
      length(p)
    ),
    method = "holm",
    n_total = NULL
) {

  out <- rep(
    NA_real_,
    length(p)
  )

  ok <-
    status == "OK" &
    is.finite(p)

  if (!any(ok)) {
    return(out)
  }

  n_use <- if (
    is.null(n_total)
  ) {
    sum(ok)
  } else {
    as.integer(n_total)
  }

  if (
    !is.finite(n_use) ||
      n_use <
      sum(ok)
  ) {
    n_use <- sum(ok)
  }

  out[ok] <- stats::p.adjust(
    p[ok],
    method = method,
    n = n_use
  )

  out
}


safe_inverse <- function(V) {

  V <- as.matrix(V)

  if (
    nrow(V) == 0L ||
      any(
        !is.finite(V)
      )
  ) {
    return(NULL)
  }

  inv <- tryCatch(
    solve(V),
    error = function(e) NULL
  )

  if (!is.null(inv)) {
    return(inv)
  }

  ee <- tryCatch(
    eigen(
      V,
      symmetric = TRUE
    ),
    error = function(e) NULL
  )

  if (is.null(ee)) {
    return(NULL)
  }

  tol <-
    max(dim(V)) *
    max(
      abs(
        ee$values
      ),
      1
    ) *
    .Machine$double.eps *
    100

  keep <- which(
    ee$values >
      tol
  )

  if (
    length(keep) == 0L
  ) {
    return(NULL)
  }

  Q <- ee$vectors[
    ,
    keep,
    drop = FALSE
  ]

  Q %*%
    diag(
      1 /
        ee$values[
          keep
        ],
      nrow = length(
        keep
      )
    ) %*%
    t(Q)
}


in_followup_interval <- function(
    event_ym,
    baseline_ym,
    end_ym,
    end_is_next_episode
) {

  after_baseline <-
    !is.na(event_ym) &
    !is.na(baseline_ym) &
    event_ym >
    baseline_ym

  before_end <- dplyr::case_when(
    is.na(end_ym) ~ FALSE,
    end_is_next_episode %in% TRUE ~
      event_ym <
      end_ym,
    TRUE ~
      event_ym <=
      end_ym
  )

  after_baseline &
    before_end
}

# ==============================================================================
# 4. LOAD AND VALIDATE LATENT SCORE LOOKUPS
# ==============================================================================

score_lookups <- list()

score_schema_audit <- list()
schema_idx <- 0L


for (
  T_now in T_VALUES
) {

  s <- readRDS(
    SCORE_FILE(
      T_now
    )
  )

  required_score_vars <- c(
    "baseline_id",
    "pidp_key",
    "baseline_wave",
    "measure",
    "T",
    "raw_current_score_sd",
    "stable_score_SD",
    "current_wpvs_SD"
  )

  missing_score_vars <- setdiff(
    required_score_vars,
    names(s)
  )

  if (
    length(
      missing_score_vars
    ) > 0L
  ) {
    stop(
      "Latent score lookup T=",
      T_now,
      " is missing: ",
      paste(
        missing_score_vars,
        collapse = ", "
      )
    )
  }

  s <- s %>%
    dplyr::mutate(
      pidp_key =
        normalize_pidp(
          pidp_key
        ),
      baseline_wave =
        suppressWarnings(
          as.integer(
            baseline_wave
          )
        ),
      T =
        suppressWarnings(
          as.integer(
            T
          )
        ),
      stable_score_SD =
        suppressWarnings(
          as.numeric(
            stable_score_SD
          )
        ),
      current_wpvs_SD =
        suppressWarnings(
          as.numeric(
            current_wpvs_SD
          )
        ),
      raw_current_score_sd =
        suppressWarnings(
          as.numeric(
            raw_current_score_sd
          )
        ),
      baseline_id =
        as.character(
          baseline_id
        ),
      measure =
        as.character(
          measure
        )
    ) %>%
    dplyr::filter(
      T ==
        T_now
    )

  if (
    anyDuplicated(
      s[
        c(
          "pidp_key",
          "baseline_wave",
          "measure"
        )
      ]
    ) > 0L
  ) {
    stop(
      "Latent score lookup T=",
      T_now,
      " is not unique by pidp x baseline wave x measure."
    )
  }

  score_lookups[[as.character(T_now)]] <- s

  schema_idx <- schema_idx + 1L

  score_schema_audit[[schema_idx]] <- s %>%
    dplyr::group_by(
      measure
    ) %>%
    dplyr::summarise(
      T = T_now,
      rows =
        dplyr::n(),
      people =
        dplyr::n_distinct(
          pidp_key
        ),
      finite_stable =
        sum(
          is.finite(
            stable_score_SD
          )
        ),
      finite_current_wpvs =
        sum(
          is.finite(
            current_wpvs_SD
          )
        ),
      correlation_stable_current_wpvs =
        safe_cor(
          stable_score_SD,
          current_wpvs_SD
        ),
      raw_current_score_sd_min =
        safe_min(
          raw_current_score_sd
        ),
      raw_current_score_sd_max =
        safe_max(
          raw_current_score_sd
        ),
      .groups = "drop"
    )
}

)

prune_matched_sets <- function(
    dat,
    case_var = "treated"
) {

  dat %>%
    dplyr::filter(
      !is.na(
        .data[[case_var]]
      ),
      !is.na(
        match_set_id
      ),
      !is.na(
        unit_pidp
      )
    ) %>%
    dplyr::group_by(
      match_set_id
    ) %>%
    dplyr::filter(
      sum(
        .data[[case_var]] ==
          1L,
        na.rm = TRUE
      ) ==
        1L,
      sum(
        .data[[case_var]] ==
          0L,
        na.rm = TRUE
      ) >=
        1L
    ) %>%
    dplyr::ungroup()
}


matched_sample_summary <- function(
    dat,
    measure,
    T_now

  if (

    return(

          0L,

          0L,
        unique_control_people =
          0L,

  controls_per_set <- dat %>%

build_lat_matched_data <- function(
    T_now,
    measure_label
) {

  s <- score_lookups[[as.character(T_now)]] %>%
    dplyr::filter(
      measure ==
        measure_label
    ) %>%
    dplyr::transmute(
      pidp_key,
      baseline_wave,
      latent_baseline_id =
        baseline_id,
      stable_z =
        stable_score_SD,
      wpvs_z =
        current_wpvs_SD,
      raw_current_score_sd
    )

  joined <- incident_match %>%
    dplyr::left_join(
      s,
      by = c(
        "unit_pidp" =
          "pidp_key",
        "baseline_wave" =
          "baseline_wave"
      ),
      relationship =
        "many-to-one"
    )

  before <- joined %>%
    dplyr::summarise(
      T =
        T_now,
      measure =
        measure_label,
      matched_index_rows =
        dplyr::n(),
      matched_sets_before_score_restriction =
        dplyr::n_distinct(
          match_set_id
        ),
      cases_before_score_restriction =
        sum(
          treated ==
            1L
        ),
      controls_before_score_restriction =
        sum(
          treated ==
            0L
        ),
      rows_with_finite_latent_scores =
        sum(
          is.finite(
            stable_z
          ) &
            is.finite(
              wpvs_z
            )
        ),
      cases_with_finite_latent_scores =
        sum(
          treated ==
            1L &
            is.finite(
              stable_z
            ) &
            is.finite(
              wpvs_z
            )
        ),
      controls_with_finite_latent_scores =
        sum(
          treated ==
            0L &
            is.finite(
              stable_z
            ) &
            is.finite(
              wpvs_z
            )
        )
    )

  analytic <- joined %>%
    dplyr::filter(
      is.finite(
        stable_z
      ),
      is.finite(
        wpvs_z
      )
    ) %>%
    prune_matched_sets(
      case_var =
        "treated"
    )

  after <- matched_sample_summary(
    analytic,
    measure_label,
    T_now
  )

  list(
    data =
      analytic,
    join_audit =
      dplyr::bind_cols(
        before,
        after %>%
          dplyr::select(
            rows_after_pruning =
              rows,
            matched_sets_after_pruning =
              matched_sets,
            cases_after_pruning =
              cases,
            controls_after_pruning =
              controls,
            people_after_pruning =
              people,
            median_controls_per_set_after_pruning =
              median_controls_per_set
          )
      ),
    sample =
      after
  )
}

lat_joint_test <- function(
    fit,
    dat,
    measure,
    T_now
) {

  out <- tibble::tibble(
    design =
      "Incident LAT matched conditional logit",
    endpoint =
      "Incident LAT formation",
    measure =
      measure,
    T =
      T_now,
    test =
      "Stable + current WPVS = 0",
    statistic =
      NA_real_,
    df =
      2L,
    p_value =
      NA_real_,
    n_rows =
      nrow(dat),
    n_sets =
      dplyr::n_distinct(
        dat$match_set_id
      ),
    n_cases =
      sum(
        dat$treated ==
          1L
      ),
    n_people =
      dplyr::n_distinct(
        dat$unit_pidp
      ),
    status =
      "MODEL_MISSING"
  )

  if (is.null(fit)) {
    return(out)
  }

  terms <- c(
    "stable_z",
    "wpvs_z"
  )

  b <- stats::coef(
    fit
  )

  V <- tryCatch(
    stats::vcov(
      fit
    ),
    error =
      function(e) NULL
  )

  if (
    is.null(V) ||
      !all(
        terms %in%
          names(b)
      ) ||
      !all(
        terms %in%
          rownames(V)
      )
  ) {
    out$status <-
      "TERM_OR_VCOV_MISSING"
    return(out)
  }

  bb <- as.numeric(
    b[
      terms
    ]
  )

  VV <- as.matrix(
    V[
      terms,
      terms,
      drop = FALSE
    ]
  )

  invV <- safe_inverse(
    VV
  )

  if (
    is.null(
      invV
    ) ||
      any(
        !is.finite(
          bb
        )
      )
  ) {
    out$status <-
      "NONINVERTIBLE_OR_NONFINITE"
    return(out)
  }

  W <- as.numeric(
    t(bb) %*%
      invV %*%
      bb
  )

  out$statistic <-
    W

  out$p_value <-
    stats::pchisq(
      W,
      df = 2L,
      lower.tail = FALSE
    )

  out$status <-
    "OK"

  out
}


lat_component_difference <- function(
    fit,
    dat,
    measure,
    T_now
) {

  out <- tibble::tibble(
    design =
      "Incident LAT matched conditional logit",
    endpoint =
      "Incident LAT formation",
    measure =
      measure,
    T =
      T_now,
    test =
      "Stable coefficient = current WPVS coefficient",
    difference_log_or =
      NA_real_,
    robust_se_difference =
      NA_real_,
    statistic_z =
      NA_real_,
    p_value =
      NA_real_,
    n_rows =
      nrow(dat),
    n_sets =
      dplyr::n_distinct(
        dat$match_set_id
      ),
    status =
      "MODEL_MISSING"
  )

  if (is.null(fit)) {
    return(out)
  }

  terms <- c(
    "stable_z",
    "wpvs_z"
  )

  b <- stats::coef(
    fit
  )

  V <- tryCatch(
    stats::vcov(
      fit
    ),
    error =
      function(e) NULL
  )

  if (
    is.null(V) ||
      !all(
        terms %in%
          names(b)
      ) ||
      !all(
        terms %in%
          rownames(V)
      )
  ) {
    out$status <-
      "TERM_OR_VCOV_MISSING"
    return(out)
  }

  diff <-
    as.numeric(
      b[["stable_z"]] -
        b[["wpvs_z"]]
    )

  var_diff <-
    V[
      "stable_z",
      "stable_z"
    ] +
    V[
      "wpvs_z",
      "wpvs_z"
    ] -
    2 *
    V[
      "stable_z",
      "wpvs_z"
    ]

  if (
    !is.finite(
      var_diff
    ) ||
      var_diff <= 0
  ) {
    out$status <-
      "INVALID_VARIANCE"
    return(out)
  }

  se_diff <-
    sqrt(
      var_diff
    )

  z <-
    diff /
    se_diff

  out$difference_log_or <-
    diff

  out$robust_se_difference <-
    se_diff

  out$statistic_z <-
    z

  out$p_value <-
    2 *
    stats::pnorm(
      abs(z),
      lower.tail = FALSE
    )

  out$status <-
    "OK"

  out
}


fit_lat_selection <- function(
    dat,
    measure,
    T_now,
    save_model = TRUE
) {

  if (
    nrow(dat) == 0L ||
      dplyr::n_distinct(
        dat$match_set_id
      ) <
      2L
  ) {

    return(
      list(
        coefficients =
          empty_lat_coefficients(
            measure,
            T_now,
            "NO_ANALYTIC_SETS"
          ),
        joint =
          lat_joint_test(
            NULL,
            dat,
            measure,
            T_now
          ),
        difference =
          lat_component_difference(
            NULL,
            dat,
            measure,
            T_now
          ),
        model =
          NULL
      )
    )
  }

  fit <- tryCatch(
    survival::clogit(
      treated ~
        stable_z +
        wpvs_z +
        strata(
          match_set_id
        ) +
        cluster(
          unit_pidp
        ),
      data =
        dat,
      method =
        "efron",
      na.action =
        stats::na.omit
    ),
    error =
      function(e) e
  )

  stub <- paste0(
    "LAT_T",
    T_now,
    "_",
    ifelse(
      measure ==
        "Life satisfaction",
      "LS",
      "DISTRESS"
    )
  )

  if (
    inherits(
      fit,
      "error"
    )
  ) {

    write_txt(
      c(
        "MODEL FAILED",
        "",
        conditionMessage(
          fit
        )
      ),
      paste0(
        stub,
        "_ERROR.txt"
      ),
      "models"
    )

    return(
      list(
        coefficients =
          empty_lat_coefficients(
            measure,
            T_now,
            paste0(
              "MODEL_ERROR: ",
              conditionMessage(
                fit
              )
            )
          ),
        joint =
          lat_joint_test(
            NULL,
            dat,
            measure,
            T_now
          ),
        difference =
          lat_component_difference(
            NULL,
            dat,
            measure,
            T_now
          ),
        model =
          NULL
      )
    )
  }

  if (
    save_model
  ) {

    saveRDS(
      fit,
      file.path(
        MODEL_DIR,
        paste0(
          stub,
          ".rds"
        )
      )
    )

    write_txt(
      capture.output(
        summary(
          fit
        )
      ),
      paste0(
        stub,
        "_summary.txt"
      ),
      "models"
    )
  }

  b <- stats::coef(
    fit
  )

  V <- stats::vcov(
    fit
  )

  extract_one <- function(
      term,
      component
  ) {

    if (
      !term %in%
        names(b) ||
        !term %in%
        rownames(V)
    ) {

      return(
        tibble::tibble(
          design =
            "Incident LAT matched conditional logit",
          endpoint =
            "Incident LAT formation",
          measure =
            measure,
          T =
            T_now,
          component =
            component,
          term =
            term,
          estimate_log_or =
            NA_real_,
          robust_se =
            NA_real_,
          odds_ratio =
            NA_real_,
          conf_low =
            NA_real_,
          conf_high =
            NA_real_,
          p_value =
            NA_real_,
          n_rows =
            nrow(
              dat
            ),
          n_sets =
            dplyr::n_distinct(
              dat$match_set_id
            ),
          n_cases =
            sum(
              dat$treated ==
                1L
            ),
          n_controls =
            sum(
              dat$treated ==
                0L
            ),
          n_people =
            dplyr::n_distinct(
              dat$unit_pidp
            ),
          status =
            "TERM_MISSING"
        )
      )
    }

    est <- as.numeric(
      b[[term]]
    )

    se <- sqrt(
      as.numeric(
        V[
          term,
          term
        ]
      )
    )

    ci <- normal_ci(
      est,
      se
    )

    tibble::tibble(
      design =
        "Incident LAT matched conditional logit",
      endpoint =
        "Incident LAT formation",
      measure =
        measure,
      T =
        T_now,
      component =
        component,
      term =
        term,
      estimate_log_or =
        est,
      robust_se =
        se,
      odds_ratio =
        exp(
          est
        ),
      conf_low =
        exp(
          ci[
            1L
          ]
        ),
      conf_high =
        exp(
          ci[
            2L
          ]
        ),
      p_value =
        2 *
        stats::pnorm(
          abs(
            est /
              se
          ),
          lower.tail =
            FALSE
        ),
      n_rows =
        nrow(
          dat
        ),
      n_sets =
        dplyr::n_distinct(
          dat$match_set_id
        ),
      n_cases =
        sum(
          dat$treated ==
            1L
        ),
      n_controls =
        sum(
          dat$treated ==
            0L
        ),
      n_people =
        dplyr::n_distinct(
          dat$unit_pidp
        ),
      status =
        "OK"
    )
  }

  coefficients <- dplyr::bind_rows(
    extract_one(
      "stable_z",
      "Stable between-person component"
    ),
    extract_one(
      "wpvs_z",
      "Current within-person component (WPVS)"
    )
  )

  list(
    coefficients =
      coefficients,
    joint =
      lat_joint_test(
        fit,
        dat,
        measure,
        T_now
      ),
    difference =
      lat_component_difference(
        fit,
        dat,
        measure,
        T_now
      ),
    model =
      fit
  )
}

# 8. SELECT FIRST LATENT-SCORE-ELIGIBLE BASELINE PER NON-OVERLAPPING EPISODE
# ==============================================================================

build_latent_episode_baselines <- function(
    T_now,
    measure_label
) {

  s <- score_lookups[[as.character(T_now)]] %>%
    dplyr::filter(
      measure ==
        measure_label
    ) %>%
    dplyr::transmute(
      pidp =
        pidp_key,
      wave_n =
        baseline_wave,
      latent_baseline_id =
        baseline_id,
      stable_z =
        stable_score_SD,
      wpvs_z =
        current_wpvs_SD,
      raw_current_score_sd
    )

  joined <- clean_ep %>%
    dplyr::left_join(
      s,
      by = c(
        "pidp",
        "wave_n"
      ),
      relationship =
        "many-to-one"
    )

  eligible_episode_support <- joined %>%
    dplyr::group_by(
      episode_uid
    ) %>%
    dplyr::summarise(
      any_latent_score =
        any(
          is.finite(
            stable_z
          ) &
            is.finite(
              wpvs_z
          )
        ),
      n_latent_score_eligible_landmarks =
        sum(
          is.finite(
            stable_z
          ) &
            is.finite(
              wpvs_z
          )
        ),
      .groups =
        "drop"
    )

  selected <- joined %>%
    dplyr::filter(
      is.finite(
        stable_z
      ),
      is.finite(
        wpvs_z
      ),
      is.finite(
        clean_ym
      ),
      is.finite(
        episode_followup_end_ym
      ),
      clean_ym <
        episode_followup_end_ym
    ) %>%
    dplyr::arrange(
      pidp,
      episode_num,
      clean_ym,
      wave_n
    ) %>%
    dplyr::group_by(
      pidp,
      episode_num,
      episode_uid
    ) %>%
    dplyr::slice_head(
      n =
        1L
    ) %>%
    dplyr::ungroup() %>%
    dplyr::mutate(
      measure =
        measure_label,
      T =
        T_now,
      baseline_wave_n =
        wave_n,
      baseline_ym =
        clean_ym,
      baseline_delay_from_episode_start =
        baseline_ym -
        episode_start_ym,
      followup_end_ym =
        episode_followup_end_ym,
      followup_months =
        followup_end_ym -
        baseline_ym,
      baseline_year =
        suppressWarnings(
          as.integer(
            interview_year
          )
        ),
      recurrent_episode =
        as.integer(
          episode_num >=
            2L
        )
    ) %>%
    dplyr::filter(
      is.finite(
        followup_months
      ),
      followup_months >
        0,
      is.finite(
        baseline_delay_from_episode_start
      ),
      baseline_delay_from_episode_start >=
        0,
      is.finite(
        age_c
      ),
      !is.na(
        sex
      )
    )

  audit <- tibble::tibble(
    T =
      T_now,
    measure =
      measure_label,
    clean_landmarks =
      nrow(
        joined
      ),
    clean_episodes =
      dplyr::n_distinct(
        joined$episode_uid
      ),
    episodes_with_any_latent_score =
      sum(
        eligible_episode_support$any_latent_score
      ),
    selected_episode_baselines =
      nrow(
        selected
      ),
    selected_people =
      dplyr::n_distinct(
        selected$pidp
      ),
    recurrent_episode_baselines =
      sum(
        selected$recurrent_episode ==
          1L
      ),
    median_baseline_delay_months =
      safe_median(
        selected$baseline_delay_from_episode_start
      ),
    median_followup_months =
      safe_median(
        selected$followup_months
      )
  )

  list(
    selected =
      selected,
    episode_support =
      eligible_episode_support,
    audit =
      audit
  )
}

# 9. ATTACH FIRST EVENT AFTER EACH LATENT BASELINE
# ==============================================================================

attach_first_event <- function(
    baselines,
    event_tbl,
    event_stub
) {

  if (
    nrow(
      baselines
    ) ==
      0L
  ) {

    out <- baselines

    out[[paste0(event_stub, "_event")]] <- integer(
      0
    )

    out[[paste0(event_stub, "_ym")]] <- numeric(
      0
    )

    out[[paste0("months_to_", event_stub)]] <- numeric(
      0
    )

    out[[paste0(event_stub, "_source")]] <- character(
      0
    )

    return(out)
  }

  base_for_join <- baselines %>%
    dplyr::select(
      episode_uid,
      pidp,
      baseline_ym,
      followup_end_ym,
      end_is_next_episode
    )

  candidates <- base_for_join %>%
    dplyr::inner_join(
      event_tbl,
      by =
        "pidp",
      relationship =
        "many-to-many"
    ) %>%
    dplyr::mutate(
      eligible_event =
        in_followup_interval(
          event_ym =
            event_ym,
          baseline_ym =
            baseline_ym,
          end_ym =
            followup_end_ym,
          end_is_next_episode =
            end_is_next_episode
        ),
      phistory_source =
        stringr::str_detect(
          event_source,
          "^PHISTORY"
        )
    ) %>%
    dplyr::filter(
      eligible_event
    ) %>%
    dplyr::arrange(
      episode_uid,
      event_ym,
      dplyr::desc(
        phistory_source
      )
    ) %>%
    dplyr::group_by(
      episode_uid
    ) %>%
    dplyr::slice_head(
      n =
        1L
    ) %>%
    dplyr::ungroup() %>%
    dplyr::transmute(
      episode_uid,
      first_event_ym =
        event_ym,
      first_event_source =
        event_source
    )

  out <- baselines %>%
    dplyr::left_join(
      candidates,
      by =
        "episode_uid",
      relationship =
        "one-to-one"
    ) %>%
    dplyr::mutate(
      event_flag =
        dplyr::if_else(
          !is.na(
            first_event_ym
          ),
          1L,
          0L
        ),
      months_to_event =
        first_event_ym -
        baseline_ym
    )

  names(out)[
    names(out) ==
      "event_flag"
  ] <- paste0(
    event_stub,
    "_event"
  )

  names(out)[
    names(out) ==
      "first_event_ym"
  ] <- paste0(
    event_stub,
    "_ym"
  )

  names(out)[
    names(out) ==
      "months_to_event"
  ] <- paste0(
    "months_to_",
    event_stub
  )

  names(out)[
    names(out) ==
      "first_event_source"
  ] <- paste0(
    event_stub,
    "_source"
  )

  out
}


attach_primary_outcomes <- function(
    baselines
) {

  out <- attach_first_event(
    baselines,
    cohab_events_hybrid,
    "cohab"
  )

  out <- attach_first_event(
    out,
    marriage_events_hybrid,
    "marriage"
  )

  out %>%
    dplyr::mutate(
      any_core_event =
        dplyr::if_else(
          cohab_event ==
            1L |
            marriage_event ==
            1L,
          1L,
          0L
        ),
      first_core_ym =
        dplyr::case_when(
          cohab_event ==
            1L &
            marriage_event ==
            1L ~
              pmin(
                cohab_ym,
                marriage_ym,
                na.rm = TRUE
              ),
          cohab_event ==
            1L ~
              cohab_ym,
          marriage_event ==
            1L ~
              marriage_ym,
          TRUE ~
            NA_real_
        ),
      months_to_first_core =
        first_core_ym -
        baseline_ym
    )
}

ENDPOINT_ORDER <- c(
  "Cohabitation",
  "Marriage",
  "Cohabitation or marriage"
)


endpoint_estimand <- function(
    endpoint
) {

  switch(
    endpoint,
    "Cohabitation" =
      paste(
        "Cause-specific hazard of first subsequent cohabitation;",
        "marriage before/same month as cohabitation ends risk as a competing event"
      ),
    "Marriage" =
      paste(
        "Hazard of first subsequent marriage;",
        "cohabitation is allowed as an intermediate state"
      ),
    "Cohabitation or marriage" =
      paste(
        "Hazard of first subsequent cohabitation or marriage,",
        "whichever occurs first"
      ),
    endpoint
  )
}


make_endpoint_data <- function(
    dat,
    endpoint
) {

  cohab_time <- dplyr::if_else(
    dat$cohab_event ==
      1L &
      is.finite(
        dat$months_to_cohab
      ) &
      dat$months_to_cohab >
      0,
    dat$months_to_cohab,
    Inf
  )

  marriage_time <- dplyr::if_else(
    dat$marriage_event ==
      1L &
      is.finite(
        dat$months_to_marriage
      ) &
      dat$months_to_marriage >
      0,
    dat$months_to_marriage,
    Inf
  )

  censor_time <-
    dat$followup_months

  if (
    endpoint ==
      "Cohabitation"
  ) {

    time <- pmin(
      cohab_time,
      marriage_time,
      censor_time,
      na.rm = TRUE
    )

    event <- as.integer(
      is.finite(
        cohab_time
      ) &
        cohab_time <
        marriage_time &
        cohab_time <=
        censor_time
    )

    competing_marriage <- as.integer(
      is.finite(
        marriage_time
      ) &
        marriage_time <=
        cohab_time &
        marriage_time <=
        censor_time
    )

  } else if (
    endpoint ==
      "Marriage"
  ) {

    time <- pmin(
      marriage_time,
      censor_time,
      na.rm = TRUE
    )

    event <- as.integer(
      is.finite(
        marriage_time
      ) &
        marriage_time <=
        censor_time
    )

    competing_marriage <- rep(
      0L,
      nrow(
        dat
      )
    )

  } else if (
    endpoint ==
      "Cohabitation or marriage"
  ) {

    core_time <- pmin(
      cohab_time,
      marriage_time,
      na.rm = TRUE
    )

    time <- pmin(
      core_time,
      censor_time,
      na.rm = TRUE
    )

    event <- as.integer(
      is.finite(
        core_time
      ) &
        core_time <=
        censor_time
    )

    competing_marriage <- rep(
      0L,
      nrow(
        dat
      )
    )

  } else {

    stop(
      "Unknown endpoint: ",
      endpoint
    )
  }

  endpoint_label <-
    endpoint

  estimand_label <-
    endpoint_estimand(
      endpoint_label
    )

  dat %>%
    dplyr::mutate(
      endpoint =
        .env$endpoint_label,
      estimand =
        .env$estimand_label,
      time =
        as.numeric(
          .env$time
        ),
      event =
        as.integer(
          .env$event
        ),
      competing_marriage =
        as.integer(
          .env$competing_marriage
        )
    ) %>%
    dplyr::filter(
      is.finite(
        time
      ),
      time >
        0,
      event %in%
        c(
          0L,
          1L
        )
    )
}

build_cox_formula <- function(
    dat
) {

  rhs <- c(
    "stable_z",
    "wpvs_z",
    "age_c",
    "I(age_c^2)"
  )

  sex_nonmissing <- dat$sex[
    !is.na(
      dat$sex
    )
  ]

  if (
    length(
      unique(
        sex_nonmissing
      )
    ) >=
      2L
  ) {
    rhs <- c(
      rhs,
      "factor(sex)"
    )
  }

  if (
    length(
      unique(
        dat$recurrent_episode
      )
    ) >=
      2L
  ) {
    rhs <- c(
      rhs,
      "recurrent_episode"
    )
  }

  n_delay_values <- length(
    unique(
      dat$baseline_delay_from_episode_start[
        is.finite(
          dat$baseline_delay_from_episode_start
        )
      ]
    )
  )

  if (
    n_delay_values >=
      4L
  ) {

    rhs <- c(
      rhs,
      "splines::ns(baseline_delay_from_episode_start, df = 3)"
    )

  } else if (
    n_delay_values >=
      2L
  ) {

    rhs <- c(
      rhs,
      "baseline_delay_from_episode_start"
    )
  }

  if (
    length(
      unique(
        dat$baseline_year
      )
    ) >=
      2L
  ) {

    rhs <- c(
      rhs,
      "strata(baseline_year)"
    )
  }

  rhs <- c(
    rhs,
    "cluster(pidp)"
  )

  stats::as.formula(
    paste(
      "survival::Surv(time, event)",
      "~",
      paste(
        rhs,
        collapse =
          " + "
      )
    )
  )
}

cox_joint_test <- function(
    fit,
    dat,
    T_now,
    measure,
    endpoint
) {

  out <- tibble::tibble(
    design =
      "Non-overlapping clean-unpartnered Cox",
    endpoint =
      endpoint,
    estimand =
      endpoint_estimand(
        endpoint
      ),
    measure =
      measure,
    T =
      T_now,
    test =
      "Stable + current WPVS = 0",
    statistic =
      NA_real_,
    df =
      2L,
    p_value =
      NA_real_,
    n_rows =
      nrow(
        dat
      ),
    n_people =
      dplyr::n_distinct(
        dat$pidp
      ),
    n_events =
      sum(
        dat$event ==
          1L
      ),
    status =
      "MODEL_MISSING"
  )

  if (
    is.null(
      fit
    )
  ) {
    return(
      out
    )
  }

  terms <- c(
    "stable_z",
    "wpvs_z"
  )

  b <- stats::coef(
    fit
  )

  V <- tryCatch(
    stats::vcov(
      fit
    ),
    error =
      function(e) NULL
  )

  if (
    is.null(
      V
    ) ||
      !all(
        terms %in%
          names(
            b
          )
      ) ||
      !all(
        terms %in%
          rownames(
            V
          )
      )
  ) {

    out$status <-
      "TERM_OR_VCOV_MISSING"

    return(
      out
    )
  }

  bb <- as.numeric(
    b[
      terms
    ]
  )

  VV <- as.matrix(
    V[
      terms,
      terms,
      drop =
        FALSE
    ]
  )

  invV <- safe_inverse(
    VV
  )

  if (
    is.null(
      invV
    ) ||
      any(
        !is.finite(
          bb
        )
      )
  ) {

    out$status <-
      "NONINVERTIBLE_OR_NONFINITE"

    return(
      out
    )
  }

  W <- as.numeric(
    t(
      bb
    ) %*%
      invV %*%
      bb
  )

  out$statistic <-
    W

  out$p_value <-
    stats::pchisq(
      W,
      df =
        2L,
      lower.tail =
        FALSE
    )

  out$status <-
    "OK"

  out
}


cox_component_difference <- function(
    fit,
    dat,
    T_now,
    measure,
    endpoint
) {

  out <- tibble::tibble(
    design =
      "Non-overlapping clean-unpartnered Cox",
    endpoint =
      endpoint,
    estimand =
      endpoint_estimand(
        endpoint
      ),
    measure =
      measure,
    T =
      T_now,
    test =
      "Stable coefficient = current WPVS coefficient",
    difference_log_hr =
      NA_real_,
    robust_se_difference =
      NA_real_,
    statistic_z =
      NA_real_,
    p_value =
      NA_real_,
    n_rows =
      nrow(
        dat
      ),
    n_people =
      dplyr::n_distinct(
        dat$pidp
      ),
    n_events =
      sum(
        dat$event ==
          1L
      ),
    status =
      "MODEL_MISSING"
  )

  if (
    is.null(
      fit
    )
  ) {
    return(out)
  }

  terms <- c(
    "stable_z",
    "wpvs_z"
  )

  b <- stats::coef(
    fit
  )

  V <- tryCatch(
    stats::vcov(
      fit
    ),
    error =
      function(e) NULL
  )

  if (
    is.null(
      V
    ) ||
      !all(
        terms %in%
          names(
            b
          )
      ) ||
      !all(
        terms %in%
          rownames(
            V
          )
      )
  ) {

    out$status <-
      "TERM_OR_VCOV_MISSING"

    return(out)
  }

  diff <-
    as.numeric(
      b[["stable_z"]] -
        b[["wpvs_z"]]
    )

  var_diff <-
    V[
      "stable_z",
      "stable_z"
    ] +
    V[
      "wpvs_z",
      "wpvs_z"
    ] -
    2 *
    V[
      "stable_z",
      "wpvs_z"
    ]

  if (
    !is.finite(
      var_diff
    ) ||
      var_diff <=
      0
  ) {

    out$status <-
      "INVALID_VARIANCE"

    return(out)
  }

  se_diff <-
    sqrt(
      var_diff
    )

  z <-
    diff /
    se_diff

  out$difference_log_hr <-
    diff

  out$robust_se_difference <-
    se_diff

  out$statistic_z <-
    z

  out$p_value <-
    2 *
    stats::pnorm(
      abs(
        z
      ),
      lower.tail =
        FALSE
    )

  out$status <-
    "OK"

  out
}


cox_ph_diagnostic <- function(
    fit,
    T_now,
    measure,
    endpoint
) {

  if (
    is.null(
      fit
    )
  ) {

    return(
      tibble::tibble(
        T =
          T_now,
        measure =
          measure,
        endpoint =
          endpoint,
        term =
          NA_character_,
        rho =
          NA_real_,
        chisq =
          NA_real_,
        p_value =
          NA_real_,
        status =
          "MODEL_MISSING"
      )
    )
  }

  zph <- tryCatch(
    survival::cox.zph(
      fit,
      transform =
        "km"
    ),
    error =
      function(e) e
  )

  if (
    inherits(
      zph,
      "error"
    )
  ) {

    return(
      tibble::tibble(
        T =
          T_now,
        measure =
          measure,
        endpoint =
          endpoint,
        term =
          NA_character_,
        rho =
          NA_real_,
        chisq =
          NA_real_,
        p_value =
          NA_real_,
        status =
          paste0(
            "COX_ZPH_ERROR: ",
            conditionMessage(
              zph
            )
          )
      )
    )
  }

  tab <- as.data.frame(
    zph$table
  )

  tab$term <- rownames(
    tab
  )

  rownames(
    tab
  ) <- NULL

  tibble::tibble(
    T =
      T_now,
    measure =
      measure,
    endpoint =
      endpoint,
    term =
      tab$term,
    rho =
      if (
        "rho" %in%
          names(
            tab
          )
      ) {
        suppressWarnings(
          as.numeric(
            tab[["rho"]]
          )
        )
      } else {
        NA_real_
      },
    chisq =
      if (
        "chisq" %in%
          names(
            tab
          )
      ) {
        suppressWarnings(
          as.numeric(
            tab[["chisq"]]
          )
        )
      } else {
        NA_real_
      },
    p_value =
      if (
        "p" %in%
          names(
            tab
          )
      ) {
        suppressWarnings(
          as.numeric(
            tab[["p"]]
          )
        )
      } else {
        NA_real_
      },
    status =
      "OK"
  )
}


fit_one_cox <- function(
    dat,
    T_now,
    measure,
    endpoint,
    run_ph = FALSE,
    save_model = TRUE
) {

  d <- make_endpoint_data(
    dat,
    endpoint
  )

  if (
    nrow(
      d
    ) <
      20L ||
      sum(
        d$event ==
          1L
      ) <
      5L
  ) {

    return(
      list(
        coefficients =
          empty_cox_coefficients(
            T_now,
            measure,
            endpoint,
            "INSUFFICIENT_EVENTS"
          ),
        joint =
          cox_joint_test(
            NULL,
            d,
            T_now,
            measure,
            endpoint
          ),
        difference =
          cox_component_difference(
            NULL,
            d,
            T_now,
            measure,
            endpoint
          ),
        ph =
          if (
            run_ph
          ) {
            cox_ph_diagnostic(
              NULL,
              T_now,
              measure,
              endpoint
            )
          } else {
            NULL
          },
        model =
          NULL,
        data =
          d
      )
    )
  }

  formula <- build_cox_formula(
    d
  )

  fit <- tryCatch(
    survival::coxph(
      formula =
        formula,
      data =
        d,
      ties =
        COX_TIES,
      na.action =
        stats::na.omit,
      x =
        TRUE,
      y =
        TRUE,
      model =
        TRUE,
      control =
        survival::coxph.control(
          iter.max =
            50
        )
    ),
    error =
      function(e) e
  )

  stub <- paste0(
    "COX_T",
    T_now,
    "_",
    ifelse(
      measure ==
        "Life satisfaction",
      "LS_",
      "DISTRESS_"
    ),
    dplyr::case_when(
      endpoint ==
        "Cohabitation" ~
          "COHAB",
      endpoint ==
        "Marriage" ~
          "MARRIAGE",
      endpoint ==
        "Cohabitation or marriage" ~
          "COHAB_OR_MARRIAGE",
      TRUE ~
          "ENDPOINT"
    )
  )

  if (
    inherits(
      fit,
      "error"
    )
  ) {

    write_txt(
      c(
        "MODEL FAILED",
        "",
        "Formula:",
        paste(
          deparse(
            formula
          ),
          collapse =
            " "
        ),
        "",
        "Error:",
        conditionMessage(
          fit
        )
      ),
      paste0(
        stub,
        "_ERROR.txt"
      ),
      "models"
    )

    fit <- NULL
  }

  if (
    !is.null(
      fit
    ) &&
      save_model
  ) {

    saveRDS(
      fit,
      file.path(
        MODEL_DIR,
        paste0(
          stub,
          ".rds"
        )
      )
    )

    write_txt(
      c(
        "FORMULA",
        paste(
          deparse(
            formula
          ),
          collapse =
            " "
        ),
        "",
        capture.output(
          summary(
            fit
          )
        )
      ),
      paste0(
        stub,
        "_summary.txt"
      ),
      "models"
    )
  }

  if (
    is.null(
      fit
    )
  ) {

    return(
      list(
        coefficients =
          empty_cox_coefficients(
            T_now,
            measure,
            endpoint,
            "MODEL_ERROR"
          ),
        joint =
          cox_joint_test(
            NULL,
            d,
            T_now,
            measure,
            endpoint
          ),
        difference =
          cox_component_difference(
            NULL,
            d,
            T_now,
            measure,
            endpoint
          ),
        ph =
          if (
            run_ph
          ) {
            cox_ph_diagnostic(
              NULL,
              T_now,
              measure,
              endpoint
            )
          } else {
            NULL
          },
        model =
          NULL,
        data =
          d
      )
    )
  }

  b <- stats::coef(
    fit
  )

  V <- stats::vcov(
    fit
  )

  extract_one <- function(
      term,
      component
  ) {

    if (
      !term %in%
        names(
          b
        ) ||
        !term %in%
        rownames(
          V
        )
    ) {

      return(
        tibble::tibble(
          design =
            "Non-overlapping clean-unpartnered Cox",
          endpoint =
            endpoint,
          estimand =
            endpoint_estimand(
              endpoint
            ),
          measure =
            measure,
          T =
            T_now,
          component =
            component,
          term =
            term,
          estimate_log_hr =
            NA_real_,
          robust_se =
            NA_real_,
          hazard_ratio =
            NA_real_,
          conf_low =
            NA_real_,
          conf_high =
            NA_real_,
          p_value =
            NA_real_,
          n_rows =
            nrow(
              d
            ),
          n_people =
            dplyr::n_distinct(
              d$pidp
            ),
          n_events =
            sum(
              d$event ==
                1L
            ),
          n_competing_marriages =
            if (
              endpoint ==
                "Cohabitation"
            ) {
              sum(
                d$competing_marriage ==
                  1L
              )
            } else {
              NA_integer_
            },
          status =
            "TERM_MISSING"
        )
      )
    }

    est <- as.numeric(
      b[[term]]
    )

    se <- sqrt(
      as.numeric(
        V[
          term,
          term
        ]
      )
    )

    if (
      !is.finite(
        se
      ) ||
        se <=
        0
    ) {

      return(
        tibble::tibble(
          design =
            "Non-overlapping clean-unpartnered Cox",
          endpoint =
            endpoint,
          estimand =
            endpoint_estimand(
              endpoint
            ),
          measure =
            measure,
          T =
            T_now,
          component =
            component,
          term =
            term,
          estimate_log_hr =
            est,
          robust_se =
            se,
          hazard_ratio =
            exp(
              est
            ),
          conf_low =
            NA_real_,
          conf_high =
            NA_real_,
          p_value =
            NA_real_,
          n_rows =
            nrow(
              d
            ),
          n_people =
            dplyr::n_distinct(
              d$pidp
            ),
          n_events =
            sum(
              d$event ==
                1L
            ),
          n_competing_marriages =
            if (
              endpoint ==
                "Cohabitation"
            ) {
              sum(
                d$competing_marriage ==
                  1L
              )
            } else {
              NA_integer_
            },
          status =
            "INVALID_SE"
        )
      )
    }

    ci <- normal_ci(
      est,
      se
    )

    tibble::tibble(
      design =
        "Non-overlapping clean-unpartnered Cox",
      endpoint =
        endpoint,
      estimand =
        endpoint_estimand(
          endpoint
        ),
      measure =
        measure,
      T =
        T_now,
      component =
        component,
      term =
        term,
      estimate_log_hr =
        est,
      robust_se =
        se,
      hazard_ratio =
        exp(
          est
        ),
      conf_low =
        exp(
          ci[
            1L
          ]
        ),
      conf_high =
        exp(
          ci[
            2L
          ]
        ),
      p_value =
        2 *
        stats::pnorm(
          abs(
            est /
              se
          ),
          lower.tail =
            FALSE
        ),
      n_rows =
        nrow(
          d
        ),
      n_people =
        dplyr::n_distinct(
          d$pidp
        ),
      n_events =
        sum(
          d$event ==
            1L
        ),
      n_competing_marriages =
        if (
          endpoint ==
            "Cohabitation"
        ) {
          sum(
            d$competing_marriage ==
              1L
          )
        } else {
          NA_integer_
        },
      status =
        "OK"
    )
  }

  coefficients <- dplyr::bind_rows(
    extract_one(
      "stable_z",
      "Stable between-person component"
    ),
    extract_one(
      "wpvs_z",
      "Current within-person component (WPVS)"
    )
  )

  list(
    coefficients =
      coefficients,
    joint =
      cox_joint_test(
        fit,
        d,
        T_now,
        measure,
        endpoint
      ),
    difference =
      cox_component_difference(
        fit,
        d,
        T_now,
        measure,
        endpoint
      ),
    ph =
      if (
        run_ph
      ) {
        cox_ph_diagnostic(
          fit,
          T_now,
          measure,
          endpoint
        )
      } else {
        NULL
      },
    model =
      fit,
    data =
      d
  )
}
