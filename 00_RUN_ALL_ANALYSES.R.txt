# ----------------------------- USER OPTIONS -----------------------------------
STOP_ON_ERROR <- TRUE
RUN_PREPARATION <- TRUE
RUN_RDAS_DIAGNOSTIC_ADDON <- TRUE
RUN_OPTIONAL_PARALLEL_DIAGNOSTIC <- FALSE

# The component scripts retain the study paths documented in
# 00_README_REPRODUCIBILITY.txt. Edit their path blocks if reproducing elsewhere.

# -------------------------- FIND THIS DIRECTORY -------------------------------
cmd <- commandArgs(trailingOnly = FALSE)
file_arg <- grep("^--file=", cmd, value = TRUE)
source_file <- tryCatch(sys.frame(1)$ofile, error = function(e) NULL)
if (length(file_arg) >= 1L) {
  master_file <- sub("^--file=", "", file_arg[1L])
  SCRIPT_DIR <- dirname(normalizePath(master_file, winslash = "/", mustWork = TRUE))
} else if (!is.null(source_file) && nzchar(source_file)) {
  SCRIPT_DIR <- dirname(normalizePath(source_file, winslash = "/", mustWork = TRUE))
} else {
  SCRIPT_DIR <- normalizePath(getwd(), winslash = "/", mustWork = TRUE)
}
setwd(SCRIPT_DIR)

LOG_DIR <- file.path(SCRIPT_DIR, "reproduction_logs")
dir.create(LOG_DIR, recursive = TRUE, showWarnings = FALSE)

# ------------------------------ RUN ORDER -------------------------------------

    "01C_RDAS_gradient_pair_balanced_diagnostics.R.txt",

    "03_prepare_current_unpartnered_selection_base.R.txt",
    "03B_prepare_unpartnered_episode_structure.R.txt",
    "UKHLS_latent_stable_WPVS_final_measurement_stage_v2_1_WAVECENTERED_AGEFIX.R.txt",

steps$run <- TRUE
steps$run[steps$step %in% c("00A", "00B", "00C")] <- RUN_PREPARATION
steps$run[steps$step == "01C"] <- RUN_RDAS_DIAGNOSTIC_ADDON
steps$run[steps$step == "90"] <- RUN_OPTIONAL_PARALLEL_DIAGNOSTIC

missing_scripts <- steps$file[
  steps$run & !file.exists(file.path(SCRIPT_DIR, steps$file))
]
if (length(missing_scripts) > 0L) {
  stop("Missing required component script(s):\n", paste(missing_scripts, collapse = "\n"))
}

rscript <- file.path(
  R.home("bin"),
  if (.Platform$OS.type == "windows") "Rscript.exe" else "Rscript"
)
if (!file.exists(rscript)) rscript <- "Rscript"

run_one <- function(step, file, role) {
  script_path <- file.path(SCRIPT_DIR, file)
  log_path <- file.path(LOG_DIR, paste0(step, "_", sub("\\.txt$", "", basename(file)), ".log"))

  cat("\n===============================================================================\n")
  cat("STEP ", step, ": ", role, "\n", sep = "")
  cat("Script: ", script_path, "\n", sep = "")
  cat("Log:    ", log_path, "\n", sep = "")
  cat("===============================================================================\n")

  status <- system2(
    rscript,
    args = shQuote(script_path),
    stdout = log_path,
    stderr = log_path,
    wait = TRUE
  )

  ok <- identical(as.integer(status), 0L)
  cat(if (ok) "COMPLETED\n" else paste0("FAILED (exit status ", status, ")\n"))

  data.frame(
    step = step,
    file = file,
    role = role,
    attempted = TRUE,
    exit_status = as.integer(status),
    success = ok,
    log = normalizePath(log_path, winslash = "/", mustWork = FALSE),
    stringsAsFactors = FALSE
  )
}

run_log <- data.frame()
for (i in seq_len(nrow(steps))) {
  s <- steps[i, ]
  if (!isTRUE(s$run)) {
    run_log <- rbind(
      run_log,
      data.frame(
        step = s$step,
        file = s$file,
        role = s$role,
        attempted = FALSE,
        exit_status = NA_integer_,
        success = NA,
        log = NA_character_,
        stringsAsFactors = FALSE
      )
    )
    next
  }

  ans <- run_one(s$step, s$file, s$role)
  run_log <- rbind(run_log, ans)

  if (!isTRUE(ans$success) && STOP_ON_ERROR) {
    utils::write.csv(
      run_log,
      file.path(LOG_DIR, "MASTER_RUN_LOG.csv"),
      row.names = FALSE
    )
    stop("Reproduction stopped at step ", s$step, ". Inspect: ", ans$log)
  }
}

utils::write.csv(
  run_log,
  file.path(LOG_DIR, "MASTER_RUN_LOG.csv"),
  row.names = FALSE
)
writeLines(capture.output(sessionInfo()), file.path(LOG_DIR, "MASTER_SESSION_INFO.txt"))

cat("See 00_README_REPRODUCIBILITY.txt and 01_CODEBOOK.txt for interpretation.\n")
