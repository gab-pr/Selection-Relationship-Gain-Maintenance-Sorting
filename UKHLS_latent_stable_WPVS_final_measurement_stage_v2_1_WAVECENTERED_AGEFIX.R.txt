################################################################################
# UKHLS LATENT STABLE-TRAIT / WPVS FINAL MEASUREMENT-STAGE WORKFLOW v2
# ACTUAL-WAVE MEAN ADJUSTMENT
#
# PURPOSE
# -------
# Final measurement-stage workflow for the model-based stable-trait /
# within-person variability score (WPVS) decomposition that will be used in
# the revised H1 psychological-selection analyses.
#
# This version implements the final measurement-stage revision: before fitting
# the stable-trait/WPVS model, each psychological score is centered on the
# ACTUAL UKHLS SURVEY-WAVE mean estimated in the adult UKHLS panel. This removes
# common secular/wave-level shifts before stable between-person and within-person
# components are constructed.
#
# This script still DOES NOT estimate the final H1 relationship-transition
# effects. It produces the final T=4 score lookup for that second-stage analysis,
# with T=5 and T=6 retained as measurement-history sensitivities.
#
# CONCEPTUAL MODEL
# ----------------
# For one psychological outcome Y measured at T repeated occasions:
#
#   Y_it = mu_t + I_i + Y*_it
#
# where:
#   I_i    = stable between-person factor
#   Y*_it  = within-person variability score (WPVS)
#
# FINAL WAVE-ADJUSTED INPUT
# -------------------------
# The model is fitted to:
#
#   Yadj_it = Y_it - mean(Y_.t)
#
# where mean(Y_.t) is the unweighted adult-sample mean of the corresponding
# psychological measure in the respondent's ACTUAL UKHLS survey wave. Thus the
# latent decomposition is performed after removing common survey-wave shifts.
# For GHQ, wave means are calculated after favorable orientation (-GHQ12).
#
# Stable and within-person latent components are constrained to be uncorrelated.
# Correlation-preserving predicted scores need not themselves be empirically
# orthogonal; their correlation is therefore reported descriptively but is NOT
# used as a validity criterion.
#
# The measurement model follows the stable-trait / within-person structure used
# in Usami's two-step approach. The correlation-preserving score calculations
# below adapt the matrix formulas in the supplied supplementary R code.
#
# IMPORTANT PROSPECTIVE RESTRICTION
# ---------------------------------
# No observation after the focal clean-unpartnered baseline is ever used to
# construct or score the person's stable/WPVS predictors.
#
# Primary specification:
#   T = 4 occasions = current clean-unpartnered baseline + 3 preceding UKHLS
#   waves, all with valid psychological measurements.
#
# Sensitivity specifications:
#   T = 5 and T = 6.
#
# For the SEM measurement models, occasions are exact consecutive UKHLS waves.
# This is intentionally stricter than the manuscript's rolling historical mean
# and more closely matches the fixed-time-point structure assumed by the
# stable-trait/WPVS model. The script ALSO reports the less restrictive number
# of baselines with >=T valid measurements available anywhere up to baseline,
# so the cost of the consecutive-wave requirement is transparent.
#
# GHQ DIRECTION
# -------------
# GHQ-12 is reverse-oriented internally:
#
#   ghq_favorable = -GHQ12
#
# so higher stable and WPVS scores uniformly mean more favorable psychological
# functioning for both life satisfaction and GHQ.
#
# MEASUREMENT-MODEL ESTIMATION SAMPLE
# -----------------------------------
# To avoid treating overlapping baseline windows from the same respondent as
# independent observations in Step 1, the SEM is estimated on the FIRST
# eligible clean-unpartnered baseline per respondent for each outcome x T.
#
# Once population measurement-model parameters are estimated, scores are
# generated for ALL eligible clean-unpartnered baselines. These score lookups
# can later be merged into the existing incident-LAT and transition workflows.
#
# SCORE METHOD
# ------------
# The script calculates:
#   1. a correlation-preserving stable-trait predictor;
#   2. correlation-preserving WPVS for all T occasions;
#   3. the current-baseline WPVS (the final occasion), which is the prospective
#      state-dependent predictor of interest.
#
# Scores are also divided by a common raw current-score SD for interpretability.
#
# OUTPUT INTERPRETATION
# ---------------------
# The most important outputs are:
#
#   00_actual_wave_means.csv
#   00b_wave_centering_audit.csv
#   01_history_support_all_clean_baselines.csv
#   02_endpoint_support_lat_cohab_marriage.csv
#   03_measurement_model_fit.csv
#   04_measurement_model_variance_and_identification_audit.csv
#   05_score_diagnostics.csv
#   06_old_vs_latent_component_correlations.csv
#   07_temporal_spacing_diagnostics.csv
#   08_feasibility_summary.csv
#
# and:
#
#   data/final_wave_adjusted_latent_score_lookup_T4.rds
#   data/final_wave_adjusted_latent_score_lookup_T5.rds
#   data/final_wave_adjusted_latent_score_lookup_T6.rds
#
# The T=4 score lookup is the primary measurement input for the later H1
# re-analysis, conditional on the final diagnostics remaining admissible.
#
# REFERENCES / DESIGN BASIS
# -------------------------
# Usami's stable-trait / WPVS two-step framework and the user's supplied
# supplementary R code. This script adapts ONLY the disaggregation/scoring step.
# It does NOT implement DTRreg/SNMM causal estimation.
################################################################################

normalize_pidp_key <- function(x) {

  z <- x

  if (
    inherits(z, "haven_labelled") ||
      inherits(z, "labelled") ||
      inherits(z, "haven_labelled_spss")
  ) {
    z <- unclass(z)
  }

  if (is.factor(z)) {
    z <- as.character(z)
  }

  if (is.numeric(z) || is.integer(z)) {
    out <- ifelse(
      is.na(z),
      NA_character_,
      format(
        z,
        scientific = FALSE,
        trim = TRUE,
        digits = 22
      )
    )
    out <- sub("\\.0+$", "", out)
    return(out)
  }

  out <- trimws(as.character(z))
  out[out %in% c("", "NA", "NaN")] <- NA_character_
  sub("\\.0+$", "", out)
}


stata_month_from_year_month <- function(year, month_num) {

  y <- suppressWarnings(as.integer(year))
  m <- suppressWarnings(as.integer(month_num))

  ifelse(
    is.na(y) | is.na(m) | m < 1L | m > 12L,
    NA_integer_,
    as.integer((y - 1960L) * 12L + (m - 1L))
  )
}


write_csv_quiet <- function(x, filename, subdir = "tables") {

  target_dir <- switch(
    subdir,
    tables = TABLE_DIR,
    diagnostics = DIAG_DIR,
    TABLE_DIR
  )

  path <- file.path(target_dir, filename)

  readr::write_csv(
    x,
    path,
    na = "",
    progress = FALSE
  )

  invisible(path)
}


safe_sd <- function(x) {

  z <- suppressWarnings(as.numeric(x))
  s <- stats::sd(z, na.rm = TRUE)

  if (!is.finite(s) || s <= 0) {
    NA_real_
  } else {
    s
  }
}


safe_cor <- function(x, y) {

  ok <- is.finite(x) & is.finite(y)

  if (sum(ok) < 3L) {
    return(NA_real_)
  }

  stats::cor(
    x[ok],
    y[ok]
  )
}


safe_fit_measure <- function(fit, measure) {

  out <- tryCatch(
    lavaan::fitMeasures(fit, measure),
    error = function(e) NA_real_
  )

  if (
    length(out) == 0L ||
      !is.finite(out[[1L]])
  ) {
    NA_real_
  } else {
    unname(out[[1L]])
  }
}


safe_lavinspect <- function(fit, what, default = NA) {

  tryCatch(
    lavaan::lavInspect(fit, what),
    error = function(e) default
  )
}

# 3. MATRIX HELPERS FOR CORRELATION-PRESERVING SCORES
# ==============================================================================

symmetrize <- function(M) {
  (M + t(M)) / 2
}


matrix_psd_audit <- function(M, tol = PSD_TOL) {

  M <- symmetrize(M)

  ee <- tryCatch(
    eigen(M, symmetric = TRUE),
    error = function(e) NULL
  )

  if (is.null(ee)) {
    return(
      list(
        ok = FALSE,
        min_eigen = NA_real_,
        max_eigen = NA_real_,
        eigenvalues = NA_real_
      )
    )
  }

  vals <- ee$values

  list(
    ok = min(vals) >= -tol,
    min_eigen = min(vals),
    max_eigen = max(vals),
    eigenvalues = vals
  )
}


matrix_sqrt_psd <- function(M, tol = PSD_TOL) {

  M <- symmetrize(M)

  ee <- eigen(
    M,
    symmetric = TRUE
  )

  vals <- ee$values

  if (min(vals) < -tol) {
    stop(
      "Matrix is not positive semidefinite. Minimum eigenvalue = ",
      format(min(vals), scientific = TRUE)
    )
  }

  vals[vals < 0] <- 0

  ee$vectors %*%
    diag(
      sqrt(vals),
      nrow = length(vals)
    ) %*%
    t(ee$vectors)
}

}

# ==============================================================================
# 5A. FINAL REVISION: REMOVE ACTUAL UKHLS SURVEY-WAVE MEANS
# ==============================================================================
#
# The nuisance time mean is estimated in the adult UKHLS analysis panel, not in
# the clean-unpartnered risk set. This avoids defining the centering constant
# from the selected population whose relationship transitions are later modeled.
#
# Means are unweighted because their role here is measurement-model centering,
# not population prevalence estimation. The final H1 transition models should
# still retain their interview-year fixed effects.
# ==============================================================================

wave_means <- panel %>%
  filter(
    is.finite(age),
    age >= MIN_AGE,
    age <= MAX_AGE
  ) %>%
  group_by(wave_num) %>%
  summarise(
    n_ls_wave_mean = sum(is.finite(life_sat)),
    ls_wave_mean = if_else(
      n_ls_wave_mean > 0L,
      mean(life_sat, na.rm = TRUE),
      NA_real_
    ),
    n_ghq_wave_mean = sum(is.finite(ghq_favorable)),
    ghq_favorable_wave_mean = if_else(
      n_ghq_wave_mean > 0L,
      mean(ghq_favorable, na.rm = TRUE),
      NA_real_
    ),
    .groups = "drop"
  )

if (
  any(!is.finite(wave_means$ls_wave_mean)) ||
    any(!is.finite(wave_means$ghq_favorable_wave_mean))
) {
  stop(
    "At least one UKHLS wave has a non-finite psychological wave mean. ",
    "Inspect wave_means before proceeding."
  )
}

write_csv_quiet(
  wave_means,
  "00_actual_wave_means.csv"
)

  left_join(
    wave_means,
    by = "wave_num",

      is.finite(life_sat),
      life_sat - ls_wave_mean,

      is.finite(ghq_favorable),

# ==============================================================================
# 7. ALL-AVAILABLE PRIOR-HISTORY SUPPORT
# ==============================================================================
#
# This section asks:
#
#   At each clean-unpartnered baseline, how many valid measurements exist at
#   any UKHLS wave up to and including the baseline?
#
# These counts do NOT require consecutive waves and provide an upper bound on
# the support available if a later model accommodates irregular/missing timing.
# ==============================================================================

history_counts <- formation_base %>%
  dplyr::select(
    baseline_id,
    pidp_key,
    wave_num
  ) %>%
  left_join(
    panel %>%
      dplyr::select(
        pidp_key,
        wave_num,
        life_sat,
        ghq_favorable,
        ls_valid_cumulative,
        ghq_valid_cumulative
      ),
    by = c(
      "pidp_key",
      "wave_num"
    ),
    relationship = "many-to-one"
  ) %>%
  mutate(
    ls_total_valid_up_to_baseline =

# 8. BUILD STRICTLY PROSPECTIVE CONSECUTIVE-WAVE WINDOWS
# ==============================================================================

build_history_window <- function(
    baselines,
    panel_data,
    measure_var,
    measure_label,
    T_now
) {

  stopifnot(
    T_now >= 2L
  )

  out <- baselines %>%
    dplyr::select(
      baseline_id,
      pidp_key,
      baseline_wave = wave_num,
      everything()
    )

  lookup_base <- panel_data %>%
    dplyr::select(
      pidp_key,
      wave_num,
      interview_month_index,
      all_of(measure_var)
    )

  # y1 = oldest included wave
  # yT = current clean-unpartnered baseline
  for (lag_now in 0:(T_now - 1L)) {

    pos <- T_now - lag_now

    lookup_now <- lookup_base %>%
      transmute(
        pidp_key,
        join_wave = wave_num,
        !!paste0("y", pos) :=
          .data[[measure_var]],
        !!paste0("m", pos) :=
          interview_month_index
      )

    out <- out %>%
      mutate(
        join_wave =
          baseline_wave - lag_now
      ) %>%
      left_join(
        lookup_now,
        by = c(
          "pidp_key",
          "join_wave"
        ),
        relationship = "many-to-one"
      ) %>%
      dplyr::select(
        -join_wave
      )
  }

  y_names <- paste0(
    "y",
    seq_len(T_now)
  )

  m_names <- paste0(
    "m",
    seq_len(T_now)
  )

  Y <- as.matrix(
    out[
      ,
      y_names,
      drop = FALSE
    ]
  )

  M <- as.matrix(
    out[
      ,
      m_names,
      drop = FALSE
    ]
  )

  complete_measure <-
    rowSums(
      is.finite(Y)
    ) == T_now

  complete_months <-
    rowSums(
      is.finite(M)
    ) == T_now

  all_complete <-
    complete_measure &
    complete_months

  temporal_span <- rep(
    NA_real_,
    nrow(out)
  )

  min_gap <- rep(
    NA_real_,
    nrow(out)
  )

  max_gap <- rep(
    NA_real_,
    nrow(out)
  )

  if (any(all_complete)) {

    idx <- which(
      all_complete
    )

    temporal_span[idx] <-
      M[
        idx,
        T_now
      ] -
      M[
        idx,
        1L
      ]

    gap_mat <- t(
      apply(
        M[
          idx,
          ,
          drop = FALSE
        ],
        1L,
        diff
      )
    )

    if (T_now == 2L) {
      gap_mat <- matrix(
        gap_mat,
        ncol = 1L
      )
    }

    min_gap[idx] <-
      apply(
        gap_mat,
        1L,
        min
      )

    max_gap[idx] <-
      apply(
        gap_mat,
        1L,
        max
      )
  }

  out %>%
    mutate(
      measure = measure_label,
      T = T_now,
      exact_consecutive_complete =
        all_complete,
      total_span_months =
        temporal_span,
      min_adjacent_gap_months =
        min_gap,
      max_adjacent_gap_months =
        max_gap
    )
}

# ==============================================================================
# 11. STABLE-TRAIT / WITHIN-PERSON SEM SYNTAX
# ==============================================================================

make_stable_wpvs_model <- function(T_now) {

  y <- paste0(
    "y",
    seq_len(T_now)
  )

  w <- paste0(
    "w",
    seq_len(T_now)
  )

  lines <- character(0)

  # Stable trait with unit loadings.
  lines <- c(
    lines,
    paste0(
      "Trait =~ ",
      paste(
        paste0(
          "1*",
          y
        ),
        collapse = " + "
      )
    )
  )

  # Single-indicator within-person latent variables.
  for (j in seq_len(T_now)) {
    lines <- c(
      lines,
      paste0(
        w[j],
        " =~ 1*",
        y[j]
      )
    )
  }

  # Observed residual variances fixed at zero: exact decomposition.
  for (j in seq_len(T_now)) {
    lines <- c(
      lines,
      paste0(
        y[j],
        " ~~ 0*",
        y[j]
      )
    )
  }

  # Stable factor orthogonal to all WPVS factors.
  for (j in seq_len(T_now)) {
    lines <- c(
      lines,
      paste0(
        "Trait ~~ 0*",
        w[j]
      )
    )
  }

  # Stable-trait variance.
  lines <- c(
    lines,
    "Trait ~~ trait_var*Trait"
  )

  # First within-person variance.
  lines <- c(
    lines,
    "w1 ~~ wpvs_var_1*w1"
  )

  # First-order within-person dynamics.
  #
  # Betas and innovation variances are allowed to differ by lag. This is less
  # restrictive than forcing stationarity and gives a direct feasibility check.
  if (T_now >= 2L) {
    for (j in 2:T_now) {

      lines <- c(
        lines,
        paste0(
          "w",
          j,
          " ~ beta_",
          j,
          "*w",
          j - 1L
        )
      )

      lines <- c(
        lines,
        paste0(
          "w",
          j,
          " ~~ wpvs_innov_var_",
          j,
          "*w",
          j
        )
      )
    }
  }

  # Stable and within-person latent means fixed at zero.
  lines <- c(
    lines,
    "Trait ~ 0*1"
  )

  for (j in seq_len(T_now)) {
    lines <- c(
      lines,
      paste0(
        w[j],
        " ~ 0*1"
      )
    )
  }

  # Observed occasion-specific means are freely estimated.
  for (j in seq_len(T_now)) {
    lines <- c(
      lines,
      paste0(
        y[j],
        " ~ mu_",
        j,
        "*1"
      )
    )
  }

  paste(
    lines,
    collapse = "\n"
  )
}


# ==============================================================================
# 12. FIT ONE MEASUREMENT MODEL
# ==============================================================================

fit_one_measurement_model <- function(
    window_data,
    measure_label,
    T_now
) {

  y_names <- paste0(
    "y",
    seq_len(T_now)
  )

  complete_data <- window_data %>%
    filter(
      exact_consecutive_complete
    ) %>%
    arrange(
      pidp_key,
      baseline_wave
    )

  # Avoid pseudo-replication from overlapping windows in Step 1 parameter
  # estimation: use first eligible baseline per person.
  fit_data <- complete_data %>%
    group_by(
      pidp_key
    ) %>%
    slice_head(
      n = 1L
    ) %>%
    ungroup()

  if (nrow(fit_data) < 50L) {

    return(
      list(
        fit = NULL,
        complete_data = complete_data,
        fit_data = fit_data,
        error = paste0(
          "Too few independent respondent windows for SEM: N=",
          nrow(fit_data)
        ),
        syntax = make_stable_wpvs_model(T_now)
      )
    )
  }

  model_syntax <-
    make_stable_wpvs_model(
      T_now
    )

  fit <- tryCatch(
    lavaan::sem(
      model = model_syntax,
      data = fit_data %>%
        dplyr::select(
          all_of(
            y_names
          )
        ),
      estimator = "MLR",
      meanstructure = TRUE,
      fixed.x = FALSE,
      std.lv = FALSE,
      missing = "listwise"
    ),
    error = function(e) e
  )

  if (inherits(
    fit,
    "error"
  )) {

    return(
      list(
        fit = NULL,
        complete_data = complete_data,
        fit_data = fit_data,
        error = conditionMessage(fit),
        syntax = model_syntax
      )
    )
  }

  list(
    fit = fit,
    complete_data = complete_data,
    fit_data = fit_data,
    error = NA_character_,
    syntax = model_syntax
  )
}


# ==============================================================================
# 13. USAMI-STYLE CORRELATION-PRESERVING SCORE CALCULATION
# ==============================================================================

score_usami_univariate <- function(
    fit,
    fit_reference_data,
    score_data,
    T_now,
    tol = PSD_TOL
) {

  y_names <- paste0(
    "y",
    seq_len(T_now)
  )

  if (is.null(fit)) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "measurement_model_missing"
        )
      )
    )
  }

  pe <- lavaan::parameterEstimates(
    fit
  )

  phi_row <- pe %>%
    filter(
      lhs == "Trait",
      op == "~~",
      rhs == "Trait"
    )

  if (nrow(phi_row) != 1L) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "trait_variance_not_uniquely_identified"
        )
      )
    )
  }

  phi <- phi_row$est[[1L]]

  if (
    !is.finite(phi) ||
      phi <= 0
  ) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "nonpositive_trait_variance",
          trait_variance = phi
        )
      )
    )
  }

  Y_fit <- as.matrix(
    fit_reference_data[
      ,
      y_names,
      drop = FALSE
    ]
  )

  Y_score <- as.matrix(
    score_data[
      ,
      y_names,
      drop = FALSE
    ]
  )

  if (
    any(
      !is.finite(
        Y_fit
      )
    ) ||
      any(
        !is.finite(
          Y_score
        )
      )
  ) {
    stop(
      "score_usami_univariate requires complete finite history windows."
    )
  }

  # Time-specific means from the fitted SEM.
  mu <- tryCatch(
    lavaan::lavInspect(
      fit,
      "mean.ov"
    ),
    error = function(e) NULL
  )

  if (
    is.null(mu) ||
      length(mu) != T_now ||
      any(
        !is.finite(
          mu
        )
      )
  ) {

    mu <- colMeans(
      Y_fit
    )
  }

  mu <- as.numeric(
    mu[
      y_names
    ]
  )

  if (
    length(mu) != T_now ||
      any(
        !is.finite(
          mu
        )
      )
  ) {
    mu <- colMeans(
      Y_fit
    )
  }

  Y_fit_c <- sweep(
    Y_fit,
    2L,
    mu,
    "-"
  )

  Y_score_c <- sweep(
    Y_score,
    2L,
    mu,
    "-"
  )

  # Following the supplied Usami code, Sigma is the observed covariance matrix.
  Sigma <- stats::cov(
    Y_fit_c
  )

  Sigma <- symmetrize(
    Sigma
  )

  sigma_audit <-
    matrix_psd_audit(
      Sigma,
      tol
    )

  if (!sigma_audit$ok) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "observed_covariance_not_psd",
          trait_variance = phi,
          sigma_min_eigen =
            sigma_audit$min_eigen
        )
      )
    )
  }

  lambda <- matrix(
    1,
    nrow = T_now,
    ncol = 1L
  )

  Sigma_inv <- safe_ginv(
    Sigma
  )

  trait_denom <- as.numeric(
    t(lambda) %*%
      Sigma_inv %*%
      lambda
  )

  if (
    !is.finite(
      trait_denom
    ) ||
      trait_denom <= 0
  ) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "invalid_trait_score_denominator",
          trait_variance = phi
        )
      )
    )
  }

  # Correlation-preserving stable-trait predictor:
  #
  # sqrt(phi) / sqrt(lambda' Sigma^-1 lambda)
  #   * lambda' Sigma^-1 Y
  trait_weight_row <-
    sqrt(phi) /
    sqrt(trait_denom) *
    t(lambda) %*%
    Sigma_inv

  stable_score <- as.numeric(
    Y_score_c %*%
      t(
        trait_weight_row
      )
  )

  # Build the stable-trait covariance contribution.
  I_mat <- phi *
    (
      lambda %*%
        t(lambda)
    )

  # Within-person covariance target.
  Psi <- symmetrize(
    Sigma -
      I_mat
  )

  psi_audit <-
    matrix_psd_audit(
      Psi,
      tol
    )

  if (!psi_audit$ok) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "within_person_covariance_not_psd",
          trait_variance = phi,
          sigma_min_eigen =
            sigma_audit$min_eigen,
          psi_min_eigen =
            psi_audit$min_eigen
        )
      )
    )
  }

  Psi_half <-
    matrix_sqrt_psd(
      Psi,
      tol
    )

  # Adaptation of the correlation-preserving WPVS matrix formula in the
  # supplied supplementary R code.
  SIG <-
    Psi %*%
    Psi_half %*%
    Sigma_inv %*%
    Psi %*%
    Psi_half

  SIG <- symmetrize(
    SIG
  )

  sig_audit <-
    matrix_psd_audit(
      SIG,
      tol
    )

  if (!sig_audit$ok) {
    return(
      list(
        scores = NULL,
        audit = tibble(
          scoring_status =
            "WPVS_intermediate_matrix_not_psd",
          trait_variance = phi,
          sigma_min_eigen =
            sigma_audit$min_eigen,
          psi_min_eigen =
            psi_audit$min_eigen,
          wpvs_intermediate_min_eigen =
            sig_audit$min_eigen
        )
      )
    )
  }

  SIG_half <-
    matrix_sqrt_psd(
      SIG,
      tol
    )

  W_wpvs <-
    Psi_half %*%
    safe_ginv(
      SIG_half
    ) %*%
    Psi %*%
    Psi_half %*%
    Sigma_inv

  WPVS <- Y_score_c %*%
    t(
      W_wpvs
    )

  colnames(WPVS) <-
    paste0(
      "wpvs_",
      seq_len(T_now)
    )

  score_tibble <- bind_cols(
    score_data %>%
      dplyr::select(
        baseline_id,
        pidp_key,
        baseline_wave
      ),
    tibble(
      stable_score_raw =
        stable_score
    ),
    as_tibble(
      WPVS
    )
  )

  # Empirical score audit on the SEM estimation sample.
  stable_fit <- as.numeric(
    Y_fit_c %*%
      t(
        trait_weight_row
      )
  )

  wpvs_fit <- Y_fit_c %*%
    t(
      W_wpvs
    )

  empirical_wpvs_cov <-
    stats::cov(
      wpvs_fit
    )

  stable_wpvs_current_cor <-
    safe_cor(
      stable_fit,
      wpvs_fit[
        ,
        T_now
      ]
    )

  audit <- tibble(
    scoring_status = "estimated",
    trait_variance = phi,
    sigma_min_eigen =
      sigma_audit$min_eigen,
    psi_min_eigen =
      psi_audit$min_eigen,
    wpvs_intermediate_min_eigen =
      sig_audit$min_eigen,
    stable_score_variance_fit_sample =
      stats::var(
        stable_fit
      ),
    target_trait_variance =
      phi,
    abs_stable_variance_minus_trait_variance =
      abs(
        stats::var(
          stable_fit
        ) -
          phi
      ),
    max_abs_wpvs_covariance_error =
      max(
        abs(
          empirical_wpvs_cov -
            Psi
        ),
        na.rm = TRUE
      ),
    stable_vs_current_wpvs_correlation_fit_sample =
      stable_wpvs_current_cor
  )

  list(
    scores = score_tibble,
    audit = audit,
    Sigma = Sigma,
    Psi = Psi,
    means = mu,
    trait_weights = trait_weight_row,
    wpvs_weights = W_wpvs
  )
}

# ==============================================================================

fit_objects <- list()
fit_rows <- list()
var_rows <- list()
score_audit_rows <- list()
score_lookup_by_T <- list()

fit_counter <- 0L
var_counter <- 0L
score_counter <- 0L


for (i in seq_len(
  nrow(MEASURE_SPECS)
)) {

  measure_label <-
    MEASURE_SPECS$measure[[i]]

  panel_var <-
    MEASURE_SPECS$panel_var[[i]]

  raw_var <-
    MEASURE_SPECS$raw_var[[i]]

  wave_mean_var <-
    MEASURE_SPECS$wave_mean_var[[i]]

  for (T_now in T_VALUES) {

    key <- paste(
      measure_label,
      T_now,
      sep = "__T"
    )

    w <- history_windows[[key]]

    fitted_obj <-
      fit_one_measurement_model(
        window_data = w,
        measure_label =
          measure_label,
        T_now = T_now
      )

    model_key <- paste0(
      ifelse(
        measure_label ==
          "Life satisfaction",
        "LS",
        "GHQ"
      ),
      "_T",
      T_now
    )

    fit_objects[[model_key]] <-
      fitted_obj

    fit_counter <- fit_counter + 1L

    if (is.null(
      fitted_obj$fit
    )) {

      fit_rows[[fit_counter]] <-
        tibble(
          model = model_key,
          measure = measure_label,
          T = T_now,
          n_all_eligible_windows =
            nrow(
              fitted_obj$complete_data
            ),
          n_independent_people_sem =
            nrow(
              fitted_obj$fit_data
            ),
          converged = FALSE,
          post_check_ok = FALSE,
          df = NA_real_,
          chisq = NA_real_,
          p_value = NA_real_,
          cfi = NA_real_,
          tli = NA_real_,
          rmsea = NA_real_,
          srmr = NA_real_,
          aic = NA_real_,
          bic = NA_real_,
          status = paste0(
            "fit_failed: ",
            fitted_obj$error
          )
        )

      next
    }

    fit <- fitted_obj$fit

    converged <-
      isTRUE(
        safe_lavinspect(
          fit,
          "converged",
          FALSE
        )
      )

    post_check <-
      safe_lavinspect(
        fit,
        "post.check",
        NA
      )

    if (
      length(post_check) != 1L
    ) {
      post_check <- NA
    }

    fit_rows[[fit_counter]] <-
      tibble(
        model = model_key,
        measure = measure_label,
        T = T_now,
        n_all_eligible_windows =
          nrow(
            fitted_obj$complete_data
          ),
        n_independent_people_sem =
          nrow(
            fitted_obj$fit_data
          ),
        converged = converged,
        post_check_ok =
          as.logical(
            post_check
          ),
        df =
          safe_fit_measure(
            fit,
            "df"
          ),
        chisq =
          safe_fit_measure(
            fit,
            "chisq"
          ),
        p_value =
          safe_fit_measure(
            fit,
            "pvalue"
          ),
        cfi =
          safe_fit_measure(
            fit,
            "cfi"
          ),
        tli =
          safe_fit_measure(
            fit,
            "tli"
          ),
        rmsea =
          safe_fit_measure(
            fit,
            "rmsea"
          ),
        srmr =
          safe_fit_measure(
            fit,
            "srmr"
          ),
        aic =
          safe_fit_measure(
            fit,
            "aic"
          ),
        bic =
          safe_fit_measure(
            fit,
            "bic"
          ),
        status =
          ifelse(
            converged,
            "estimated",
            "not_converged"
          )
      )

    saveRDS(
      fit,
      file.path(
        MODEL_DIR,
        paste0(
          model_key,
          "_stable_WPVS_measurement_model.rds"
        )
      )
    )

    pe <- lavaan::parameterEstimates(
      fit
    )

    trait_var <- pe %>%
      filter(
        lhs == "Trait",
        op == "~~",
        rhs == "Trait"
      ) %>%
      pull(
        est
      )

    wpvs_variances <- pe %>%
      filter(
        op == "~~",
        lhs == rhs,
        grepl(
          "^w[0-9]+$",
          lhs
        )
      ) %>%
      dplyr::select(
        latent = lhs,
        estimate = est
      )

    var_counter <- var_counter + 1L

    var_rows[[var_counter]] <-
      tibble(
        model = model_key,
        measure = measure_label,
        T = T_now,
        trait_variance =
          ifelse(
            length(trait_var) == 1L,
            trait_var,
            NA_real_
          ),
        min_wpvs_or_innovation_variance =
          ifelse(
            nrow(wpvs_variances) > 0L,
            min(
              wpvs_variances$estimate,
              na.rm = TRUE
            ),
            NA_real_
          ),
        n_nonpositive_wpvs_or_innovation_variances =
          sum(
            wpvs_variances$estimate <= 0,
            na.rm = TRUE
          ),
        sem_sample_small_flag =
          nrow(
            fitted_obj$fit_data
          ) <
          DESCRIPTIVE_SMALL_SEM_N
      )

    score_obj <-
      score_usami_univariate(
        fit =
          fit,
        fit_reference_data =
          fitted_obj$fit_data,
        score_data =
          fitted_obj$complete_data,

          T_now
      )

    score_counter <- score_counter + 1L

    score_audit_rows[[score_counter]] <-
      bind_cols(
        tibble(
          model = model_key,
          measure = measure_label,
          T = T_now,
          n_scored =
            ifelse(
              is.null(
                score_obj$scores
              ),
              0L,
              nrow(
                score_obj$scores
              )
            )
        ),
        score_obj$audit
      )

    if (
      !is.null(
        score_obj$scores
      )
    ) {

      scores <- score_obj$scores

      # Scale both latent components by the SAME SD of the RAW current score,
      # not by the SD of the wave-adjusted score. This preserves the manuscript's
      # common observed-score metric for comparing stable and state components.
      baseline_raw_lookup <- fitted_obj$complete_data %>%
        dplyr::select(
          baseline_id,
          pidp_key,
          baseline_wave
        ) %>%
        left_join(
          panel %>%
            transmute(
              pidp_key,
              baseline_wave = wave_num,
              baseline_raw_score = .data[[raw_var]],
              baseline_wave_mean = .data[[wave_mean_var]],
              baseline_wave_adjusted_score = .data[[panel_var]],
              baseline_interview_year = interview_year,
              baseline_interview_month = interview_month
            ),
          by = c(
            "pidp_key",
            "baseline_wave"
          ),
          relationship = "many-to-one"
        )

      current_sd <- safe_sd(
        baseline_raw_lookup$baseline_raw_score
      )

      if (!is.finite(current_sd) || current_sd <= 0) {
        stop(
          "Could not obtain a valid raw current-score SD for ",
          measure_label,
          " T=",
          T_now
        )
      }

      scores <- scores %>%
        left_join(
          baseline_raw_lookup %>%
            dplyr::select(
              baseline_id,
              baseline_raw_score,
              baseline_wave_mean,
              baseline_wave_adjusted_score,
              baseline_interview_year,
              baseline_interview_month
            ),
          by = "baseline_id",
          relationship = "many-to-one"
        ) %>%
        mutate(
          measure =
            measure_label,
          T =
            T_now,
          wave_adjustment =
            "actual UKHLS survey-wave mean removed before decomposition",
          raw_current_score_sd =
            current_sd,
          stable_score_SD =
            stable_score_raw /
            current_sd,
          current_wpvs_raw =
            .data[[paste0("wpvs_", T_now)]],
          current_wpvs_SD =
            current_wpvs_raw /
            current_sd
        )

      if (
        is.null(
          score_lookup_by_T[[as.character(T_now)]]
        )
      ) {

        score_lookup_by_T[[as.character(T_now)]] <- scores

      } else {

        score_lookup_by_T[[as.character(T_now)]] <- bind_rows(
          score_lookup_by_T[[as.character(T_now)]],
          scores
        )
      }

      # Save scoring matrices for full reproducibility.
      saveRDS(
        list(
          Sigma = score_obj$Sigma,
          Psi = score_obj$Psi,
          means = score_obj$means,
          trait_weights =
            score_obj$trait_weights,
          wpvs_weights =
            score_obj$wpvs_weights
        ),
        file.path(
          MODEL_DIR,
          paste0(
            model_key,
            "_correlation_preserving_scoring_matrices.rds"
          )
        )
      )
    }
  }
}
