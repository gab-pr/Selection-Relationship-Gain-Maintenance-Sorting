################################################################################
# UKHLS PATHWAY-BASED ANTICIPATION ADD-ON V1.2 (PIDP TYPE FIX)
#
# PURPOSE
# -------
# This stand-alone add-on implements pathway-based tests of psychological
# anticipation without replacing or modifying the validated upstream workflows.
#
# CONCEPTUAL DESIGN
# -----------------
# A. COHABITATION PATHWAY TEST
#    Clean incident LAT formation -> future pathway within 36 months:
#      (1) probable LAT -> cohabitation,
#      (2) remains LAT through the fixed horizon,
#      (3) observed LAT exit/no partner through the fixed horizon.
#    Future cohabiters are matched to incident LAT relationships that remain LAT.
#    Psychological trajectories are followed from the pre-LAT period through
#    post-LAT relationship-duration stages. The primary estimand is differential
#    within-person psychological change after LAT formation.
#
# B. BALANCED LAT TARGET-BIN SENSITIVITIES
#    Re-estimate the future-cohabitation vs stable-LAT contrast separately for
#    12-23 and 24-35 months, requiring outcome-complete observations at both the
#    first-LAT reference and the target stage within each retained matched set.
#
# C. SUBJECTIVE COHABITATION ANTICIPATION (NCRR11 CHANGE)
#    Test whether within-LAT strengthening of intention to live together
#    (previous interview -> current interview) predicts subsequent psychological
#    change (current interview -> next interview), preserving temporal order.
#
# D. SUBJECTIVE MARRIAGE ANTICIPATION (NCRR12 CHANGE)
#    Audit repeated within-LAT variation in marriage likelihood and test whether
#    strengthening marriage expectation predicts subsequent psychological change.
#    NCRR12 is explicitly treated as LAT-stage marriage expectation, NOT as a
#    cohabiter-specific marriage-expectation measure.
#
# E. FIXED-HORIZON MARRIAGE PATHWAY TEST
#    Cohabitation onset -> pathway within 36 months:
#      (1) marriage within 36 months,
#      (2) remains cohabiting/unmarried through 36 months,
#      (3) breakup within 36 months.
#    Future marriers are matched to cohabitations stable through the horizon.
#    Psychological trajectories are compared across cohabitation-duration stages.
#
# INTERPRETIVE RULE
# -----------------
# Favorable post-relationship-formation trajectory divergence among future
# progressors is CONSISTENT WITH a transition-bound developmental/anticipatory
# process, but it is not a causal effect of anticipation. Subjective expectation
# change provides complementary evidence more specific to anticipation.
#
# IMPORTANT
# ---------
# * Objective time-to-event proximity is NOT the primary anticipation test here.
# * Controls are fixed-horizon stable nonprogressors, not "never transitioners."
# * Breakup pathways are kept separate and are not pooled with stable controls.
# * All primary pathway models estimate CHANGE relative to relationship onset;
#   psychological level differences alone are not treated as anticipation.
################################################################################

write_csv_quiet <- function(x, filename, subdir = "tables") {
  path <- file.path(OUT_DIR, subdir, filename)
  readr::write_csv(x, path, na = "", progress = FALSE)
  invisible(path)
}

write_txt <- function(x, filename, subdir = "tables") {
  path <- file.path(OUT_DIR, subdir, filename)
  writeLines(as.character(x), path, useBytes = TRUE)
  invisible(path)
}

safe_sd <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) < 2L) return(NA_real_)
  stats::sd(z)
}

safe_mean <- function(x) {
  z <- suppressWarnings(as.numeric(x))
  z <- z[is.finite(z)]
  if (length(z) == 0L) return(NA_real_)
  mean(z)
}

safe_nobs <- function(model) {
  if (is.null(model)) return(NA_integer_)
  out <- tryCatch(stats::nobs(model), error = function(e) NA_integer_)
  suppressWarnings(as.integer(out))
}

save_model_bundle <- function(model, name) {
  if (is.null(model)) return(invisible(NULL))

  saveRDS(model, file.path(OUT_DIR, "models", paste0(name, ".rds")))

  b <- stats::coef(model)
  if (length(b) > 0L) {
    se <- fixest::se(model)
    pv <- fixest::pvalue(model)
    ci <- stats::confint(model)

    tab <- tibble::tibble(
      term = names(b),
      estimate = as.numeric(b),
      std_error = as.numeric(se[names(b)]),
      conf_low = as.numeric(ci[names(b), 1]),
      conf_high = as.numeric(ci[names(b), 2]),
      p_value = as.numeric(pv[names(b)]),
      nobs = safe_nobs(model)
    )

    write_csv_quiet(tab, paste0(name, "_coefficients.csv"))
  }

  write_txt(capture.output(summary(model)), paste0(name, "_summary.txt"))
  invisible(model)
}

safe_feols <- function(formula, data, name, ...) {
  fit <- tryCatch(
    fixest::feols(formula, data = data, ...),
    error = function(e) e
  )

  if (inherits(fit, "error")) {
    write_txt(
      c("MODEL FAILED", "", conditionMessage(fit)),
      paste0(name, "_ERROR.txt")
    )
    return(NULL)
  }

  save_model_bundle(fit, name)
  fit
}

safe_feglm <- function(formula, data, family, name, ...) {
  fit <- tryCatch(
    fixest::feglm(formula, data = data, family = family, ...),
    error = function(e) e
  )

  if (inherits(fit, "error")) {
    write_txt(
      c("MODEL FAILED", "", conditionMessage(fit)),
      paste0(name, "_ERROR.txt")
    )
    return(NULL)
  }

  save_model_bundle(fit, name)
  fit
}

extract_term <- function(
    model,
    term,
    analysis,
    outcome,
    contrast = NA_character_
) {
  out <- tibble::tibble(
    analysis = analysis,
    outcome = outcome,
    contrast = contrast,
    term = term,
    estimate = NA_real_,
    std_error = NA_real_,
    conf_low = NA_real_,
    conf_high = NA_real_,
    p_value = NA_real_,
    nobs = if (is.null(model)) NA_integer_ else safe_nobs(model),
    status = if (is.null(model)) "model_missing" else "term_missing"
  )

  if (is.null(model)) return(out)

  b <- stats::coef(model)
  if (!term %in% names(b)) return(out)

  se <- fixest::se(model)
  pv <- fixest::pvalue(model)
  ci <- stats::confint(model)

  tibble::tibble(
    analysis = analysis,
    outcome = outcome,
    contrast = contrast,
    term = term,
    estimate = as.numeric(b[term]),
    std_error = as.numeric(se[term]),
    conf_low = as.numeric(ci[term, 1]),
    conf_high = as.numeric(ci[term, 2]),
    p_value = as.numeric(pv[term]),
    nobs = safe_nobs(model),
    status = "estimated"
  )
}

extract_first_matching_term <- function(
    model,
    required_substrings,
    analysis,
    outcome,
    contrast
) {
  if (is.null(model)) {
    return(extract_term(NULL, NA_character_, analysis, outcome, contrast))
  }

  nm <- names(stats::coef(model))
  keep <- rep(TRUE, length(nm))
  for (s in required_substrings) {
    keep <- keep & grepl(s, nm, fixed = TRUE)
  }
  hits <- nm[keep]

  if (length(hits) == 0L) {
    return(extract_term(model, "__NO_MATCH__", analysis, outcome, contrast))
  }

  extract_term(model, hits[[1L]], analysis, outcome, contrast)
}

wald_terms <- function(model, terms, analysis, outcome, test) {
  base <- tibble::tibble(
    analysis = analysis,
    outcome = outcome,
    test = test,
    statistic = NA_real_,
    df = NA_integer_,
    p_value = NA_real_,
    n_terms = length(terms),
    matched_terms = paste(terms, collapse = " | "),
    nobs = if (is.null(model)) NA_integer_ else safe_nobs(model),
    status = if (is.null(model)) "model_missing" else "not_estimated"
  )

  if (is.null(model) || length(terms) == 0L) return(base)

  b_all <- stats::coef(model)
  V_all <- stats::vcov(model)
  terms <- intersect(terms, names(b_all))
  if (length(terms) == 0L) return(base)

  b <- as.numeric(b_all[terms])
  V <- as.matrix(V_all[terms, terms, drop = FALSE])

  if (any(!is.finite(b)) || any(!is.finite(V))) {
    base$status <- "nonfinite_coef_or_vcov"
    return(base)
  }

  qrV <- qr(V)
  rankV <- qrV$rank
  if (rankV == 0L) {
    base$status <- "vcov_rank_zero"
    return(base)
  }

  ee <- eigen(V, symmetric = TRUE)
  tol <- max(dim(V)) * max(abs(ee$values), 1) * .Machine$double.eps * 100
  idx <- which(ee$values > tol)
  if (length(idx) == 0L) {
    base$status <- "generalized_inverse_failed"
    return(base)
  }
  Q <- ee$vectors[, idx, drop = FALSE]
  Vinv <- Q %*% diag(1 / ee$values[idx], nrow = length(idx)) %*% t(Q)
  rankV <- length(idx)

  W <- as.numeric(t(b) %*% Vinv %*% b)
  p <- stats::pchisq(W, df = rankV, lower.tail = FALSE)

  tibble::tibble(
    analysis = analysis,
    outcome = outcome,
    test = test,
    statistic = W,
    df = as.integer(rankV),
    p_value = p,
    n_terms = length(terms),
    matched_terms = paste(terms, collapse = " | "),
    nobs = safe_nobs(model),
    status = ifelse(rankV == length(terms), "estimated", "estimated_reduced_rank")
  )
}

find_terms_all <- function(model, required_substrings) {
  if (is.null(model)) return(character(0))
  nm <- names(stats::coef(model))
  keep <- rep(TRUE, length(nm))
  for (s in required_substrings) {
    keep <- keep & grepl(s, nm, fixed = TRUE)
  }
  nm[keep]
}

choose_one_in_window <- function(data, id_var, time_var, lower, upper, target) {
  data %>%
    dplyr::filter(
      .data[[time_var]] >= lower,
      .data[[time_var]] <= upper
    ) %>%
    dplyr::mutate(.distance_to_target = abs(.data[[time_var]] - target)) %>%
    dplyr::group_by(.data[[id_var]]) %>%
    dplyr::arrange(.distance_to_target, .data[[time_var]], .by_group = TRUE) %>%
    dplyr::slice_head(n = 1L) %>%
    dplyr::ungroup() %>%
    dplyr::select(-.distance_to_target)
}


empty_match_index <- function() {
  tibble::tibble(
    comparison = character(),
    match_set_id = character(),
    unit_pidp = character(),
    unit_episode_id = character(),
    condition = character(),
    treated = numeric(),
    weight = numeric(),
    n_controls = integer()
  )
}

# Robust summary used for both LAT and cohabitation matching.
# It explicitly returns a zero-match row rather than evaluating min()/max()
# on an empty treated subset. This prevents Inf/-Inf diagnostic values and
# makes failed matching immediately visible in the exported summary.
summarise_match_index <- function(match_index, comparison_label) {
  if (nrow(match_index) == 0L) {
    return(tibble::tibble(
      comparison = comparison_label,
      matched_sets = 0L,
      treated_people = 0L,
      control_people = 0L,
      median_controls_per_set = NA_real_,
      min_controls_per_set = NA_integer_,
      max_controls_per_set = NA_integer_,
      match_status = "NO_MATCHED_SETS"
    ))
  }

  treated_rows <- match_index %>%
    dplyr::filter(condition == "treated")
  control_rows <- match_index %>%
    dplyr::filter(condition == "control")

  if (nrow(treated_rows) == 0L) {
    return(tibble::tibble(
      comparison = comparison_label,
      matched_sets = 0L,
      treated_people = 0L,
      control_people = dplyr::n_distinct(control_rows$unit_pidp),
      median_controls_per_set = NA_real_,
      min_controls_per_set = NA_integer_,
      max_controls_per_set = NA_integer_,
      match_status = "INVALID_NO_TREATED_ROWS"
    ))
  }

  controls_per_set <- treated_rows$n_controls
  controls_per_set <- controls_per_set[!is.na(controls_per_set)]

  tibble::tibble(
    comparison = comparison_label,
    matched_sets = dplyr::n_distinct(treated_rows$match_set_id),
    treated_people = dplyr::n_distinct(treated_rows$unit_pidp),
    control_people = dplyr::n_distinct(control_rows$unit_pidp),
    median_controls_per_set = if (length(controls_per_set) > 0L)
      stats::median(controls_per_set) else NA_real_,
    min_controls_per_set = if (length(controls_per_set) > 0L)
      min(controls_per_set) else NA_integer_,
    max_controls_per_set = if (length(controls_per_set) > 0L)
      max(controls_per_set) else NA_integer_,
    match_status = if (nrow(control_rows) == 0L)
      "INVALID_NO_CONTROL_ROWS" else "MATCHED"
  )
}

match_index_condition_audit <- function(match_index, comparison_label) {
  if (nrow(match_index) == 0L) {
    return(tibble::tibble(
      comparison = comparison_label,
      condition = "NO_ROWS",
      rows = 0L,
      people = 0L,
      matched_sets = 0L
    ))
  }

  match_index %>%
    dplyr::count(condition, name = "rows") %>%
    dplyr::left_join(
      match_index %>%
        dplyr::group_by(condition) %>%
        dplyr::summarise(
          people = dplyr::n_distinct(unit_pidp),
          matched_sets = dplyr::n_distinct(match_set_id),
          .groups = "drop"
        ),
      by = "condition"
    ) %>%
    dplyr::mutate(comparison = comparison_label, .before = 1)
}

build_simple_match_index <- function(
    event_data,
    treated_path,
    control_path,
    comparison_label,
    onset_type = c("LAT", "COHAB")
) {
  onset_type <- match.arg(onset_type)

  treated <- event_data %>%
    dplyr::filter(path_group == treated_path) %>%
    dplyr::arrange(pidp, onset_ym)

  controls <- event_data %>%
    dplyr::filter(path_group == control_path)

  if (nrow(treated) == 0L || nrow(controls) == 0L) return(empty_match_index())

  tr <- treated %>%
    dplyr::rename_with(~ paste0("tr_", .x)) %>%
    dplyr::mutate(match_set_id = paste0(comparison_label, "_", dplyr::row_number()))

  ct <- controls %>% dplyr::rename_with(~ paste0("ct_", .x))

  if (onset_type == "LAT") {
    candidates <- tr %>%
      dplyr::inner_join(
        ct,
        by = c(
          "tr_sex" = "ct_sex",
          "tr_positive_wave" = "ct_positive_wave"
        ),
        relationship = "many-to-many"
      ) %>%
      dplyr::filter(
        abs(tr_age - ct_age) <= MATCH_AGE_CALIPER,
        abs(tr_baseline_month - ct_baseline_month) <= MATCH_BASELINE_MONTH_CALIPER,
        abs(tr_baseline_to_positive_months - ct_baseline_to_positive_months) <=
          MATCH_FOLLOWUP_INTERVAL_CALIPER
      )
  } else {
    candidates <- tr %>%
      dplyr::inner_join(
        ct,
        by = c(
          "tr_sex" = "ct_sex",
          "tr_baseline_wave" = "ct_baseline_wave"
        ),
        relationship = "many-to-many"
      ) %>%
      dplyr::filter(
        abs(tr_age - ct_age) <= MATCH_AGE_CALIPER,
        abs(tr_onset_ym - ct_onset_ym) <= MATCH_COHAB_START_CALIPER
      )
  }

  retained <- candidates %>%
    dplyr::count(match_set_id, name = "n_controls") %>%
    dplyr::filter(n_controls >= MIN_CONTROLS_PER_SET)

  candidates <- candidates %>%
    dplyr::inner_join(retained, by = "match_set_id")

  if (nrow(candidates) == 0L) return(empty_match_index())

  treated_index <- candidates %>%
    dplyr::distinct(
      match_set_id,
      tr_pidp,
      tr_episode_id,
      tr_onset_ym,
      n_controls
    ) %>%
    dplyr::transmute(
      comparison = comparison_label,
      match_set_id,
      unit_pidp = tr_pidp,
      unit_episode_id = tr_episode_id,
      condition = "treated",
      treated = 1,
      weight = 1,
      n_controls
    )

  control_index <- candidates %>%
    dplyr::distinct(
      match_set_id,
      ct_pidp,
      ct_episode_id,
      ct_onset_ym,
      n_controls
    ) %>%
    dplyr::transmute(
      comparison = comparison_label,
      match_set_id,
      unit_pidp = ct_pidp,
      unit_episode_id = ct_episode_id,
      condition = "control",
      treated = 0,
      weight = 1 / n_controls,
      n_controls
    )

  dplyr::bind_rows(treated_index, control_index)
}

make_matched_stage_stack <- function(match_index, stage_data) {
  if (nrow(match_index) == 0L) return(tibble::tibble())

  match_index %>%
    dplyr::inner_join(
      stage_data,
      by = c(
        "unit_pidp" = "pidp",
        "unit_episode_id" = "episode_id"
      ),
      relationship = "many-to-many"
    ) %>%
    dplyr::mutate(
      match_person_fe = paste(match_set_id, unit_pidp, sep = "__P__"),
      match_stage_fe = paste(match_set_id, stage, sep = "__S__")
    )
}

onset_balance_audit <- function(match_index, event_data, variables, label) {
  if (nrow(match_index) == 0L) {
    return(tibble::tibble(
      comparison = label, variable = variables,
      mean_abs_set_difference = NA_real_,
      median_abs_set_difference = NA_real_,
      max_abs_set_difference = NA_real_,
      matched_sets = 0L
    ))
  }

  dat <- match_index %>%
    dplyr::inner_join(
      event_data,
      by = c(
        "unit_pidp" = "pidp",
        "unit_episode_id" = "episode_id"
      ),
      relationship = "many-to-one"
    )

  purrr::map_dfr(variables, function(v) {
    sm <- dat %>%
      dplyr::filter(!is.na(.data[[v]])) %>%
      dplyr::group_by(match_set_id, condition) %>%
      dplyr::summarise(
        m = stats::weighted.mean(.data[[v]], w = weight, na.rm = TRUE),
        .groups = "drop"
      ) %>%
      tidyr::pivot_wider(names_from = condition, values_from = m) %>%
      dplyr::filter(!is.na(treated), !is.na(control)) %>%
      dplyr::mutate(abs_difference = abs(treated - control))

    tibble::tibble(
      comparison = label,
      variable = v,
      mean_abs_set_difference = if (nrow(sm) == 0L) NA_real_ else mean(sm$abs_difference),
      median_abs_set_difference = if (nrow(sm) == 0L) NA_real_ else stats::median(sm$abs_difference),
      max_abs_set_difference = if (nrow(sm) == 0L) NA_real_ else max(sm$abs_difference),
      matched_sets = nrow(sm)
    )
  })
}

# Matched event-study model. Matched-set x person fixed effects absorb stable
# unit differences; matched-set x stage fixed effects absorb common stage shocks.
fit_matched_trajectory_direct <- function(
    stack,
    outcome_var,
    stage_reference,
    model_name
) {
  if (nrow(stack) == 0L || !outcome_var %in% names(stack)) return(NULL)

  dat <- stack %>%
    dplyr::filter(!is.na(.data[[outcome_var]])) %>%
    dplyr::mutate(stage = stats::relevel(factor(as.character(stage)), ref = stage_reference))

  if (nrow(dat) == 0L) return(NULL)

  fml <- stats::as.formula(
    paste0(
      outcome_var,
      " ~ i(stage, treated, ref = '", stage_reference,
      "') | match_person_fe + match_stage_fe"
    )
  )

  safe_feols(
    fml,
    data = dat,
    name = model_name,
    weights = ~weight,
    vcov = ~unit_pidp
  )
}

fit_balanced_target <- function(
    match_index,
    stage_data,
    target_stage,
    outcome_var,
    outcome_label,
    comparison_label,
    model_stub,
    reference_stage
) {
  two_stage <- stage_data %>%
    dplyr::filter(stage %in% c(reference_stage, target_stage)) %>%
    dplyr::filter(!is.na(.data[[outcome_var]]))

  tmp <- match_index %>%
    dplyr::inner_join(
      two_stage,
      by = c(
        "unit_pidp" = "pidp",
        "unit_episode_id" = "episode_id"
      ),
      relationship = "many-to-many"
    )

  if (nrow(tmp) == 0L) {
    empty_result <- extract_term(
      NULL, NA_character_, comparison_label, outcome_label,
      paste0(target_stage, " vs ", reference_stage, " differential change")
    ) %>%
      dplyr::mutate(
        target_stage = target_stage,
        matched_sets = 0L,
        treated_people = 0L,
        control_people = 0L
      )
    return(list(result = empty_result, model = NULL, sample = tibble::tibble()))
  }

  availability <- tmp %>%
    dplyr::distinct(match_set_id, unit_pidp, condition, stage) %>%
    dplyr::count(match_set_id, unit_pidp, condition, name = "n_stage") %>%
    dplyr::filter(n_stage == 2L)

  tmp <- tmp %>%
    dplyr::inner_join(
      availability %>% dplyr::select(match_set_id, unit_pidp, condition),
      by = c("match_set_id", "unit_pidp", "condition")
    )

  # Recalculate the number of outcome-complete controls within each balanced
  # two-stage matched set. Use distinct names because match_index already carries
  # an upstream n_controls column; joining another column with the same name would
  # create n_controls.x/n_controls.y and leave no plain n_controls variable.
  set_ok <- tmp %>%
    dplyr::distinct(match_set_id, unit_pidp, condition) %>%
    dplyr::group_by(match_set_id) %>%
    dplyr::summarise(
      n_treated_balanced = sum(condition == "treated"),
      n_controls_balanced = sum(condition == "control"),
      .groups = "drop"
    ) %>%
    dplyr::filter(
      n_treated_balanced == 1L,
      n_controls_balanced >= 1L
    )

  tmp <- tmp %>%
    dplyr::inner_join(set_ok, by = "match_set_id") %>%
    dplyr::mutate(
      weight_balanced = dplyr::if_else(
        condition == "treated", 1, 1 / n_controls_balanced
      ),
      match_person_fe = paste(match_set_id, unit_pidp, sep = "__P__"),
      match_stage_fe = paste(match_set_id, stage, sep = "__S__"),
      stage = stats::relevel(factor(as.character(stage)), ref = reference_stage)
    )

  if (nrow(tmp) == 0L) {
    empty_result <- extract_term(
      NULL, NA_character_, comparison_label, outcome_label,
      paste0(target_stage, " vs ", reference_stage, " differential change")
    ) %>%
      dplyr::mutate(
        target_stage = target_stage,
        matched_sets = 0L,
        treated_people = 0L,
        control_people = 0L
      )
    return(list(result = empty_result, model = NULL, sample = tmp))
  }

  fml <- stats::as.formula(
    paste0(
      outcome_var,
      " ~ i(stage, treated, ref = '", reference_stage,
      "') | match_person_fe + match_stage_fe"
    )
  )

  model <- safe_feols(
    fml,
    data = tmp,
    name = paste0(model_stub, "_", target_stage, "_", outcome_label),
    weights = ~weight_balanced,
    vcov = ~unit_pidp
  )

  result <- extract_first_matching_term(
    model,
    required_substrings = c(target_stage),
    analysis = comparison_label,
    outcome = outcome_label,
    contrast = paste0(target_stage, " vs ", reference_stage, " differential change")
  ) %>%
    dplyr::mutate(
      target_stage = target_stage,
      matched_sets = dplyr::n_distinct(tmp$match_set_id),
      treated_people = dplyr::n_distinct(tmp$unit_pidp[tmp$condition == "treated"]),
      control_people = dplyr::n_distinct(tmp$unit_pidp[tmp$condition == "control"])
    )

  list(result = result, model = model, sample = tmp)
}

# UKHLS pidp is sometimes stored as a labelled/numeric value and sometimes as
# character in saved analytic objects. All person identifiers are normalized to
# plain character strings before any joins. For numeric IDs, format(...,
# scientific = FALSE) avoids scientific notation and preserves the integer-like
# identifier representation used in the incident-LAT objects.
id_chr <- function(x) {
  if (is.factor(x)) x <- as.character(x)
  if (inherits(x, "haven_labelled")) x <- unclass(x)
  if (is.numeric(x) || is.integer(x)) {
    return(format(x, scientific = FALSE, trim = TRUE, digits = 22))
  }
  as.character(x)
}

# Normalize categorical codes that are stored as numeric in some upstream RDS
# objects and as character in others. In the validated v4.1 panel, sex is coded
# numerically (1/2), whereas the incident-LAT workflow converts it to character.
# Unclass haven-labelled vectors before conversion so the underlying 1/2 codes are
# preserved rather than mixing value labels with numeric codes.
code_chr <- function(x) {
  if (is.factor(x)) return(as.character(x))
  if (inherits(x, "haven_labelled")) x <- unclass(x)
  if (is.numeric(x) || is.integer(x)) {
    return(format(x, scientific = FALSE, trim = TRUE, digits = 22))
  }
  as.character(x)
}

# IMPORTANT TIME-ORIGIN HARMONIZATION
# ---------------------------------
# The incident-LAT workflow stores month indices in Stata monthly-date units:
#   (year - 1960) * 12 + (month - 1).
# The validated v4.1 objects use:
#   year * 12 + month - 1.
# Therefore incident month indices must be shifted by exactly 1960 * 12 = 23520
# before they are compared or combined with v4.1 interview_ym/start_ym variables.
# Differences measured wholly within the incident workflow are unchanged by this
# additive shift. The shift is applied once here to every incident absolute-month
# field used downstream.
V41_YM_OFFSET_FROM_STATA_MONTH <- 1960L * 12L

    prev_valid_lat_start = dplyr::lag(lat_relationship_start_ym),
    valid_gap = interview_ym - prev_valid_ym,
    start_date_shift = abs(lat_relationship_start_ym - prev_valid_lat_start),
    new_lat_episode = ncrr1 == 1 & (
      is.na(prev_valid_ncrr1) |
        prev_valid_ncrr1 != 1 |
        is.na(valid_gap) |
        valid_gap > MAX_GAP_WITHIN_LAT_EPISODE_MONTHS |
        (
          !is.na(lat_relationship_start_ym) &
            !is.na(prev_valid_lat_start) &
            start_date_shift > MAX_LAT_START_DATE_SHIFT_MONTHS
        )
    ),
    lat_episode_number = cumsum(dplyr::coalesce(new_lat_episode, FALSE)),
    lat_episode_id = dplyr::if_else(
      ncrr1 == 1,
      paste0(pidp, "_LAT_", lat_episode_number),
      NA_character_

  dplyr::ungroup() %>%
  dplyr::select(
    pidp, interview_ym, lat_episode_id, new_lat_episode, start_date_shift,
    ncrr1
  )

  dplyr::left_join(
    next_valid_after_episode %>%
      dplyr::select(
        lat_episode_id, next_valid_ym, next_valid_ncrr1,
        next_valid_wave
      ),
    by = "lat_episode_id",
    relationship = "one-to-one"
  ) %>%
  dplyr::mutate(
    months_LATstart_to_cohab = linked_cohab_start_ym - episode_reference_start_ym,
    months_LATstart_to_exit = next_valid_ym - episode_reference_start_ym,
    observed_LAT_duration_from_start =
      last_lat_interview_ym - episode_reference_start_ym,

    cohab_within_36 =
      !is.na(months_LATstart_to_cohab) &
      months_LATstart_to_cohab > 0 &
      months_LATstart_to_cohab <= PATHWAY_HORIZON_MONTHS,

    observed_exit_within_36 =
      !is.na(months_LATstart_to_exit) &
      months_LATstart_to_exit > 0 &
      months_LATstart_to_exit <= PATHWAY_HORIZON_MONTHS &
      next_valid_ncrr1 == 2,

    observed_stable_LAT_near_36 = has_LAT_obs_near_36,

    path_group = dplyr::case_when(
      cohab_within_36 ~ "future_cohab_36",
      !cohab_within_36 & observed_exit_within_36 ~ "future_breakup_36",
      !cohab_within_36 & !observed_exit_within_36 &
        observed_stable_LAT_near_36 ~ "stable_LAT_36",
      TRUE ~ "other_or_censored"
    )
  )

write_csv_quiet(

    dplyr::mutate(percent = 100 * LAT_episodes / sum(LAT_episodes)),

lat_subjective <- lat_rows %>%
  dplyr::arrange(lat_episode_id, interview_ym) %>%
  dplyr::group_by(lat_episode_id) %>%
  dplyr::mutate(
    prev_interview_ym = dplyr::lag(interview_ym),
    next_interview_ym = dplyr::lead(interview_ym),
    prev_gap = interview_ym - prev_interview_ym,
    next_gap = next_interview_ym - interview_ym,

    prev_ncrr11 = dplyr::lag(ncrr11_intend_yes),
    next_ncrr11 = dplyr::lead(ncrr11_intend_yes),
    delta_ncrr11 = dplyr::if_else(
      !is.na(ncrr11_intend_yes) & !is.na(prev_ncrr11) &
        prev_gap >= PSYCH_MIN_GAP_MONTHS & prev_gap <= PSYCH_MAX_GAP_MONTHS,
      ncrr11_intend_yes - prev_ncrr11,
      NA_real_
    ),

    next_ls = dplyr::lead(life_satisfaction),
    next_ghqwb = dplyr::lead(ghq_wb),
    next_ls_change_z = dplyr::if_else(
      !is.na(life_satisfaction) & !is.na(next_ls) &
        next_gap >= PSYCH_MIN_GAP_MONTHS & next_gap <= PSYCH_MAX_GAP_MONTHS,
      (next_ls - life_satisfaction) / lat_ls_sd,
      NA_real_
    ),
    next_ghqwb_change_z = dplyr::if_else(
      !is.na(ghq_wb) & !is.na(next_ghqwb) &
        next_gap >= PSYCH_MIN_GAP_MONTHS & next_gap <= PSYCH_MAX_GAP_MONTHS,
      (next_ghqwb - ghq_wb) / lat_ghqwb_sd,
      NA_real_
    ),

    ncrr11_switch = dplyr::case_when(
      prev_ncrr11 == 0 & ncrr11_intend_yes == 1 ~ "no_to_yes",
      prev_ncrr11 == 1 & ncrr11_intend_yes == 0 ~ "yes_to_no",
      prev_ncrr11 == 0 & ncrr11_intend_yes == 0 ~ "stable_no",
      prev_ncrr11 == 1 & ncrr11_intend_yes == 1 ~ "stable_yes",
      TRUE ~ NA_character_
    )
  ) %>%
  dplyr::ungroup()

# Exploratory construct-validity sensitivity among LAT episodes conservatively
# linked to a subsequent cohabitation spell. Outcome is whether that linked
# cohabitation spell is observed to progress to marriage. This is NOT a complete
# same-partner marriage construct-validity test for all LAT relationships.
C_construct_rows <- ncrr12_rows %>%
  dplyr::inner_join(
    lat_pathways %>%
      dplyr::filter(!is.na(linked_cohab_spell_id)) %>%
      dplyr::select(
        lat_episode_id,
        linked_cohab_spell_outcome,
        linked_marriage_start_ym
      ),
    by = "lat_episode_id",
    relationship = "many-to-one"
  ) %>%
  dplyr::mutate(
    linked_cohab_later_married = dplyr::case_when(
      linked_cohab_spell_outcome == "marriage" ~ 1,
      linked_cohab_spell_outcome %in% c("breakup", "ongoing") ~ 0,
      TRUE ~ NA_real_
    )
  )

C_construct_model <- safe_feglm(
  linked_cohab_later_married ~
    ncrr12_score_z + splines::ns(lat_duration_months, df = 3) +
    age + i(sex) | wave,
  data = C_construct_rows %>%
    dplyr::filter(!is.na(linked_cohab_later_married)),
  family = "binomial",
  name = "C_NCRR12_level_predicts_marriage_via_linked_cohab",
  vcov = ~pidp
)

C_construct_result <- extract_term(
  C_construct_model,
  "ncrr12_score_z",
  "NCRR12 construct-validity sensitivity",
  "Observed marriage after probable LAT -> cohabitation linkage",
  "1-SD higher LAT marriage expectation"
) %>%
  dplyr::mutate(
    odds_ratio = exp(estimate),
    or_conf_low = exp(conf_low),
    or_conf_high = exp(conf_high),
    caution = paste(
      "Exploratory: restricted to LAT episodes conservatively linked to",
      "cohabitation; does not capture all direct or unlinked marriages."
    )
  )
write_csv_quiet(C_construct_result, "C04_NCRR12_construct_validity_sensitivity.csv")
