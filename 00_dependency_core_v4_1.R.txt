# =============================================================================
# UKHLS REVISED FORMAL ANALYSIS WORKFLOW - VERSION 4.1
# Selection, relationship development, marriage proximity, and breakup proximity
# =============================================================================
#
# Identification and inferential sequence:
# A. Pre-relationship psychological differentiation/selection
# B. Distal 25-60m pre-proximity diagnostic
# C. Relationship development + lagged coupled-change analyses
#    + non-binary C-to-D distal diagnostic
# D_M. Marriage-proximity ATT risk-set analyses of RECENT CHANGE
# D_B. Breakup-proximity ATT risk-set analyses of RECENT CHANGE
# E. Sequential explanatory attenuation on identical common samples
#    + inverse-probability-of-history-availability weighting (IPHAW)
# F. Lagged NCRR12 expectation triangulation in LAT relationships
# G. Multiple-relationship within-person robustness
#
# CONFIRMATORY FAMILY:
#   D_M, 1-12 months before marriage, D1b estimator, GHQ and life satisfaction.
#   Two-sided p-values are Holm-adjusted across these two co-primary outcomes.
#
# SECONDARY MULTIPLICITY FAMILIES:
#   Marriage timing: D1b at 13-24m and 25-60m x GHQ/LS (Holm across 4).
#   Breakup timing: D1b at 1-12m, 13-24m, 25-60m x GHQ/LS (Holm across 6).
#
# D1b is the fixed confirmatory estimator because its estimand is the observed
# history-complete eligible population and it does not require modeling the
# missing-history process. D1c is a MANDATORY missing-history sensitivity that
# reweights the same observed-history cases toward the eligible risk-set
# population in which the survey/interview schedule provided STRUCTURAL
# OPPORTUNITY for the required prior psychological history to be observed.
# D1c cannot create/rescue confirmatory support, but a stable and materially
# discordant D1c result QUALIFIES a supported D1b result as sensitive to
# selective psychological-history availability.
#
# D0/D1a/D2/D3 and A/B/C/E/F/G are explanatory, diagnostic, robustness, or
# triangulation analyses rather than additional confirmatory tests.
#
# No analysis is interpreted as a causal "anticipation effect" or as a
# causal mediation/decomposition. E is descriptive explanatory attenuation.
# =============================================================================

num_clean <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  x[x < 0] <- NA_real_
  x
}

ym_index <- function(year, month) {
  year <- num_clean(year)
  month <- num_clean(month)
  out <- rep(NA_real_, length(year))
  ok <- !is.na(year) & !is.na(month) &
    month >= 1 & month <= 12 & year >= 1900 & year <= 2100
  out[ok] <- year[ok] * 12 + month[ok] - 1
  out
}

zsafe <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  s <- sd(x, na.rm = TRUE)
  if (is.na(s) || s == 0) return(rep(NA_real_, length(x)))
  (x - mean(x, na.rm = TRUE)) / s
}

last_nonmissing_lag <- function(x) {
  x <- suppressWarnings(as.numeric(x))
  out <- rep(NA_real_, length(x))
  last <- NA_real_
  for (i in seq_along(x)) {
    out[i] <- last
    if (!is.na(x[i])) last <- x[i]
  }
  out
}

  x <- suppressWarnings(as.numeric(x))

  out <- rep(NA_real_, length(x))

  for (i in seq_along(x)) {
    out[i] <- last_time
    if (!is.na(x[i]) && !is.na(time[i])) last_time <- time[i]
  }

last_nonmissing_change_lag <- function(change, time) {
  change <- suppressWarnings(as.numeric(change))
  time <- suppressWarnings(as.numeric(time))
  out_change <- rep(NA_real_, length(change))
  out_time <- rep(NA_real_, length(change))
  last_change <- NA_real_
  last_time <- NA_real_
  for (i in seq_along(change)) {
    out_change[i] <- last_change
    out_time[i] <- last_time
    if (!is.na(change[i]) && !is.na(time[i])) {
      last_change <- change[i]
      last_time <- time[i]
    }
  }
  list(change = out_change, time = out_time)
}

save_fixest_model <- function(model, name) {
  saveRDS(model, file.path(OUT_DIR, "models", paste0(name, ".rds")))
  cf <- coef(model)
  se_ <- se(model)
  p_ <- pvalue(model)
  ci_ <- confint(model)
  out <- tibble(
    term = names(cf),
    estimate = as.numeric(cf),
    std_error = as.numeric(se_),
    conf_low = as.numeric(ci_[, 1]),
    conf_high = as.numeric(ci_[, 2]),
    p_value = as.numeric(p_)
  )
  write_csv_quiet(out, paste0(name, "_coefficients.csv"))
  write_txt(capture.output(summary(model)), paste0(name, "_summary.txt"))
  invisible(model)
}

weighted_mean <- function(x, w) {
  ok <- !is.na(x) & !is.na(w) & w > 0
  if (!any(ok)) return(NA_real_)
  sum(x[ok] * w[ok]) / sum(w[ok])
}

weighted_var <- function(x, w) {
  ok <- !is.na(x) & !is.na(w) & w > 0
  if (sum(ok) < 2L) return(NA_real_)
  m <- weighted_mean(x[ok], w[ok])
  sum(w[ok] * (x[ok] - m)^2) / sum(w[ok])
}

smd_two_group <- function(x, treated, w = NULL) {
  if (is.null(w)) w <- rep(1, length(x))
  mt <- weighted_mean(x[treated == 1], w[treated == 1])
  mc <- weighted_mean(x[treated == 0], w[treated == 0])
  vt <- weighted_var(x[treated == 1], w[treated == 1])
  vc <- weighted_var(x[treated == 0], w[treated == 0])
  denom <- sqrt((vt + vc) / 2)
  if (is.na(denom) || denom == 0) return(NA_real_)
  (mt - mc) / denom
}


effective_sample_size <- function(w) {
  w <- w[is.finite(w) & !is.na(w) & w > 0]
  if (length(w) == 0L) return(NA_real_)
  (sum(w)^2) / sum(w^2)
}

# Re-trim a FINAL combined weight separately within treated/control status.
# This is applied after ATT x history-weight multiplication.
trim_weight_by_treatment <- function(
  w,
  treated,
  lower = FINAL_WEIGHT_TRIM_LOWER,
  upper = FINAL_WEIGHT_TRIM_UPPER
) {
  out <- rep(NA_real_, length(w))

  groups <- sort(unique(treated[!is.na(treated)]))

  for (g in groups) {
    idx <- treated == g & is.finite(w) & !is.na(w) & w > 0

    if (!any(idx)) next

    vals <- w[idx]

    if (length(vals) > 1L) {
      q <- stats::quantile(
        vals,
        probs = c(lower, upper),
        na.rm = TRUE,
        names = FALSE
      )
      out[idx] <- pmin(pmax(vals, q[1]), q[2])
    } else {
      out[idx] <- vals
    }
  }

  out
}

make_final_weight_variants <- function(att_weight, history_weight, treated) {
  raw <- att_weight * history_weight

  list(
    raw = raw,
    trim_1_99 = trim_weight_by_treatment(
      raw, treated,
      FINAL_WEIGHT_TRIM_LOWER,
      FINAL_WEIGHT_TRIM_UPPER
    ),
    trim_2p5_97p5 = trim_weight_by_treatment(
      raw, treated,
      FINAL_WEIGHT_TIGHT_LOWER,
      FINAL_WEIGHT_TIGHT_UPPER
    )
  )
}

weight_stability_diagnostics <- function(data, weight_var, stage) {
  w <- data[[weight_var]]

  one_row <- function(idx, group_label) {
    idx <- idx & is.finite(w) & !is.na(w) & w > 0
    n <- sum(idx)
    ess <- if (n == 0L) NA_real_ else effective_sample_size(w[idx])
    ratio <- if (n == 0L || is.na(ess)) NA_real_ else ess / n

    tibble(
      stage = stage,
      group = group_label,
      n_nominal = n,
      ess = ess,
      ess_ratio = ratio,
      mean_weight = if (n == 0L) NA_real_ else mean(w[idx]),
      p99_weight = if (n == 0L) NA_real_ else
        as.numeric(stats::quantile(w[idx], .99, na.rm = TRUE)),
      max_weight = if (n == 0L) NA_real_ else max(w[idx])
    )
  }

  diag <- bind_rows(
    one_row(rep(TRUE, nrow(data)), "overall"),
    one_row(data$treated == 1, "treated"),
    one_row(data$treated == 0, "control")
  )

  valid_ratios <- diag$ess_ratio[!is.na(diag$ess_ratio)]

  stage_status <- case_when(
    length(valid_ratios) == 0L ~ "ESS unavailable",
    any(valid_ratios < ESS_SEVERE_RATIO) ~
      "severely unstable: report only; do not use as robustness evidence",
    any(valid_ratios < ESS_ACCEPTABLE_RATIO) ~
      "caution: report only; do not use as affirmative robustness evidence",
    TRUE ~ "acceptable ESS for robustness interpretation"
  )

  diag %>%
    mutate(
      stage_status = stage_status,
      acceptable_for_robustness =
        stage_status == "acceptable ESS for robustness interpretation"
    )
}

smd_to_target <- function(target_x, selected_x, selected_w) {
  target_x <- suppressWarnings(as.numeric(target_x))
  selected_x <- suppressWarnings(as.numeric(selected_x))
  target_sd <- stats::sd(target_x, na.rm = TRUE)
  if (is.na(target_sd) || target_sd == 0) return(NA_real_)
  (weighted_mean(selected_x, selected_w) -
     mean(target_x, na.rm = TRUE)) / target_sd
}

extract_term_stats <- function(model, term = "treated") {
  if (is.null(model)) {
    return(tibble(
      term = term,
      estimate = NA_real_,
      std_error = NA_real_,
      conf_low = NA_real_,
      conf_high = NA_real_,
      p_value = NA_real_,
      nobs = NA_integer_
    ))
  }

  cf <- coef(model)
  if (!term %in% names(cf)) {
    return(tibble(
      term = term,
      estimate = NA_real_,
      std_error = NA_real_,
      conf_low = NA_real_,
      conf_high = NA_real_,
      p_value = NA_real_,
      nobs = as.integer(stats::nobs(model))
    ))
  }

  ci <- tryCatch(
    confint(model),
    error = function(e) NULL
  )

  if (is.null(ci) || is.null(rownames(ci)) || !term %in% rownames(ci)) {
    ci_low <- NA_real_
    ci_high <- NA_real_
  } else {
    ci_low <- as.numeric(ci[term, 1])
    ci_high <- as.numeric(ci[term, 2])
  }

  tibble(
    term = term,
    estimate = unname(cf[term]),
    std_error = unname(se(model)[term]),
    conf_low = ci_low,
    conf_high = ci_high,
    p_value = unname(pvalue(model)[term]),
    nobs = as.integer(stats::nobs(model))
  )
}

attenuation_row <- function(
  base_model,
  adjusted_model,
  arm,
  horizon,
  outcome,
  stage,
  common_sample
) {
  b <- extract_term_stats(base_model, "treated")
  a <- extract_term_stats(adjusted_model, "treated")

  interpretable <-
    !is.na(b$estimate) &
    !is.na(a$estimate) &
    abs(b$estimate) > 1e-8 &
    sign(b$estimate) == sign(a$estimate)

  attenuation_pct <- if (interpretable) {
    100 * (1 - a$estimate / b$estimate)
  } else {
    NA_real_
  }

  tibble(
    arm = arm,
    horizon = horizon,
    outcome = outcome,
    stage = stage,
    common_sample = common_sample,
    base_estimate = b$estimate,
    adjusted_estimate = a$estimate,
    attenuation_pct = attenuation_pct,
    base_n = b$nobs,
    adjusted_n = a$nobs,
    interpretation_flag = if_else(
      interpretable,
      "descriptive attenuation estimable",
      "do not interpret percentage: zero/near-zero or sign change"
    )
  )
}

model_comparison_row <- function(

  )
}

safe_wald_row <- function(model, keep, test_name, outcome, arm = "marriage") {
  if (is.null(model)) {
    return(tibble(
      test = test_name,
      arm = arm,
      outcome = outcome,
      stat = NA_real_,
      p_value = NA_real_,
      df1 = NA_real_,
      df2 = NA_real_,
      status = "model_missing"
    ))
  }

  wt <- tryCatch(
    fixest::wald(model, keep = keep, print = FALSE),
    error = function(e) e
  )

  if (inherits(wt, "error") || length(wt) == 1L && is.na(wt)) {
    return(tibble(
      test = test_name,
      arm = arm,
      outcome = outcome,
      stat = NA_real_,
      p_value = NA_real_,
      df1 = NA_real_,
      df2 = NA_real_,
      status = "wald_unavailable"
    ))
  }

  tibble(
    test = test_name,
    arm = arm,
    outcome = outcome,
    stat = unname(wt["stat"]),
    p_value = unname(wt["p"]),
    df1 = unname(wt["df1"]),
    df2 = unname(wt["df2"]),
    status = "estimated"
  )
}

# Structural-opportunity helpers ------------------------------------------------

exists_prior_module_in_window <- function(
  times,
  module,
  j,
  min_gap,
  max_gap
) {
  if (j <= 1L) return(FALSE)
  idx <- seq_len(j - 1L)
  gaps <- times[j] - times[idx]
  any(
    module[idx] &
      !is.na(gaps) &
      gaps >= min_gap &
      gaps <= max_gap,
    na.rm = TRUE
  )
}

compute_change_opportunity <- function(
  times,
  module,
  min_gap,
  max_gap
) {
  n <- length(times)
  out <- rep(FALSE, n)
  if (n <= 1L) return(out)

  for (k in 2:n) {
    if (!isTRUE(module[k])) next
    idx <- seq_len(k - 1L)
    gaps <- times[k] - times[idx]
    out[k] <- any(
      module[idx] &
        !is.na(gaps) &
        gaps >= min_gap &
        gaps <= max_gap,
      na.rm = TRUE
    )
  }

  out
}

exists_prior_change_opp <- function(
  times,
  change_opp,
  j,
  max_gap
) {
  if (j <= 1L) return(FALSE)
  idx <- seq_len(j - 1L)
  gaps <- times[j] - times[idx]
  any(
    change_opp[idx] &
      !is.na(gaps) &
      gaps > 0 &
      gaps <= max_gap,
    na.rm = TRUE
  )
}

source_row_ambiguous <- function(
  source_time,
  times,
  ambiguous
) {
  if (is.na(source_time)) return(NA)
  hit <- which(!is.na(times) & times == source_time)
  if (length(hit) == 0L) return(NA)
  any(ambiguous[hit], na.rm = TRUE)
}

# Bias-reduced availability model ---------------------------------------------
# All candidate specifications use brglm2. There is no ordinary-GLM fallback.
# Formula reduction is only a numerical fallback if the richer bias-reduced
# specification cannot be fitted (e.g., a one-level factor after restriction).

fit_bias_reduced_availability <- function(dat) {

  formulas <- list(
    full = .H ~
      treated +
      factor(sex) +
      wave_n +
      age +
      cohab_duration_months +
      prior_outcome +
      log1p(n_prior_interviews) +
      log1p(n_prior_outcome_obs) +
      log1p(panel_tenure_months),

    reduced = .H ~
      treated +
      wave_n +
      age +
      cohab_duration_months +
      prior_outcome +
      log1p(n_prior_interviews) +
      log1p(panel_tenure_months),

    minimal = .H ~
      treated +
      age +
      cohab_duration_months +
      prior_outcome
  )

  for (nm in names(formulas)) {
    fit <- tryCatch(
      suppressWarnings(
        stats::glm(
          formulas[[nm]],
          data = dat,
          family = stats::binomial("logit"),
          method = brglm2::brglmFit,
          type = "AS_mean",
          control = stats::glm.control(maxit = 100)
        )
      ),
      error = function(e) NULL
    )

    if (!is.null(fit)) {
      p <- tryCatch(
        stats::predict(fit, type = "response"),
        error = function(e) rep(NA_real_, nrow(dat))
      )

      if (
        length(p) == nrow(dat) &&
          any(is.finite(p), na.rm = TRUE)
      ) {
        return(list(
          model = fit,
          formula_tier = nm
        ))
      }
    }
  }

  list(
    model = NULL,
    formula_tier = "failed"
  )
}

estimate_ipha_stage <- function(
  data,
  availability_var,
  opportunity_var,
  stage_name,
  enforce_availability_subset = TRUE
) {

  source <- data %>%
    mutate(
      .H = as.integer(.data[[availability_var]]),
      .S = as.integer(.data[[opportunity_var]])
    )

  n_mismatch <- sum(
    source$.H == 1 & source$.S != 1,
    na.rm = TRUE
  )

  if (isTRUE(enforce_availability_subset) && n_mismatch > 0L) {
    stop(
      "Structural-opportunity mismatch at ",
      stage_name,
      ": ",
      n_mismatch,
      " available histories occur outside opportunity support."
    )
  }

  support_rates <- source %>%
    group_by(treated) %>%
    summarise(
      n_source = n(),
      n_opportunity = sum(.S == 1, na.rm = TRUE),
      n_available_in_opportunity = sum(.H == 1 & .S == 1, na.rm = TRUE),
      availability_given_opportunity = if_else(
        n_opportunity > 0,
        n_available_in_opportunity / n_opportunity,
        NA_real_
      ),
      .groups = "drop"
    ) %>%
    mutate(
      stage = stage_name,
      support_is_structural_subset = enforce_availability_subset,
      n_available_outside_support = n_mismatch,
      sparse_selected_arm =
        n_available_in_opportunity < IPHAW_MIN_SELECTED_PER_ARM_DIAGNOSTIC,
      stage_sparse_support = any(sparse_selected_arm)
    )

  write_csv_quiet(
    support_rates,
    paste0(stage_name, "_STRUCTURAL_SUPPORT.csv"),
    subdir = "diagnostics"
  )

  dat <- source %>%
    filter(
      .S == 1,
      !is.na(landmark_id),
      !is.na(.H),
      !is.na(treated),
      !is.na(wave_n),
      !is.na(sex),
      !is.na(age),
      !is.na(cohab_duration_months),
      !is.na(prior_outcome),
      !is.na(n_prior_interviews),
      !is.na(n_prior_outcome_obs),
      !is.na(panel_tenure_months)
    )

  empty_weights <- function(ids = character(0)) {
    tibble(
      landmark_id = ids,
      .H = NA_integer_,
      .S = NA_integer_,
      p_hat = NA_real_,
      ipha_weight_raw = NA_real_,
      ipha_weight_trim = NA_real_
    )
  }

  if (nrow(dat) == 0L) {
    return(list(
      weights = empty_weights(),
      model = NULL,
      formula_tier = "no_support",
      support_diagnostics = support_rates
    ))
  }

  if (length(unique(dat$.H)) == 1L) {
    p_hat <- rep(mean(dat$.H), nrow(dat))
    ipha_raw <- ifelse(dat$.H == 1, 1, NA_real_)
    ipha_trim <- ipha_raw
    fit <- NULL
    formula_tier <- "constant_availability"
  } else {

    fitted <- fit_bias_reduced_availability(dat)
    fit <- fitted$model
    formula_tier <- fitted$formula_tier

    if (is.null(fit)) {
      write_txt(
        paste0(
          "BIAS-REDUCED IPHAW MODEL FAILED\n",
          "No brglm2 specification produced usable predictions."
        ),
        paste0(stage_name, "_IPHAW_ERROR.txt"),
        subdir = "diagnostics"
      )

      return(list(
        weights = tibble(
          landmark_id = dat$landmark_id,
          .H = dat$.H,
          .S = dat$.S,
          p_hat = NA_real_,
          ipha_weight_raw = NA_real_,
          ipha_weight_trim = NA_real_
        ),
        model = NULL,
        formula_tier = "failed",
        support_diagnostics = support_rates
      ))
    }

    saveRDS(
      fit,
      file.path(
        OUT_DIR,
        "models",
        paste0(stage_name, "_IPHAW_brglm2.rds")
      )
    )

    write_txt(
      c(
        paste0("Formula tier: ", formula_tier),
        "Estimator: brglm2 mean bias-reduced logistic regression (AS_mean)",
        "Target: structural-opportunity support only",
        "",
        capture.output(summary(fit))
      ),
      paste0(stage_name, "_IPHAW_brglm2_summary.txt"),
      subdir = "diagnostics"
    )

    p_hat <- suppressWarnings(
      stats::predict(fit, type = "response")
    )

    # Numerical safeguard only. Structural zero probabilities have already
    # been removed by the opportunity restriction.
    p_hat <- pmin(
      pmax(p_hat, IPHAW_PROB_EPS),
      1 - IPHAW_PROB_EPS
    )

    numerator <- ave(
      dat$.H,
      dat$treated,
      FUN = function(z) mean(z == 1)
    )

    ipha_raw <- ifelse(
      dat$.H == 1,
      numerator / p_hat,
      NA_real_
    )

    ipha_trim <- trim_weight_by_treatment(
      ipha_raw,
      dat$treated,
      IPHAW_TRIM_LOWER,
      IPHAW_TRIM_UPPER
    )
  }

  diag_dat <- dat %>%
    mutate(
      p_hat = p_hat,
      ipha_weight_raw = ipha_raw,
      ipha_weight_trim = ipha_trim
    )

  propensity_summary <- diag_dat %>%
    group_by(treated, .H) %>%
    summarise(
      n = n(),
      p_hat_mean = mean(p_hat, na.rm = TRUE),
      p_hat_min = if (all(is.na(p_hat))) NA_real_ else min(p_hat, na.rm = TRUE),
      p_hat_p05 = if (all(is.na(p_hat))) NA_real_ else quantile(p_hat, 0.05, na.rm = TRUE),
      p_hat_median = if (all(is.na(p_hat))) NA_real_ else median(p_hat, na.rm = TRUE),
      p_hat_p95 = if (all(is.na(p_hat))) NA_real_ else quantile(p_hat, 0.95, na.rm = TRUE),
      p_hat_max = if (all(is.na(p_hat))) NA_real_ else max(p_hat, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      stage = stage_name,
      formula_tier = formula_tier
    )

  write_csv_quiet(
    propensity_summary,
    paste0(stage_name, "_propensity_summary.csv"),
    subdir = "diagnostics"
  )

  wdat <- diag_dat %>%
    filter(
      .H == 1,
      is.finite(ipha_weight_raw),
      ipha_weight_raw > 0
    )

  weight_summary <- wdat %>%
    group_by(treated) %>%
    summarise(
      n = n(),
      raw_mean = mean(ipha_weight_raw, na.rm = TRUE),
      raw_p99 = if (n() > 0) quantile(ipha_weight_raw, 0.99, na.rm = TRUE) else NA_real_,
      raw_max = if (n() > 0) max(ipha_weight_raw, na.rm = TRUE) else NA_real_,
      trim_mean = mean(ipha_weight_trim, na.rm = TRUE),
      trim_p99 = if (n() > 0) quantile(ipha_weight_trim, 0.99, na.rm = TRUE) else NA_real_,
      trim_max = if (n() > 0) max(ipha_weight_trim, na.rm = TRUE) else NA_real_,
      ess_raw = effective_sample_size(ipha_weight_raw),
      ess_trim = effective_sample_size(ipha_weight_trim),
      .groups = "drop"
    )

  write_csv_quiet(
    weight_summary,
    paste0(stage_name, "_weight_summary.csv"),
    subdir = "diagnostics"
  )

  target_vars <- c(
    "age",
    "cohab_duration_months",
    "prior_outcome",
    "n_prior_interviews",
    "n_prior_outcome_obs",
    "panel_tenure_months"
  )

  target_balance <- map_dfr(target_vars, function(v) {
    tibble(
      variable = v,
      smd_selected_unweighted_vs_opportunity_target = smd_to_target(
        dat[[v]],
        wdat[[v]],
        rep(1, nrow(wdat))
      ),
      smd_selected_IPHAW_vs_opportunity_target = smd_to_target(
        dat[[v]],
        wdat[[v]],
        wdat$ipha_weight_trim
      )
    )
  })

  write_csv_quiet(
    target_balance,
    paste0(stage_name, "_target_balance.csv"),
    subdir = "diagnostics"
  )

  list(
    weights = diag_dat %>%
      select(
        landmark_id,
        .H,
        .S,
        p_hat,
        ipha_weight_raw,
        ipha_weight_trim
      ),
    model = fit,
    formula_tier = formula_tier,
    support_diagnostics = support_rates
  )
}

read_one_wave <- function(path, wave) {
  nms <- safe_read_names(path)
  resolved <- map_chr(variable_specs, ~ find_wave_var(nms, wave, .x))
  names(resolved) <- names(variable_specs)
  selected <- unique(na.omit(unname(resolved)))
  dat <- haven::read_dta(path, col_select = dplyr::any_of(selected))

  get_standard <- function(concept) {
    v <- resolved[[concept]]
    if (is.na(v) || !v %in% names(dat)) {
      rep(NA_real_, nrow(dat))
    } else {
      suppressWarnings(as.numeric(dat[[v]]))
    }
  }

  tibble(
    wave = wave,
    wave_n = wave_number(wave),
    pidp = get_standard("pidp"),
    interview_year = num_clean(get_standard("interview_year")),
    interview_month = num_clean(get_standard("interview_month")),
    age = num_clean(get_standard("age")),
    sex = num_clean(get_standard("sex")),
    mastat_dv = num_clean(get_standard("mastat_dv")),
    ghq = num_clean(get_standard("ghq")),
    life_satisfaction = num_clean(get_standard("life_satisfaction")),
    rdas_satisfaction = num_clean(get_standard("rdas_satisfaction")),
    rdas_cohesion = num_clean(get_standard("rdas_cohesion")),
    ncrr1 = num_clean(get_standard("ncrr1")),
    ncrr11 = num_clean(get_standard("ncrr11")),
    ncrr12 = num_clean(get_standard("ncrr12")),
    ncrrm = num_clean(get_standard("ncrrm")),
    ncrry4 = num_clean(get_standard("ncrry4"))
  ) %>%
    mutate(
      interview_ym = ym_index(interview_year, interview_month),
      lat_relationship_start_ym = ym_index(ncrry4, ncrrm),
      ghq = if_else(ghq >= 0 & ghq <= 36, ghq, NA_real_),
      life_satisfaction = if_else(
        life_satisfaction >= 1 & life_satisfaction <= 7,
        life_satisfaction, NA_real_
      ),
      rdas_satisfaction = if_else(rdas_satisfaction >= 0, rdas_satisfaction, NA_real_),
      rdas_cohesion = if_else(rdas_cohesion >= 0, rdas_cohesion, NA_real_)
    )
}

    usable_breakup_date_sens =
      cohab_ongoing == 0 &
      cohend == 1 &
      cohab_end_if %in% BREAKUP_DATE_FLAGS_SENSITIVITY,

    months_to_breakup = if_else(
      cohab_ongoing == 0 & cohend == 1,
      months_remaining_in_cohab,
      NA_real_
    ),

    # Retained as an audit variable and explicitly used in sample-flow output.
    validated_breakup_24m =
      usable_breakup_date_sens &
      months_to_breakup >= 1 &
      months_to_breakup <= 24,

    # Operational comparison group for BOTH marriage and breakup arms:
    # the respondent is observed to remain in the same cohabiting spell for
    # at least another 60 months. Thus neither marriage nor breakup occurs
    # during the reference horizon.
    continued_cohab_60m =
      months_remaining_in_cohab >= CONTROL_HORIZON_MONTHS &
      (cohab_ongoing == 1 | cohab_end_if %in% c(0, 1)),

    marriage_horizon_state = case_when(
      exact_marriage_date &
        !is.na(months_to_marriage) &
        months_to_marriage >= ifelse(EXCLUDE_SAME_MONTH_MARRIAGE, 1, 0) &
        months_to_marriage <= 12 ~ "marriage_1_12m",
      exact_marriage_date &
        !is.na(months_to_marriage) &
        months_to_marriage >= 13 &
        months_to_marriage <= 24 ~ "marriage_13_24m",
      exact_marriage_date &
        !is.na(months_to_marriage) &
        months_to_marriage >= 25 &
        months_to_marriage <= 60 ~ "marriage_25_60m",
      continued_cohab_60m ~ "continued_cohab_60m",
      TRUE ~ NA_character_
    ),

    breakup_horizon_state = case_when(
      exact_breakup_date &
        !is.na(months_to_breakup) &
        months_to_breakup >= 1 &
        months_to_breakup <= 12 ~ "breakup_1_12m",
      exact_breakup_date &
        !is.na(months_to_breakup) &
        months_to_breakup >= 13 &
        months_to_breakup <= 24 ~ "breakup_13_24m",
      exact_breakup_date &
        !is.na(months_to_breakup) &
        months_to_breakup >= 25 &
        months_to_breakup <= 60 ~ "breakup_25_60m",
      continued_cohab_60m ~ "continued_cohab_60m",
      TRUE ~ NA_character_
    )
  )

# -----------------------------------------------------------------------------
# 6B. STRUCTURAL OPPORTUNITY FOR H1/H2/H3
# -----------------------------------------------------------------------------
# Validated against the dedicated v2 audit: observed H1/H2/H3 histories must be
# subsets of these opportunity populations. Opportunity is based on EXISTENCE
# of any admissible scheduled prior measurement/pair on the full cohab_panel;
# it does not depend on whether the respondent answered the item.

cohab_panel <- cohab_panel %>%
  left_join(
    module_schedule_core,
    by = c("wave", "wave_n")
  ) %>%
  mutate(
    ghq_module = coalesce(ghq_module, FALSE),
    ls_module = coalesce(ls_module, FALSE),
    rdas_sat_module = coalesce(rdas_sat_module, FALSE),
    rdas_coh_module = coalesce(rdas_coh_module, FALSE)
  ) %>%
  arrange(
    relationship_spell_id,
    interview_ym,
    wave_n
  ) %>%
  group_by(relationship_spell_id) %>%
  group_modify(~ {

    d <- .x
    tt <- d$interview_ym
    n <- nrow(d)

    ghq_change_opp <- compute_change_opportunity(
      tt,
      d$ghq_module,
      PRIOR_PSYCH_MIN_GAP_MONTHS,
      PRIOR_PSYCH_MAX_GAP_MONTHS
    )

    ls_change_opp <- compute_change_opportunity(
      tt,
      d$ls_module,
      PRIOR_PSYCH_MIN_GAP_MONTHS,
      PRIOR_PSYCH_MAX_GAP_MONTHS
    )

    sat_change_opp <- compute_change_opportunity(
      tt,
      d$rdas_sat_module,
      PRIOR_RDAS_MIN_GAP_MONTHS,
      PRIOR_RDAS_MAX_GAP_MONTHS
    )

    coh_change_opp <- compute_change_opportunity(
      tt,
      d$rdas_coh_module,
      PRIOR_RDAS_MIN_GAP_MONTHS,
      PRIOR_RDAS_MAX_GAP_MONTHS
    )

    H1_ghq <- rep(FALSE, n)
    H1_ls <- rep(FALSE, n)
    H2 <- rep(FALSE, n)
    H3 <- rep(FALSE, n)

    for (j in seq_len(n)) {

      H1_ghq[j] <- exists_prior_change_opp(
        tt,
        ghq_change_opp,
        j,
        PRIOR_CHANGE_MAX_GAP_MONTHS
      )

      H1_ls[j] <- exists_prior_change_opp(
        tt,
        ls_change_opp,
        j,
        PRIOR_CHANGE_MAX_GAP_MONTHS
      )

      sat_level_opp <- exists_prior_module_in_window(
        tt,
        d$rdas_sat_module,
        j,
        PRIOR_RDAS_MIN_GAP_MONTHS,
        PRIOR_RDAS_MAX_GAP_MONTHS
      )

      coh_level_opp <- exists_prior_module_in_window(
        tt,
        d$rdas_coh_module,
        j,
        PRIOR_RDAS_MIN_GAP_MONTHS,
        PRIOR_RDAS_MAX_GAP_MONTHS
      )

      H2[j] <- sat_level_opp & coh_level_opp

      H3[j] <-
        exists_prior_change_opp(
          tt,
          sat_change_opp,
          j,
          PRIOR_CHANGE_MAX_GAP_MONTHS
        ) &
        exists_prior_change_opp(
          tt,
          coh_change_opp,
          j,
          PRIOR_CHANGE_MAX_GAP_MONTHS
        )
    }

    amb <- d$ambiguous_active_cohab

    d %>%
      mutate(
        H1_ghq_opportunity = H1_ghq,
        H1_ls_opportunity = H1_ls,
        H2_opportunity = H2,
        H3_opportunity = H3,

        H1_ghq_source_ambiguous = purrr::map_lgl(
          prior_ghq_change_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        ),
        H1_ls_source_ambiguous = purrr::map_lgl(
          prior_ls_change_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        ),
        H2_sat_source_ambiguous = purrr::map_lgl(
          prior_rdas_satisfaction_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        ),
        H2_coh_source_ambiguous = purrr::map_lgl(
          prior_rdas_cohesion_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        ),
        H3_sat_source_ambiguous = purrr::map_lgl(
          prior_rdas_sat_change_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        ),
        H3_coh_source_ambiguous = purrr::map_lgl(
          prior_rdas_coh_change_ym,
          ~ isTRUE(source_row_ambiguous(.x, tt, amb))
        )
      )
  }) %>%
  ungroup()

make_pre_rel_sample <- function(min_gap) {
  pre_rel_candidates %>%
    filter(
      months_to_next_union >= min_gap,
      months_to_next_union <= PRE_REL_MAX_GAP_MONTHS,
      focal_union_outcome %in% c("marriage", "breakup", "stable_cohab")

      ls_z = zsafe(life_satisfaction),
      ghq_change_z = zsafe(ghq_change),
      ls_change_z = zsafe(ls_change)
    )
}

# 8. RISK-SET CONSTRUCTION HELPER

add_riskset_id <- function(dat) {
  dat %>%
    mutate(
      duration_stratum = floor(
        cohab_duration_months / RISKSET_DURATION_BIN_MONTHS
      ),
      age_stratum = floor(age / RISKSET_AGE_BIN_YEARS),
      riskset_stratum = interaction(
        wave, sex, duration_stratum, age_stratum, drop = TRUE
      )
    )
}

make_att_riskset <- function(dat, treated_var = "treated") {
  dat <- dat %>%
    filter(
      !is.na(riskset_stratum),
      !is.na(.data[[treated_var]])
    )

  counts <- dat %>%
    group_by(riskset_stratum) %>%
    summarise(
      n_treated = sum(.data[[treated_var]] == 1, na.rm = TRUE),
      n_control = sum(.data[[treated_var]] == 0, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    filter(n_treated > 0, n_control > 0)

  dat %>%
    inner_join(counts, by = "riskset_stratum") %>%
    mutate(
      att_weight = if_else(
        .data[[treated_var]] == 1,
        1,
        n_treated / n_control
      )
    )
}

# C4: reverse temporal ordering, prior psychological change -> later RDAS change.
run_C4 <- function(
  outcome_change,
  prior_psych_change,
  prior_psych_ok,
  prior_rdas_level,
  model_name
) {
  dat <- development %>%
    mutate(
      outcome_change_tmp = .data[[outcome_change]],
      prior_psych_change_tmp = .data[[prior_psych_change]],
      prior_psych_ok_tmp = .data[[prior_psych_ok]],
      prior_rdas_level_tmp = .data[[prior_rdas_level]]
    ) %>%
    filter(
      !is.na(outcome_change_tmp),
      prior_psych_ok_tmp,
      !is.na(prior_rdas_level_tmp)
    ) %>%
    mutate(
      outcome_change_z = zsafe(outcome_change_tmp),
      prior_psych_change_z = zsafe(prior_psych_change_tmp),
      prior_rdas_level_z = zsafe(prior_rdas_level_tmp)
    )

  safe_feols(
    outcome_change_z ~
      prior_psych_change_z +
      prior_rdas_level_z +
      ns(cohab_duration_months, df = 4) |
      relationship_spell_id + wave,
    data = dat,
    name = model_name,
    vcov = ~pidp
  )
}

# 11. ANALYSES D_M AND D_B: PROXIMITY RISK-SET ANALYSES
# -----------------------------------------------------------------------------
#
# D_M: exact marriage proximity
#   1-12m, 13-24m, 25-60m versus continued cohabitation >=60m
#
# D_B: exact validated-breakup proximity
#   1-12m, 13-24m, 25-60m versus continued cohabitation >=60m
#
# Outcome: RECENT within-relationship psychological change.
#
# Risk sets: wave x sex x 6-month duration x 2-year age.
# Treated observations receive weight 1. Controls receive n_treated/n_control
# within risk set, yielding an ATT-oriented coarsened risk-set comparison.
#
# MISSING PRIOR CHANGE
# --------------------
# There is NO mean-fill + missing-indicator method in this version.
# Each contrast is estimated as:
#
# D0  Broad sample: prior outcome level only.
# D1a History-complete baseline: same formula, restricted to observed prior
#     psychological change.
# D1b Same D1a sample + observed prior psychological change.
# D1c D1b + IPHAW sensitivity for selective psychological-history availability.
# D2a RDAS-level-complete common-sample baseline.
# D2b Same D2a sample + prior RDAS satisfaction/cohesion levels.
# D2c D2b + sequential IPHAW sensitivity.
# D3a RDAS-change-complete common-sample baseline.
# D3b Same D3a sample + prior RDAS satisfaction/cohesion changes.
# D3c D3b + sequential IPHAW sensitivity.
#
# E below quantifies DESCRIPTIVE attenuation only between nested models fitted
# on identical common samples. These are not causal mediated proportions.

prepare_proximity_pair <- function(
  data,
  state_var,
  treated_state,
  outcome = c("ghq", "ls")
) {
  outcome <- match.arg(outcome)

  dat <- data %>%
    filter(
      .data[[state_var]] %in% c(
        treated_state,
        "continued_cohab_60m"
      )
    ) %>%
    mutate(
      treated = as.integer(.data[[state_var]] == treated_state)
    )

  if (outcome == "ghq") {
    dat <- dat %>%
      filter(
        !is.na(spell_ghq_change),
        !is.na(prior_spell_ghq)
      ) %>%
      mutate(
        outcome_change = spell_ghq_change,
        prior_outcome = prior_spell_ghq,
        prior_outcome_change = prior_ghq_change,
        n_prior_outcome_obs = n_prior_ghq_obs,
        psych_history_available =
          usable_prior_ghq_change & !is.na(prior_ghq_change),
        H1_opportunity = H1_ghq_opportunity,
        H1_source_ambiguous = H1_ghq_source_ambiguous,

        strict_outcome_change = strict_spell_ghq_change,
        strict_prior_outcome = strict_prior_spell_ghq,
        strict_prior_outcome_change = strict_prior_ghq_change,
        strict_psych_history_available =
          strict_usable_prior_ghq_change & !is.na(strict_prior_ghq_change)
      )
  } else {
    dat <- dat %>%
      filter(
        !is.na(spell_ls_change),
        !is.na(prior_spell_ls)
      ) %>%
      mutate(
        outcome_change = spell_ls_change,
        prior_outcome = prior_spell_ls,
        prior_outcome_change = prior_ls_change,
        n_prior_outcome_obs = n_prior_ls_obs,
        psych_history_available =
          usable_prior_ls_change & !is.na(prior_ls_change),
        H1_opportunity = H1_ls_opportunity,
        H1_source_ambiguous = H1_ls_source_ambiguous,

        strict_outcome_change = strict_spell_ls_change,
        strict_prior_outcome = strict_prior_spell_ls,
        strict_prior_outcome_change = strict_prior_ls_change,
        strict_psych_history_available =
          strict_usable_prior_ls_change & !is.na(strict_prior_ls_change)
      )
  }

  dat %>%
    mutate(
      rdas_level_available =
        recent_prior_both_rdas &
        !is.na(prior_rdas_satisfaction) &
        !is.na(prior_rdas_cohesion),

      rdas_change_available =
        usable_prior_rdas_change &
        !is.na(prior_rdas_sat_change) &
        !is.na(prior_rdas_coh_change),

      H2_source_ambiguous =
        H2_sat_source_ambiguous | H2_coh_source_ambiguous,

      H3_source_ambiguous =
        H3_sat_source_ambiguous | H3_coh_source_ambiguous,

      D2_opportunity =
        H1_opportunity & H2_opportunity,

      D3_opportunity =
        H1_opportunity & H2_opportunity & H3_opportunity,

      joint_history_available =
        psych_history_available &
        rdas_level_available &
        rdas_change_available,

      strict_rdas_level_available =
        strict_recent_prior_both_rdas &
        !is.na(strict_prior_rdas_satisfaction) &
        !is.na(strict_prior_rdas_cohesion),

      strict_rdas_change_available =
        strict_usable_prior_rdas_change &
        !is.na(strict_prior_rdas_sat_change) &
        !is.na(strict_prior_rdas_coh_change)
    )
}

# Rebuild ATT weights after each complete-history restriction. This prevents
# old broad-sample n_treated/n_control values from leaking into nested samples.
make_att_riskset_v2 <- function(dat) {
  dat <- dat %>%
    select(-any_of(c("n_treated", "n_control", "att_weight"))) %>%
    filter(!is.na(riskset_stratum))

  make_att_riskset(dat, treated_var = "treated")
}

export_proximity_sample <- function(dat, stub, state_var) {
  if (nrow(dat) == 0L) return(invisible(NULL))

  write_csv_quiet(
    dat %>%
      group_by(.data[[state_var]]) %>%
      summarise(
        n_landmarks = n(),
        n_persons = n_distinct(pidp),
        weighted_mass_ATT = sum(att_weight, na.rm = TRUE),
        .groups = "drop"
      ),
    paste0(stub, "_sample.csv")
  )

  balance_vars <- c(
    "cohab_duration_months",
    "age",
    "prior_outcome"
  )

  if ("prior_outcome_change" %in% names(dat) &&
      any(!is.na(dat$prior_outcome_change))) {
    balance_vars <- c(balance_vars, "prior_outcome_change")
  }

  balance <- map_dfr(balance_vars, function(v) {
    tibble(
      variable = v,
      smd_unweighted = smd_two_group(dat[[v]], dat$treated),
      smd_ATT = smd_two_group(dat[[v]], dat$treated, dat$att_weight)
    )
  })

  write_csv_quiet(
    balance,
    paste0(stub, "_balance.csv"),
    subdir = "diagnostics"
  )
}

fit_strict_unambiguous_sensitivity <- function(
  pair,
  prefix
) {

  if (!RUN_STRICT_UNAMBIGUOUS_HISTORY_SENSITIVITY) {
    return(list(
      D1b_strict = NULL,
      D2b_strict = NULL,
      D3b_strict = NULL,
      counts = tibble()
    ))
  }

  d1_dat <- pair %>%
    filter(
      !is.na(strict_outcome_change),
      !is.na(strict_prior_outcome),
      strict_psych_history_available,
      !is.na(strict_prior_outcome_change)
    ) %>%
    mutate(
      strict_y = strict_outcome_change,
      strict_y_lag = strict_prior_outcome,
      strict_dy_lag = strict_prior_outcome_change
    )

  d1_rs <- make_att_riskset_v2(d1_dat)

  D1b_strict <- safe_feols(
    strict_y ~
      treated +
      strict_y_lag +
      strict_dy_lag |
      riskset_stratum,
    data = d1_rs,
    name = paste0(prefix, "_STRICT_D1b_unambiguous_history"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  d2_dat <- d1_dat %>%
    filter(
      strict_rdas_level_available,
      !is.na(strict_prior_rdas_satisfaction),
      !is.na(strict_prior_rdas_cohesion)
    ) %>%
    mutate(
      strict_rdas_sat_lag = strict_prior_rdas_satisfaction,
      strict_rdas_coh_lag = strict_prior_rdas_cohesion
    )

  d2_rs <- make_att_riskset_v2(d2_dat)

  D2b_strict <- safe_feols(
    strict_y ~
      treated +
      strict_y_lag +
      strict_dy_lag +
      strict_rdas_sat_lag +
      strict_rdas_coh_lag |
      riskset_stratum,
    data = d2_rs,
    name = paste0(prefix, "_STRICT_D2b_unambiguous_history"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  d3_dat <- d2_dat %>%
    filter(
      strict_rdas_change_available,
      !is.na(strict_prior_rdas_sat_change),
      !is.na(strict_prior_rdas_coh_change)
    ) %>%
    mutate(
      strict_rdas_sat_change_lag = strict_prior_rdas_sat_change,
      strict_rdas_coh_change_lag = strict_prior_rdas_coh_change
    )

  d3_rs <- make_att_riskset_v2(d3_dat)

  D3b_strict <- safe_feols(
    strict_y ~
      treated +
      strict_y_lag +
      strict_dy_lag +
      strict_rdas_sat_lag +
      strict_rdas_coh_lag +
      strict_rdas_sat_change_lag +
      strict_rdas_coh_change_lag |
      riskset_stratum,
    data = d3_rs,
    name = paste0(prefix, "_STRICT_D3b_unambiguous_history"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  list(
    D1b_strict = D1b_strict,
    D2b_strict = D2b_strict,
    D3b_strict = D3b_strict,
    counts = tibble(
      stage = c("strict_D1", "strict_D2", "strict_D3"),
      n_landmarks = c(nrow(d1_rs), nrow(d2_rs), nrow(d3_rs)),
      n_persons = c(
        n_distinct(d1_rs$pidp),
        n_distinct(d2_rs$pidp),
        n_distinct(d3_rs$pidp)
      )
    )
  )
}

run_proximity_sequence <- function(
  data,
  state_var,
  treated_state,
  outcome,
  arm
) {

  pair <- prepare_proximity_pair(
    data = data,
    state_var = state_var,
    treated_state = treated_state,
    outcome = outcome
  )

  prefix <- paste0(
    "D_", arm, "_", outcome, "_", treated_state,
    "_vs_continued_cohab_60m"
  )

  # ---------------------------------------------------------------------------
  # D0: broad eligible sample, prior level only.
  # ---------------------------------------------------------------------------
  broad_rs <- make_att_riskset_v2(pair)
  export_proximity_sample(
    broad_rs,
    paste0(prefix, "_D0_broad"),
    state_var
  )

  broad_rs <- broad_rs %>%
    mutate(
      H1_raw = as.integer(psych_history_available),
      H2_raw = as.integer(rdas_level_available),
      H3_raw = as.integer(rdas_change_available),
      H1_opp_raw = as.integer(H1_opportunity),
      H2_opp_raw = as.integer(H2_opportunity),
      H3_opp_raw = as.integer(H3_opportunity)
    )

  opportunity_diagnostics <- broad_rs %>%
    group_by(treated) %>%
    summarise(
      n = n(),
      H1_opportunity_n = sum(H1_opportunity, na.rm = TRUE),
      H1_available_n = sum(psych_history_available, na.rm = TRUE),
      H1_available_given_opp_rate = if_else(
        H1_opportunity_n > 0,
        sum(psych_history_available & H1_opportunity, na.rm = TRUE) /
          H1_opportunity_n,
        NA_real_
      ),
      H1_available_without_opp = sum(
        psych_history_available & !H1_opportunity,
        na.rm = TRUE
      ),
      H2_opportunity_n = sum(H2_opportunity, na.rm = TRUE),
      H2_available_n = sum(rdas_level_available, na.rm = TRUE),
      H2_available_given_opp_rate = if_else(
        H2_opportunity_n > 0,
        sum(rdas_level_available & H2_opportunity, na.rm = TRUE) /
          H2_opportunity_n,
        NA_real_
      ),
      H2_available_without_opp = sum(
        rdas_level_available & !H2_opportunity,
        na.rm = TRUE
      ),
      H3_opportunity_n = sum(H3_opportunity, na.rm = TRUE),
      H3_available_n = sum(rdas_change_available, na.rm = TRUE),
      H3_available_given_opp_rate = if_else(
        H3_opportunity_n > 0,
        sum(rdas_change_available & H3_opportunity, na.rm = TRUE) /
          H3_opportunity_n,
        NA_real_
      ),
      H3_available_without_opp = sum(
        rdas_change_available & !H3_opportunity,
        na.rm = TRUE
      ),
      .groups = "drop"
    ) %>%
    mutate(
      arm = arm,
      horizon = treated_state,
      outcome = outcome
    )

  write_csv_quiet(
    opportunity_diagnostics,
    paste0(prefix, "_STRUCTURAL_OPPORTUNITY_DIAGNOSTIC.csv"),
    subdir = "diagnostics"
  )

  mismatch_total <- opportunity_diagnostics %>%
    summarise(
      n = sum(
        H1_available_without_opp +
          H2_available_without_opp +
          H3_available_without_opp,
        na.rm = TRUE
      )
    ) %>%
    pull(n)

  if (!is.na(mismatch_total) && mismatch_total > 0L) {
    stop(
      "Validated structural-opportunity definition failed inside ",
      prefix,
      ". Inspect the structural-opportunity diagnostic before continuing."
    )
  }

  availability_pattern <- bind_rows(
    broad_rs %>%
      count(
        treated,
        H1_raw,
        H2_raw,
        H3_raw,
        H1_opp_raw,
        H2_opp_raw,
        H3_opp_raw,
        name = "n"
      ) %>%
      group_by(treated) %>%
      mutate(
        proportion_within_treatment = n / sum(n),
        scope = if_else(treated == 1, "treated", "control")
      ) %>%
      ungroup(),

    broad_rs %>%
      count(
        H1_raw,
        H2_raw,
        H3_raw,
        H1_opp_raw,
        H2_opp_raw,
        H3_opp_raw,
        name = "n"
      ) %>%
      mutate(
        treated = NA_integer_,
        proportion_within_treatment = n / sum(n),
        scope = "overall"
      )
  ) %>%
    mutate(
      H2_without_H1 = H2_raw == 1 & H1_raw == 0,
      H3_without_H2 = H3_raw == 1 & H2_raw == 0,
      H3_without_H1 = H3_raw == 1 & H1_raw == 0,
      arm = arm,
      horizon = treated_state,
      outcome = outcome
    )

  write_csv_quiet(
    availability_pattern,
    paste0(prefix, "_RAW_H1_H2_H3_AVAILABILITY_AND_OPPORTUNITY.csv"),
    subdir = "diagnostics"
  )

  availability_nonmonotone_summary <- broad_rs %>%
    summarise(
      n = n(),
      pct_H2_without_H1 = 100 * mean(
        rdas_level_available & !psych_history_available,
        na.rm = TRUE
      ),
      pct_H3_without_H2 = 100 * mean(
        rdas_change_available & !rdas_level_available,
        na.rm = TRUE
      ),
      pct_H3_without_H1 = 100 * mean(
        rdas_change_available & !psych_history_available,
        na.rm = TRUE
      ),
      pct_joint_complete = 100 * mean(
        joint_history_available,
        na.rm = TRUE
      )
    ) %>%
    mutate(
      arm = arm,
      horizon = treated_state,
      outcome = outcome,
      interpretation = paste(
        "Raw availability need not be naturally monotone.",
        "Sequential D2/D3 probabilities are estimated conditionally within",
        "the corresponding validated structural-opportunity targets."
      )
    )

  write_csv_quiet(
    availability_nonmonotone_summary,
    paste0(prefix, "_AVAILABILITY_NESTING_DIAGNOSTIC.csv"),
    subdir = "diagnostics"
  )

  D0 <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome |
      riskset_stratum,
    data = broad_rs,
    name = paste0(prefix, "_D0_broad_prior_level"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  # ---------------------------------------------------------------------------
  # D1: complete psychological history + mandatory D1c sensitivity.
  # ---------------------------------------------------------------------------
  # D1b: ATT in the observed-history-complete population.
  # D1c: same observed-history rows, but ATT and history weights target the
  #      H1 STRUCTURAL-OPPORTUNITY population.

  D1_support <- broad_rs %>%
    filter(H1_opportunity) %>%
    make_att_riskset_v2() %>%
    rename(D1_target_att_weight = att_weight)

  h1 <- estimate_ipha_stage(
    D1_support,
    availability_var = "psych_history_available",
    opportunity_var = "H1_opportunity",
    stage_name = paste0(prefix, "_H1_psych_history_D1_target")
  )

  h1w <- h1$weights %>%
    rename(
      H1_available = .H,
      H1_supported = .S,
      H1_p_hat = p_hat,
      H1_weight_raw = ipha_weight_raw,
      H1_weight_trim = ipha_weight_trim
    )

  history_complete <- broad_rs %>%
    filter(
      psych_history_available,
      !is.na(prior_outcome_change)
    )

  history_rs_base <- make_att_riskset_v2(history_complete)

  export_proximity_sample(
    history_rs_base,
    paste0(prefix, "_D1_history_complete"),
    state_var
  )

  D1a <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome |
      riskset_stratum,
    data = history_rs_base,
    name = paste0(prefix, "_D1a_history_baseline"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  D1b <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change |
      riskset_stratum,
    data = history_rs_base,
    name = paste0(prefix, "_D1b_plus_prior_psych_change"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  history_rs <- history_rs_base %>%
    left_join(
      h1w,
      by = "landmark_id"
    ) %>%
    left_join(
      D1_support %>%
        select(
          landmark_id,
          D1_target_att_weight
        ),
      by = "landmark_id"
    ) %>%
    filter(
      H1_available == 1,
      !is.na(H1_weight_raw),
      !is.na(D1_target_att_weight)
    )

  w1 <- make_final_weight_variants(
    history_rs$D1_target_att_weight,
    history_rs$H1_weight_raw,
    history_rs$treated
  )

  history_rs <- history_rs %>%
    mutate(
      combined_weight_H1_raw = w1$raw,
      combined_weight_H1 = w1$trim_1_99,
      combined_weight_H1_tight = w1$trim_2p5_97p5
    )

  D1c <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change |
      riskset_stratum,
    data = history_rs,
    name = paste0(
      prefix,
      "_D1c_H1opportunity_target_IPHAW_finaltrim"
    ),
    weights = ~combined_weight_H1,
    vcov = ~pidp
  )

  D1c_tight <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change |
      riskset_stratum,
    data = history_rs,
    name = paste0(
      prefix,
      "_D1c_H1opportunity_target_IPHAW_finaltrim_2p5_97p5"
    ),
    weights = ~combined_weight_H1_tight,
    vcov = ~pidp
  )

  diag_H1_raw <- weight_stability_diagnostics(
    history_rs,
    "combined_weight_H1_raw",
    "D1c H1-opportunity-target ATT x H1 raw product before final trimming"
  )

  diag_H1 <- weight_stability_diagnostics(
    history_rs,
    "combined_weight_H1",
    "D1c H1-opportunity-target ATT x H1 final-trimmed IPHAW"
  )

  write_csv_quiet(
    bind_rows(
      diag_H1_raw,
      diag_H1
    ),
    paste0(prefix, "_D1c_FINAL_WEIGHT_DIAGNOSTICS.csv"),
    subdir = "diagnostics"
  )

  # ---------------------------------------------------------------------------
  # D2: H1 + RDAS-level complete. IPHAW target = H1 & H2 opportunity support.
  # ---------------------------------------------------------------------------

  D2_support <- broad_rs %>%
    filter(D2_opportunity) %>%
    make_att_riskset_v2() %>%
    rename(D2_target_att_weight = att_weight)

  h1_D2 <- estimate_ipha_stage(
    D2_support,
    availability_var = "psych_history_available",
    opportunity_var = "D2_opportunity",
    stage_name = paste0(prefix, "_H1_for_D2_target")
  )

  D2_support_H1 <- D2_support %>%
    mutate(
      D2_H2_conditional_support = psych_history_available
    )

  h2_D2 <- estimate_ipha_stage(
    D2_support_H1,
    availability_var = "rdas_level_available",
    opportunity_var = "D2_H2_conditional_support",
    stage_name = paste0(prefix, "_H2_RDAS_level_D2_target"),
    enforce_availability_subset = FALSE
  )

  h1_D2w <- h1_D2$weights %>%
    rename(
      H1_D2_weight_raw = ipha_weight_raw,
      H1_D2_weight_trim = ipha_weight_trim
    ) %>%
    select(
      landmark_id,
      H1_D2_weight_raw,
      H1_D2_weight_trim
    )

  h2_D2w <- h2_D2$weights %>%
    rename(
      H2_D2_weight_raw = ipha_weight_raw,
      H2_D2_weight_trim = ipha_weight_trim
    ) %>%
    select(
      landmark_id,
      H2_D2_weight_raw,
      H2_D2_weight_trim
    )

  rdas_level_complete <- broad_rs %>%
    filter(
      psych_history_available,
      rdas_level_available,
      !is.na(prior_outcome_change),
      !is.na(prior_rdas_satisfaction),
      !is.na(prior_rdas_cohesion)
    )

  rdas_level_rs_base <- make_att_riskset_v2(rdas_level_complete)

  export_proximity_sample(
    rdas_level_rs_base,
    paste0(prefix, "_D2_RDAS_level_complete"),
    state_var
  )

  D2a <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change |
      riskset_stratum,
    data = rdas_level_rs_base,
    name = paste0(prefix, "_D2a_RDAS_level_common_sample_baseline"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  D2b <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion |
      riskset_stratum,
    data = rdas_level_rs_base,
    name = paste0(prefix, "_D2b_plus_prior_RDAS_levels"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  rdas_level_rs <- rdas_level_rs_base %>%
    left_join(
      D2_support %>%
        select(
          landmark_id,
          D2_target_att_weight
        ),
      by = "landmark_id"
    ) %>%
    left_join(h1_D2w, by = "landmark_id") %>%
    left_join(h2_D2w, by = "landmark_id") %>%
    filter(
      !is.na(D2_target_att_weight),
      !is.na(H1_D2_weight_raw),
      !is.na(H2_D2_weight_raw)
    )

  w2 <- make_final_weight_variants(
    rdas_level_rs$D2_target_att_weight,
    rdas_level_rs$H1_D2_weight_raw *
      rdas_level_rs$H2_D2_weight_raw,
    rdas_level_rs$treated
  )

  rdas_level_rs <- rdas_level_rs %>%
    mutate(
      combined_weight_H2_raw = w2$raw,
      combined_weight_H2 = w2$trim_1_99,
      combined_weight_H2_tight = w2$trim_2p5_97p5
    )

  D2c <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion |
      riskset_stratum,
    data = rdas_level_rs,
    name = paste0(prefix, "_D2c_D2opportunity_target_IPHAW_finaltrim"),
    weights = ~combined_weight_H2,
    vcov = ~pidp
  )

  D2c_tight <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion |
      riskset_stratum,
    data = rdas_level_rs,
    name = paste0(prefix, "_D2c_D2opportunity_target_IPHAW_finaltrim_2p5_97p5"),
    weights = ~combined_weight_H2_tight,
    vcov = ~pidp
  )

  diag_H2_raw <- weight_stability_diagnostics(
    rdas_level_rs,
    "combined_weight_H2_raw",
    "D2c D2-opportunity-target ATT x H1 x H2 raw product"
  )

  diag_H2 <- weight_stability_diagnostics(
    rdas_level_rs,
    "combined_weight_H2",
    "D2c D2-opportunity-target ATT x H1 x H2 final-trimmed IPHAW"
  )

  write_csv_quiet(
    bind_rows(
      diag_H2_raw,
      diag_H2
    ),
    paste0(prefix, "_D2c_FINAL_WEIGHT_DIAGNOSTICS.csv"),
    subdir = "diagnostics"
  )

  # ---------------------------------------------------------------------------
  # D3: H1 + H2 + RDAS-change complete.
  # IPHAW target = H1 & H2 & H3 structural opportunity support.
  # ---------------------------------------------------------------------------

  D3_support <- broad_rs %>%
    filter(D3_opportunity) %>%
    make_att_riskset_v2() %>%
    rename(D3_target_att_weight = att_weight)

  h1_D3 <- estimate_ipha_stage(
    D3_support,
    availability_var = "psych_history_available",
    opportunity_var = "D3_opportunity",
    stage_name = paste0(prefix, "_H1_for_D3_target")
  )

  D3_support_H1 <- D3_support %>%
    mutate(
      D3_H2_conditional_support = psych_history_available
    )

  h2_D3 <- estimate_ipha_stage(
    D3_support_H1,
    availability_var = "rdas_level_available",
    opportunity_var = "D3_H2_conditional_support",
    stage_name = paste0(prefix, "_H2_for_D3_target"),
    enforce_availability_subset = FALSE
  )

  D3_support_H12 <- D3_support %>%
    mutate(
      D3_H3_conditional_support =
        psych_history_available & rdas_level_available
    )

  h3_D3 <- estimate_ipha_stage(
    D3_support_H12,
    availability_var = "rdas_change_available",
    opportunity_var = "D3_H3_conditional_support",
    stage_name = paste0(prefix, "_H3_RDAS_change_D3_target"),
    enforce_availability_subset = FALSE
  )

  h1_D3w <- h1_D3$weights %>%
    rename(H1_D3_weight_raw = ipha_weight_raw) %>%
    select(landmark_id, H1_D3_weight_raw)

  h2_D3w <- h2_D3$weights %>%
    rename(H2_D3_weight_raw = ipha_weight_raw) %>%
    select(landmark_id, H2_D3_weight_raw)

  h3_D3w <- h3_D3$weights %>%
    rename(H3_D3_weight_raw = ipha_weight_raw) %>%
    select(landmark_id, H3_D3_weight_raw)

  rdas_change_complete <- broad_rs %>%
    filter(
      psych_history_available,
      rdas_level_available,
      rdas_change_available,
      !is.na(prior_outcome_change),
      !is.na(prior_rdas_satisfaction),
      !is.na(prior_rdas_cohesion),
      !is.na(prior_rdas_sat_change),
      !is.na(prior_rdas_coh_change)
    )

  rdas_change_rs_base <- make_att_riskset_v2(rdas_change_complete)

  export_proximity_sample(
    rdas_change_rs_base,
    paste0(prefix, "_D3_RDAS_change_complete"),
    state_var
  )

  D3a <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion |
      riskset_stratum,
    data = rdas_change_rs_base,
    name = paste0(prefix, "_D3a_RDAS_change_common_sample_baseline"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  D3b <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion +
      prior_rdas_sat_change +
      prior_rdas_coh_change |
      riskset_stratum,
    data = rdas_change_rs_base,
    name = paste0(prefix, "_D3b_plus_prior_RDAS_changes"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  rdas_change_rs <- rdas_change_rs_base %>%
    left_join(
      D3_support %>%
        select(
          landmark_id,
          D3_target_att_weight
        ),
      by = "landmark_id"
    ) %>%
    left_join(h1_D3w, by = "landmark_id") %>%
    left_join(h2_D3w, by = "landmark_id") %>%
    left_join(h3_D3w, by = "landmark_id") %>%
    filter(
      !is.na(D3_target_att_weight),
      !is.na(H1_D3_weight_raw),
      !is.na(H2_D3_weight_raw),
      !is.na(H3_D3_weight_raw)
    )

  w3 <- make_final_weight_variants(
    rdas_change_rs$D3_target_att_weight,
    rdas_change_rs$H1_D3_weight_raw *
      rdas_change_rs$H2_D3_weight_raw *
      rdas_change_rs$H3_D3_weight_raw,
    rdas_change_rs$treated
  )

  rdas_change_rs <- rdas_change_rs %>%
    mutate(
      combined_weight_H3_raw = w3$raw,
      combined_weight_H3 = w3$trim_1_99,
      combined_weight_H3_tight = w3$trim_2p5_97p5
    )

  D3c <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion +
      prior_rdas_sat_change +
      prior_rdas_coh_change |
      riskset_stratum,
    data = rdas_change_rs,
    name = paste0(prefix, "_D3c_D3opportunity_target_IPHAW_finaltrim"),
    weights = ~combined_weight_H3,
    vcov = ~pidp
  )

  D3c_tight <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion +
      prior_rdas_sat_change +
      prior_rdas_coh_change |
      riskset_stratum,
    data = rdas_change_rs,
    name = paste0(prefix, "_D3c_D3opportunity_target_IPHAW_finaltrim_2p5_97p5"),
    weights = ~combined_weight_H3_tight,
    vcov = ~pidp
  )

  diag_H3_raw <- weight_stability_diagnostics(
    rdas_change_rs,
    "combined_weight_H3_raw",
    "D3c D3-opportunity-target ATT x H1 x H2 x H3 raw product"
  )

  diag_H3 <- weight_stability_diagnostics(
    rdas_change_rs,
    "combined_weight_H3",
    "D3c D3-opportunity-target ATT x H1 x H2 x H3 final-trimmed sequential IPHAW"
  )

  write_csv_quiet(
    bind_rows(
      diag_H3_raw,
      diag_H3
    ),
    paste0(prefix, "_D3c_FINAL_WEIGHT_DIAGNOSTICS.csv"),
    subdir = "diagnostics"
  )

  # Direct joint-history selection model on exactly the D3 opportunity target.
  hjoint_D3 <- estimate_ipha_stage(
    D3_support,
    availability_var = "joint_history_available",
    opportunity_var = "D3_opportunity",

  )

  hjoint_D3w <- hjoint_D3$weights %>%
    rename(Hjoint_D3_weight_raw = ipha_weight_raw) %>%

  rdas_change_rs <- rdas_change_rs %>%
    left_join(
      hjoint_D3w,

    )

  wj <- make_final_weight_variants(
    rdas_change_rs$D3_target_att_weight,
    rdas_change_rs$Hjoint_D3_weight_raw,
    rdas_change_rs$treated
  )

  rdas_change_rs <- rdas_change_rs %>%
    mutate(
      combined_weight_joint_raw = wj$raw,
      combined_weight_joint = wj$trim_1_99
    )

  D3d_joint <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change +
      prior_rdas_satisfaction +
      prior_rdas_cohesion +
      prior_rdas_sat_change +
      prior_rdas_coh_change |
      riskset_stratum,
    data = rdas_change_rs,
    name = paste0(prefix, "_D3d_joint_history_D3target_IPHAW_finaltrim"),

    vcov = ~pidp
  )

  diag_joint_raw <- weight_stability_diagnostics(
    rdas_change_rs,
    "combined_weight_joint_raw",
    "D3d D3-opportunity-target ATT x joint-history raw product"
  )

  diag_joint <- weight_stability_diagnostics(
    rdas_change_rs,
    "combined_weight_joint",
    "D3d D3-opportunity-target ATT x joint-history final-trimmed IPHAW"
  )

  strict_models <- fit_strict_unambiguous_sensitivity(
    pair,
    prefix
  )

  weight_diagnostics <- bind_rows(
    diag_H1_raw,
    diag_H1,
    diag_H2_raw,
    diag_H2,
    diag_H3_raw,
    diag_H3,
    diag_joint_raw,

  ) %>%

      arm = arm,
      horizon = treated_state,

  support_diagnostics <- bind_rows(
    h1$support_diagnostics,
    h1_D2$support_diagnostics,
    h2_D2$support_diagnostics,
    h1_D3$support_diagnostics,
    h2_D3$support_diagnostics,
    h3_D3$support_diagnostics,
    hjoint_D3$support_diagnostics
  ) %>%
    mutate(
      arm = arm,
      horizon = treated_state,
      outcome = outcome
    )

  # ---------------------------------------------------------------------------
  # E fully common-sample accounting sequence.
  # ---------------------------------------------------------------------------
  # All four models use EXACTLY rdas_change_rs_base and the same complete-sample ATT
  # weights. This is descriptive explanatory accounting, NOT causal mediation.

  E0_common <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome |
      riskset_stratum,
    data = rdas_change_rs_base,
    name = paste0(prefix, "_E0_common_prior_level_only"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  E1_common <- safe_feols(
    outcome_change ~
      treated +
      prior_outcome +
      prior_outcome_change |
      riskset_stratum,
    data = rdas_change_rs_base,
    name = paste0(prefix, "_E1_common_plus_prior_psych_change"),
    weights = ~att_weight,
    vcov = ~pidp
  )

  common_models <- list(
    "E0 prior outcome level" = E0_common,
    "E1 + prior psychological change" = E1_common,
    "E2 + prior RDAS levels" = D3a,
    "E3 + prior RDAS changes" = D3b
  )

  common_accounting <- imap_dfr(
    common_models,
    function(mod, stage_name) {
      st <- extract_term_stats(mod, "treated")
      tibble(
        arm = arm,
        horizon = treated_state,
        outcome = outcome,
        stage = stage_name,
        estimate = st$estimate,
        std_error = st$std_error,
        conf_low = st$conf_low,
        conf_high = st$conf_high,
        p_value = st$p_value,
        nobs = st$nobs
      )
    }
  )

  if (nrow(common_accounting) > 0L) {
    base_beta <- common_accounting$estimate[1]

    common_accounting <- common_accounting %>%
      mutate(
        previous_estimate = lag(estimate),
        cumulative_attenuation_pct = if_else(
          !is.na(base_beta) &
            abs(base_beta) > 1e-8 &
            !is.na(estimate) &
            sign(estimate) == sign(base_beta),
          100 * (1 - estimate / base_beta),
          NA_real_
        ),
        incremental_attenuation_pct = if_else(
          !is.na(previous_estimate) &
            abs(previous_estimate) > 1e-8 &
            !is.na(estimate) &
            sign(estimate) == sign(previous_estimate),
          100 * (1 - estimate / previous_estimate),
          NA_real_
        ),
        accounting_note = paste(
          "Descriptive common-sample attenuation only;",
          "not a causal mediated share or mutually exclusive mechanism percentage."
        )
      )
  }

  attenuation <- bind_rows(
    attenuation_row(
      D1a,
      D1b,
      arm,
      treated_state,
      outcome,
      "add prior psychological change",
      "psychological-history-complete"
    ),
    attenuation_row(
      D2a,
      D2b,
      arm,
      treated_state,
      outcome,
      "add prior RDAS levels",
      "psychological-history + RDAS-level-complete"
    ),
    attenuation_row(
      D3a,
      D3b,
      arm,
      treated_state,
      outcome,
      "add prior RDAS changes",
      "psychological-history + RDAS-change-complete"
    )
  )

  sensitivity <- bind_rows(
    model_comparison_row(
      D0,
      D1a,
      arm,
      treated_state,
      outcome,
      "broad sample vs history-complete baseline: sample restriction"
    ),
    model_comparison_row(
      D1b,
      D1c,
      arm,
      treated_state,
      outcome,
      "D1b complete-history ATT vs H1-opportunity-target D1c IPHAW"
    ),
    model_comparison_row(
      D1c,
      D1c_tight,
      arm,
      treated_state,
      outcome,
      "D1c final-weight trimming 1/99 vs 2.5/97.5"
    ),
    model_comparison_row(
      D2b,
      D2c,
      arm,
      treated_state,
      outcome,
      "D2 complete-history ATT vs D2-opportunity-target sequential IPHAW"
    ),
    model_comparison_row(
      D2c,
      D2c_tight,
      arm,
      treated_state,
      outcome,
      "D2 final-weight trimming 1/99 vs 2.5/97.5"
    ),
    model_comparison_row(
      D3b,
      D3c,
      arm,
      treated_state,
      outcome,
      "D3 complete-history ATT vs D3-opportunity-target sequential IPHAW"
    ),
    model_comparison_row(
      D3c,
      D3c_tight,
      arm,
      treated_state,
      outcome,
      "D3 final-weight trimming 1/99 vs 2.5/97.5"
    ),
    model_comparison_row(
      D3c,
      D3d_joint,
      arm,
      treated_state,
      outcome,
      "D3 sequential vs direct joint history IPHAW on same D3 opportunity target"
    ),
    model_comparison_row(
      D1b,
      strict_models$D1b_strict,
      arm,
      treated_state,
      outcome,
      "primary D1b vs strict unambiguous-source D1b sensitivity"
    ),
    model_comparison_row(
      D2b,
      strict_models$D2b_strict,
      arm,
      treated_state,
      outcome,
      "D2b vs strict unambiguous-source D2b sensitivity"
    ),
    model_comparison_row(
      D3b,
      strict_models$D3b_strict,
      arm,
      treated_state,
      outcome,
      "D3b vs strict unambiguous-source D3b sensitivity"
    )
  )

  list(
    models = list(
      D0 = D0,
      D1a = D1a,
      D1b = D1b,
      D1c = D1c,
      D1c_tight = D1c_tight,
      D2a = D2a,
      D2b = D2b,
      D2c = D2c,
      D2c_tight = D2c_tight,
      D3a = D3a,
      D3b = D3b,
      D3c = D3c,
      D3c_tight = D3c_tight,
      D3d_joint = D3d_joint,
      D1b_strict = strict_models$D1b_strict,
      D2b_strict = strict_models$D2b_strict,
      D3b_strict = strict_models$D3b_strict
    ),
    attenuation = attenuation,
    common_accounting = common_accounting,
    sensitivity = sensitivity,
    weight_diagnostics = weight_diagnostics,
    support_diagnostics = support_diagnostics,
    opportunity_diagnostics = opportunity_diagnostics,
    availability_patterns = availability_pattern,
    strict_history_counts = strict_models$counts,

    # Retained for preregistered D1b-vs-D1c respondent-cluster bootstrap/BCa.
    broad_data = broad_rs,
    history_data = history_rs_base,

    n_broad = nrow(broad_rs),
    n_history = nrow(history_rs_base),
    n_rdas_level = nrow(rdas_level_rs_base),
    n_rdas_change = nrow(rdas_change_rs_base)
  )
}

# -----------------------------------------------------------------------------
# 11C-0. MANDATORY D1b-vs-D1c SELECTIVE-HISTORY SENSITIVITY
# -----------------------------------------------------------------------------
# D1b is confirmatory and targets observed history-complete cases.
# D1c is mandatory sensitivity and targets the H1 structural-opportunity
# population using opportunity-target ATT weights plus H1 IPHAW.
#
# Material disagreement:
#   (a) D1b and D1c point estimates reverse sign; OR
#   (b) the prespecified 95% respondent-cluster BCa CI for
#           beta_D1c - beta_D1b
#       excludes zero.
# If BCa is not numerically reliable, the respondent-cluster percentile CI is
# the prespecified fallback. D1c can qualify but never rescue D1b.
# -----------------------------------------------------------------------------

estimate_H1_weights_no_output <- function(dat) {

  D1_support <- dat %>%
    filter(H1_opportunity) %>%
    make_att_riskset_v2() %>%
    rename(D1_target_att_weight = att_weight)

  if (nrow(D1_support) == 0L) {
    return(tibble())
  }

  source <- D1_support %>%
    mutate(
      .H = as.integer(psych_history_available)
    ) %>%
    filter(
      !is.na(.H),
      !is.na(treated),
      !is.na(wave_n),
      !is.na(sex),
      !is.na(age),
      !is.na(cohab_duration_months),
      !is.na(prior_outcome),
      !is.na(n_prior_interviews),
      !is.na(n_prior_outcome_obs),
      !is.na(panel_tenure_months)
    )

  if (nrow(source) == 0L) {
    return(tibble())
  }

  if (length(unique(source$.H)) == 1L) {
    p_hat <- rep(mean(source$.H), nrow(source))
    hweight <- ifelse(source$.H == 1, 1, NA_real_)
  } else {
    fitted <- fit_bias_reduced_availability(source)

    if (is.null(fitted$model)) {
      return(tibble())
    }

    p_hat <- suppressWarnings(
      stats::predict(
        fitted$model,
        type = "response"
      )
    )

    p_hat <- pmin(
      pmax(p_hat, IPHAW_PROB_EPS),
      1 - IPHAW_PROB_EPS
    )

    numerator <- ave(
      source$.H,
      source$treated,
      FUN = function(z) mean(z == 1)
    )

    hweight <- ifelse(
      source$.H == 1,
      numerator / p_hat,
      NA_real_
    )
  }

  source %>%
    transmute(
      landmark_id,
      H1_available = .H,
      H1_p_hat = p_hat,
      H1_weight_raw = hweight,
      D1_target_att_weight
    )
}

estimate_D1b_D1c_from_broad <- function(broad_data) {

  broad_rs <- tryCatch(
    make_att_riskset_v2(broad_data),
    error = function(e) NULL
  )

  if (
    is.null(broad_rs) ||
      nrow(broad_rs) == 0L
  ) {
    return(
      c(
        beta_D1b = NA_real_,
        beta_D1c = NA_real_,
        difference = NA_real_
      )
    )
  }

  h1w <- estimate_H1_weights_no_output(
    broad_rs
  )

  if (nrow(h1w) == 0L) {
    return(
      c(
        beta_D1b = NA_real_,
        beta_D1c = NA_real_,
        difference = NA_real_
      )
    )
  }

  history_complete <- broad_rs %>%
    filter(
      psych_history_available,
      !is.na(prior_outcome_change)
    )

  history_rs <- tryCatch(
    make_att_riskset_v2(history_complete),
    error = function(e) NULL
  )

  if (
    is.null(history_rs) ||
      nrow(history_rs) == 0L
  ) {
    return(
      c(
        beta_D1b = NA_real_,
        beta_D1c = NA_real_,
        difference = NA_real_
      )
    )
  }

  history_rs <- history_rs %>%
    left_join(
      h1w,
      by = "landmark_id"
    ) %>%
    filter(
      H1_available == 1,
      is.finite(H1_weight_raw),
      H1_weight_raw > 0,
      is.finite(D1_target_att_weight),
      D1_target_att_weight > 0
    )

  if (
    nrow(history_rs) == 0L ||
      length(unique(history_rs$treated)) < 2L
  ) {
    return(
      c(
        beta_D1b = NA_real_,
        beta_D1c = NA_real_,
        difference = NA_real_
      )
    )
  }

  w1 <- make_final_weight_variants(
    history_rs$D1_target_att_weight,
    history_rs$H1_weight_raw,
    history_rs$treated
  )

  history_rs <- history_rs %>%
    mutate(
      combined_weight_H1 = w1$trim_1_99
    ) %>%
    filter(
      is.finite(att_weight),
      att_weight > 0,
      is.finite(combined_weight_H1),
      combined_weight_H1 > 0
    )

  D1b_fit <- tryCatch(
    suppressWarnings(
      fixest::feols(
        outcome_change ~
          treated +
          prior_outcome +
          prior_outcome_change |
          riskset_stratum,
        data = history_rs,
        weights = ~att_weight,
        notes = FALSE
      )
    ),
    error = function(e) NULL
  )

  D1c_fit <- tryCatch(
    suppressWarnings(
      fixest::feols(
        outcome_change ~
          treated +
          prior_outcome +
          prior_outcome_change |
          riskset_stratum,
        data = history_rs,
        weights = ~combined_weight_H1,
        notes = FALSE
      )
    ),
    error = function(e) NULL
  )

  beta_b <- if (
    is.null(D1b_fit) ||
      !"treated" %in% names(coef(D1b_fit))
  ) {
    NA_real_
  } else {
    unname(coef(D1b_fit)["treated"])
  }

  beta_c <- if (
    is.null(D1c_fit) ||
      !"treated" %in% names(coef(D1c_fit))
  ) {
    NA_real_
  } else {
    unname(coef(D1c_fit)["treated"])
  }

  c(
    beta_D1b = beta_b,
    beta_D1c = beta_c,
    difference = ifelse(
      is.na(beta_b) | is.na(beta_c),
      NA_real_,
      beta_c - beta_b
    )
  )
}

bootstrap_one_D1b_D1c <- function(broad_data) {

  ids <- unique(
    broad_data$pidp[
      !is.na(broad_data$pidp)
    ]
  )

  if (length(ids) < 2L) {
    return(
      c(
        beta_D1b = NA_real_,
        beta_D1c = NA_real_,
        difference = NA_real_
      )
    )
  }

  drawn_ids <- sample(
    ids,
    size = length(ids),
    replace = TRUE
  )

  boot_dat <- purrr::map2_dfr(
    drawn_ids,
    seq_along(drawn_ids),
    function(id_value, draw_index) {

      broad_data %>%
        filter(pidp == id_value) %>%
        mutate(
          pidp_boot = draw_index,
          landmark_id = paste0(
            landmark_id,
            "__boot_",
            draw_index
          )
        )
    }
  )

  estimate_D1b_D1c_from_broad(
    boot_dat
  )
}

jackknife_D1b_D1c <- function(broad_data) {

  ids <- unique(
    broad_data$pidp[
      !is.na(broad_data$pidp)
    ]
  )

  purrr::map_dfr(
    seq_along(ids),
    function(i) {

      est <- estimate_D1b_D1c_from_broad(
        broad_data %>%
          filter(pidp != ids[i])
      )

      tibble(
        jackknife_index = i,
        omitted_pidp = ids[i],
        beta_D1b = unname(est["beta_D1b"]),
        beta_D1c = unname(est["beta_D1c"]),
        difference = unname(est["difference"])
      )
    }
  )
}

bca_interval <- function(
  observed,
  bootstrap_values,
  jackknife_values,
  ci_level = 0.95
) {

  boot <- bootstrap_values[
    is.finite(bootstrap_values) &
      !is.na(bootstrap_values)
  ]

  jack <- jackknife_values[
    is.finite(jackknife_values) &
      !is.na(jackknife_values)
  ]

  if (
    length(boot) < 20L ||
      length(jack) < 10L ||
      !is.finite(observed)
  ) {
    return(list(
      low = NA_real_,
      high = NA_real_,
      z0 = NA_real_,
      acceleration = NA_real_,
      valid = FALSE
    ))
  }

  less_prop <- mean(boot < observed)
  eps <- 0.5 / length(boot)
  less_prop <- min(
    max(less_prop, eps),
    1 - eps
  )
  z0 <- stats::qnorm(less_prop)

  jack_mean <- mean(jack)
  u <- jack_mean - jack
  denom <- 6 * (sum(u^2)^(3 / 2))

  acceleration <- if (
    is.finite(denom) && denom > 0
  ) {
    sum(u^3) / denom
  } else {
    NA_real_
  }

  if (
    !is.finite(z0) ||
      !is.finite(acceleration)
  ) {
    return(list(
      low = NA_real_,
      high = NA_real_,
      z0 = z0,
      acceleration = acceleration,
      valid = FALSE
    ))
  }

  alpha <- 1 - ci_level
  za <- stats::qnorm(
    c(alpha / 2, 1 - alpha / 2)
  )

  adj <- stats::pnorm(
    z0 +
      (z0 + za) /
        (1 - acceleration * (z0 + za))
  )

  if (
    any(!is.finite(adj)) ||
      any(adj <= 0) ||
      any(adj >= 1) ||
      adj[1] >= adj[2]
  ) {
    return(list(
      low = NA_real_,
      high = NA_real_,
      z0 = z0,
      acceleration = acceleration,
      valid = FALSE
    ))
  }

  q <- stats::quantile(
    boot,
    probs = adj,
    na.rm = TRUE,
    names = FALSE
  )

  list(
    low = q[1],
    high = q[2],
    z0 = z0,
    acceleration = acceleration,
    valid = TRUE
  )
}

run_confirmatory_ipha_disagreement <- function(
  result,
  outcome_label,
  seed
) {

  d1b <- extract_term_stats(
    result$models$D1b,
    "treated"
  )

  d1c <- extract_term_stats(
    result$models$D1c,
    "treated"
  )

  beta_b <- d1b$estimate[1]
  beta_c <- d1c$estimate[1]

  observed_difference <- if (
    is.na(beta_b) || is.na(beta_c)
  ) {
    NA_real_
  } else {
    beta_c - beta_b
  }

  outcome_sd <- stats::sd(
    result$history_data$outcome_change,
    na.rm = TRUE
  )

  observed_difference_sd <- if (
    is.na(observed_difference) ||
      !is.finite(outcome_sd) ||
      outcome_sd <= 0
  ) {
    NA_real_
  } else {
    observed_difference / outcome_sd
  }

  d1c_diag <- result$weight_diagnostics %>%
    filter(
      stage ==
        "D1c H1-opportunity-target ATT x H1 final-trimmed IPHAW"
    )

  d1c_ess_acceptable <- if (
    nrow(d1c_diag) == 0L ||
      any(is.na(d1c_diag$acceptable_for_robustness))
  ) {
    FALSE
  } else {
    all(d1c_diag$acceptable_for_robustness)
  }

  d1c_min_ess_ratio <- if (
    nrow(d1c_diag) == 0L ||
      all(is.na(d1c_diag$ess_ratio))
  ) {
    NA_real_
  } else {
    min(d1c_diag$ess_ratio, na.rm = TRUE)
  }

  sign_reversal <- if (
    is.na(beta_b) ||
      is.na(beta_c) ||
      beta_b == 0 ||
      beta_c == 0
  ) {
    FALSE
  } else {
    sign(beta_b) != sign(beta_c)
  }

  boot_reps <- tibble(
    replicate = integer(0),
    beta_D1b = numeric(0),
    beta_D1c = numeric(0),
    difference = numeric(0)
  )

  if (
    RUN_CONFIRMATORY_IPHAW_BOOTSTRAP &&
      !is.null(result$broad_data) &&
      nrow(result$broad_data) > 0L
  ) {

    set.seed(seed)

    boot_matrix <- t(
      replicate(
        CONFIRMATORY_IPHAW_BOOT_REPS,
        bootstrap_one_D1b_D1c(
          result$broad_data
        )
      )
    )

    boot_reps <- tibble(
      replicate = seq_len(nrow(boot_matrix)),
      beta_D1b = boot_matrix[, "beta_D1b"],
      beta_D1c = boot_matrix[, "beta_D1c"],
      difference = boot_matrix[, "difference"]
    )
  }

  valid_boot <- boot_reps %>%
    filter(
      is.finite(difference),
      !is.na(difference)
    )

  n_boot_requested <- if (
    RUN_CONFIRMATORY_IPHAW_BOOTSTRAP
  ) {
    CONFIRMATORY_IPHAW_BOOT_REPS
  } else {
    0L
  }

  n_boot_valid <- nrow(valid_boot)

  boot_success_ratio <- if (
    n_boot_requested > 0L
  ) {
    n_boot_valid / n_boot_requested
  } else {
    NA_real_
  }

  boot_reliable <-
    !is.na(boot_success_ratio) &&
    boot_success_ratio >= CONFIRMATORY_IPHAW_BOOT_MIN_SUCCESS

  alpha_boot <- 1 - CONFIRMATORY_IPHAW_BOOT_CI

  if (n_boot_valid >= 20L) {
    pct_q <- stats::quantile(
      valid_boot$difference,
      probs = c(
        alpha_boot / 2,
        1 - alpha_boot / 2
      ),
      na.rm = TRUE,
      names = FALSE
    )
    pct_low <- pct_q[1]
    pct_high <- pct_q[2]
  } else {
    pct_low <- NA_real_
    pct_high <- NA_real_
  }

  jack_reps <- tibble(
    jackknife_index = integer(0),
    omitted_pidp = numeric(0),
    beta_D1b = numeric(0),
    beta_D1c = numeric(0),
    difference = numeric(0)
  )

  if (
    RUN_CONFIRMATORY_IPHAW_BCA &&
      !is.null(result$broad_data) &&
      nrow(result$broad_data) > 0L
  ) {
    jack_reps <- jackknife_D1b_D1c(
      result$broad_data
    )
  }

  n_jack_requested <- if (
    RUN_CONFIRMATORY_IPHAW_BCA
  ) {
    n_distinct(result$broad_data$pidp)
  } else {
    0L
  }

  valid_jack <- jack_reps %>%
    filter(
      is.finite(difference),
      !is.na(difference)
    )

  n_jack_valid <- nrow(valid_jack)

  jack_success_ratio <- if (
    n_jack_requested > 0L
  ) {
    n_jack_valid / n_jack_requested
  } else {
    NA_real_
  }

  jack_reliable <-
    !is.na(jack_success_ratio) &&
    jack_success_ratio >= CONFIRMATORY_IPHAW_JACKKNIFE_MIN_SUCCESS

  bca <- if (
    RUN_CONFIRMATORY_IPHAW_BCA &&
      boot_reliable &&
      jack_reliable
  ) {
    bca_interval(
      observed = observed_difference,
      bootstrap_values = valid_boot$difference,
      jackknife_values = valid_jack$difference,
      ci_level = CONFIRMATORY_IPHAW_BOOT_CI
    )
  } else {
    list(
      low = NA_real_,
      high = NA_real_,
      z0 = NA_real_,
      acceleration = NA_real_,
      valid = FALSE
    )
  }

  bca_reliable <-
    isTRUE(bca$valid) &&
    boot_reliable &&
    jack_reliable

  ci_method_used <- case_when(
    bca_reliable ~ "BCa respondent-cluster bootstrap",
    boot_reliable & !is.na(pct_low) & !is.na(pct_high) ~
      "percentile respondent-cluster bootstrap fallback",
    TRUE ~ "no reliable bootstrap CI"
  )

  decision_ci_low <- if (bca_reliable) bca$low else pct_low
  decision_ci_high <- if (bca_reliable) bca$high else pct_high

  decision_ci_reliable <- if (bca_reliable) {
    TRUE
  } else {
    boot_reliable && !is.na(pct_low) && !is.na(pct_high)
  }

  decision_ci_excludes_zero <-
    decision_ci_reliable &&
    !is.na(decision_ci_low) &&
    !is.na(decision_ci_high) &&
    (
      decision_ci_low > 0 ||
        decision_ci_high < 0
    )

  material_disagreement <-
    d1c_ess_acceptable &&
    (
      sign_reversal ||
        decision_ci_excludes_zero
    )

  sensitivity_status <- case_when(
    is.na(beta_b) | is.na(beta_c) ~
      paste(
        "D1b or D1c estimate unavailable;",
        "sensitivity cannot be adjudicated"
      ),

    !d1c_ess_acceptable ~
      paste(
        "D1c has inadequate ESS under the prespecified >=.50 rule;",
        "selective-history robustness cannot be assessed reliably"
      ),

    material_disagreement ~
      paste(
        "MATERIAL D1b-D1c DISAGREEMENT: the confirmatory complete-case",
        "estimate is sensitive to selective history availability within the",
        "H1 structural-opportunity population; generalization is uncertain"
      ),

    !decision_ci_reliable ~
      paste(
        "D1c ESS is acceptable but neither the BCa nor percentile",
        "respondent-cluster interval is reliable; selective-history",
        "robustness is incompletely adjudicated"
      ),

    TRUE ~
      paste(
        "No prespecified material D1b-D1c disagreement detected;",
        "the confirmatory estimate is robust to observed predictors of",
        "history availability within the H1 structural-opportunity population",
        "under the IPHAW assumptions"
      )
  )

  summary_row <- tibble(
    outcome = outcome_label,
    beta_D1b = beta_b,
    beta_D1c = beta_c,
    beta_D1c_minus_D1b = observed_difference,
    difference_in_outcome_SD_units = observed_difference_sd,
    sign_reversal = sign_reversal,
    D1c_min_ESS_ratio = d1c_min_ess_ratio,
    D1c_ESS_acceptable = d1c_ess_acceptable,

    bootstrap_reps_requested = n_boot_requested,
    bootstrap_reps_valid = n_boot_valid,
    bootstrap_success_ratio = boot_success_ratio,
    bootstrap_reliable = boot_reliable,
    percentile_CI_low = pct_low,
    percentile_CI_high = pct_high,

    jackknife_reps_requested = n_jack_requested,
    jackknife_reps_valid = n_jack_valid,
    jackknife_success_ratio = jack_success_ratio,
    jackknife_reliable = jack_reliable,

    BCa_valid = bca_reliable,
    BCa_bias_correction_z0 = bca$z0,
    BCa_acceleration = bca$acceleration,
    BCa_CI_low = bca$low,
    BCa_CI_high = bca$high,

    decision_CI_method = ci_method_used,
    decision_CI_low = decision_ci_low,
    decision_CI_high = decision_ci_high,
    decision_CI_reliable = decision_ci_reliable,
    decision_CI_excludes_zero = decision_ci_excludes_zero,
    material_D1b_D1c_disagreement = material_disagreement,
    sensitivity_status = sensitivity_status
  )

  list(
    summary = summary_row,
    bootstrap_replicates = boot_reps,
    jackknife_replicates = jack_reps
  )
}

# -----------------------------------------------------------------------------
# 11D. C-TO-D DISTAL DIAGNOSTIC - NO MECHANICAL SIGNIFICANCE GATE
# -----------------------------------------------------------------------------
# B, C5b and D25-60 do NOT form a three-test OR gate. They address different
# estimands (level, trajectory interaction, recent change), so combining their
# p-values into a binary pass/fail rule would be statistically artificial.
#
# The PRIMARY temporal diagnostic is D_M D1b at 25-60m because it has the same
# outcome definition, covariate structure and risk-set framework as the
# confirmatory D_M D1b 1-12m contrast. B and C5b are contextual diagnostics.
#
# Interpretation is based on the magnitude/direction/CI sequence:
#   D1b 25-60m -> 13-24m -> 1-12m.
# No p<.05-any-of-three flag is created.

get_treated <- function(x, model_name = "D1b") {
  extract_term_stats(x$models[[model_name]], "treated")
}

B_ghq <- extract_term_stats(B1_GHQ_DISTAL, "distal_treated")
B_ls <- extract_term_stats(B1_LS_DISTAL, "distal_treated")
B_p_holm <- stats::p.adjust(c(B_ghq$p_value, B_ls$p_value), method = "holm")

# 12. ANALYSIS F: TEMPORALLY ORDERED NCRR12 TRIANGULATION
# -----------------------------------------------------------------------------
#
# F1:
#   marriage expectation at t -> psychological change from t to t+1
#
# This is LAT/non-coresident relationship evidence, not a direct replication
# of the cohabitation pathway.

lat_panel <- panel %>%
  filter(
    ncrr1 == 1,
    !is.na(ncrr12),
    !is.na(lat_relationship_start_ym),
    lat_relationship_start_ym <= interview_ym
  ) %>%
  mutate(
    lat_relationship_id = paste(pidp, lat_relationship_start_ym, sep = "_"),
    lat_duration_months = interview_ym - lat_relationship_start_ym
  ) %>%
  arrange(lat_relationship_id, interview_ym) %>%
  group_by(lat_relationship_id) %>%
  mutate(
    next_interview_ym = lead(interview_ym),
    next_ghq = lead(ghq),
    next_ls = lead(life_satisfaction),
    next_ncrr12 = lead(ncrr12),
    next_gap_months = next_interview_ym - interview_ym,
    next_ghq_change = if_else(
      !is.na(next_ghq) & !is.na(ghq) &
        next_gap_months >= PRIOR_PSYCH_MIN_GAP_MONTHS &
        next_gap_months <= PRIOR_PSYCH_MAX_GAP_MONTHS,
      next_ghq - ghq,
      NA_real_
    ),
    next_ls_change = if_else(
      !is.na(next_ls) & !is.na(life_satisfaction) &
        next_gap_months >= PRIOR_PSYCH_MIN_GAP_MONTHS &
        next_gap_months <= PRIOR_PSYCH_MAX_GAP_MONTHS,
      next_ls - life_satisfaction,
      NA_real_
    ),
    n_lat_observations = n()
  ) %>%
  ungroup() %>%
  filter(n_lat_observations >= 2) %>%
  mutate(
    ncrr12_factor = factor(ncrr12),
    ghq_z = zsafe(ghq),
    ls_z = zsafe(life_satisfaction),
    next_ghq_change_z = zsafe(next_ghq_change),
    next_ls_change_z = zsafe(next_ls_change)
  )

write_csv_quiet(
  lat_panel %>% count(ncrr12_factor, name = "n"),
  "F00_NCRR12_category_counts.csv"
)

F1_GHQ_EXPECTATION_TO_CHANGE <- safe_feols(

# 15. ROBUST ACTUAL MODEL N FROM SAVED .RDS FILES

model_rds_files <- list.files(
  file.path(OUT_DIR, "models"),
  pattern = "\\.rds$",
  full.names = TRUE,
  ignore.case = TRUE
)

extract_saved_model_n <- function(path) {
  model_name <- sub("\\.rds$", "", basename(path), ignore.case = TRUE)
  obj <- tryCatch(readRDS(path), error = function(e) e)

  if (inherits(obj, "error")) {
    return(tibble(
      model = model_name,
      nobs_actual = NA_integer_,
      status = paste0("readRDS_error: ", conditionMessage(obj))
    ))
  }

  n <- tryCatch(stats::nobs(obj), error = function(e) NA_integer_)

  tibble(
    model = model_name,
    nobs_actual = if (
      length(n) == 0L || is.na(n[1])
    ) NA_integer_ else as.integer(n[1]),
    status = if (
      length(n) == 0L || is.na(n[1])
    ) "nobs_unavailable" else "estimated"
  )
}

if (length(model_rds_files) == 0L) {
  actual_model_counts <- tibble(
    model = character(0),
    nobs_actual = integer(0),
    status = character(0)
  )
} else {
  actual_model_counts <- bind_rows(
    lapply(model_rds_files, extract_saved_model_n)
  ) %>%
    arrange(model)
}

write_csv_quiet(
  actual_model_counts,
  "99_actual_model_nobs.csv"
)
